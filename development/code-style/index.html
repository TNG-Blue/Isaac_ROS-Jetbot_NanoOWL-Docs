    Code Style Guide¶

    Coding standards and style conventions for Isaac ROS Jetbot NanoOWL project.

    🎯 Overview¶

    Consistent code style improves: - ✅ Readability: Easier to understand - ✅ Maintainability: Simpler to modify - ✅ Collaboration: Reduces friction in reviews - ✅ Quality: Catches errors early

    📋 General Principles¶

    Code Quality Standards¶

    1. Clarity over Cleverness: Write obvious code, not clever code
    2. DRY (Don't Repeat Yourself): Extract common logic
    3. KISS (Keep It Simple, Stupid): Simplest solution wins
    4. YAGNI (You Aren't Gonna Need It): Don't add unused features
    5. Separation of Concerns: One responsibility per module

    Naming Philosophy¶

    • Descriptive: calculate_distance() not calc_d()
    • Consistent: Stick to established patterns
    • Searchable: Avoid single-letter names (except loop counters)
    • Pronounceable: Easy to discuss in conversation

    🐍 Python Style¶

    PEP 8 Compliance¶

    ROS2 Node Style¶

    Python
    import rclpy
from rclpy.node import Node
from sensor_msgs.msg import Image
from std_msgs.msg import String


class MyNode(Node):
    """ROS2 node description.

    Subscribes to: /input_topic (Image)
    Publishes to: /output_topic (String)
    """

    def __init__(self):
        """Initialize node."""
        super().__init__('my_node')

        # Declare parameters
        self.declare_parameter('param_name', 'default_value')

        # Create subscribers
        self.subscription = self.create_subscription(
            Image,
            '/input_topic',
            self.callback,
            10
        )

        # Create publishers
        self.publisher = self.create_publisher(
            String,
            '/output_topic',
            10
        )

        # Create timers
        self.timer = self.create_timer(0.1, self.timer_callback)

        self.get_logger().info('Node initialized')

    def callback(self, msg: Image) -> None:
        """Process incoming image message.

        Args:
            msg: Image message from camera
        """
        self.get_logger().debug('Received image')
        # Process message
        result = String()
        result.data = "processed"
        self.publisher.publish(result)

    def timer_callback(self) -> None:
        """Periodic timer callback."""
        pass


def main(args=None):
    """Main entry point."""
    rclpy.init(args=args)
    node = MyNode()

    try:
        rclpy.spin(node)
    except KeyboardInterrupt:
        pass
    finally:
        node.destroy_node()
        rclpy.shutdown()


if __name__ == '__main__':
    main()

    ROS2 Node Style¶

    C++
    #include "jetbot_detection/object_detector.hpp"

namespace jetbot_detection
{

ObjectDetector::ObjectDetector(const rclcpp::NodeOptions & options)
: Node("object_detector", options)
{
  // Declare parameters
  this->declare_parameter<std::string>("model_path", "/models/yolov8n.engine");
  this->declare_parameter<double>("confidence_threshold", 0.5);

  // Get parameters
  model_path_ = this->get_parameter("model_path").as_string();
  confidence_threshold_ = this->get_parameter("confidence_threshold").as_double();

  // Create subscriber
  image_sub_ = this->create_subscription<sensor_msgs::msg::Image>(
    "/camera/image_raw",
    10,
    std::bind(&ObjectDetector::imageCallback, this, std::placeholders::_1));

  // Create publisher
  detection_pub_ = this->create_publisher<vision_msgs::msg::Detection2DArray>(
    "/detections",
    10);

  // Initialize engine
  if (!initializeEngine()) {
    RCLCPP_ERROR(this->get_logger(), "Failed to initialize TensorRT engine");
    throw std::runtime_error("Engine initialization failed");
  }

  RCLCPP_INFO(this->get_logger(), "Object detector initialized");
}

void ObjectDetector::imageCallback(const sensor_msgs::msg::Image::SharedPtr msg)
{
  RCLCPP_DEBUG(this->get_logger(), "Received image: %dx%d", msg->width, msg->height);

  // Process image
  auto detections = std::make_shared<vision_msgs::msg::Detection2DArray>();
  // ... processing logic

  detection_pub_->publish(*detections);
}

}  // namespace jetbot_detection

#include "rclcpp_components/register_node_macro.hpp"
RCLCPP_COMPONENTS_REGISTER_NODE(jetbot_detection::ObjectDetector)

    CUDA Kernel Style¶

    CUDA
    /**
 * @brief Preprocessing kernel for YUV to RGB conversion
 * @param input Input YUV image
 * @param output Output RGB image
 * @param width Image width
 * @param height Image height
 */
__global__ void yuvToRgbKernel(
    const uint8_t* __restrict__ input,
    uint8_t* __restrict__ output,
    int width,
    int height)
{
    int x = blockIdx.x * blockDim.x + threadIdx.x;
    int y = blockIdx.y * blockDim.y + threadIdx.y;

    if (x >= width || y >= height) {
        return;
    }

    int idx = y * width + x;

    // YUV to RGB conversion
    float Y = input[idx];
    float U = input[width * height + (y / 2) * (width / 2) + (x / 2)];
    float V = input[width * height + (width * height) / 4 + (y / 2) * (width / 2) + (x / 2)];

    float R = Y + 1.402f * (V - 128.0f);
    float G = Y - 0.344f * (U - 128.0f) - 0.714f * (V - 128.0f);
    float B = Y + 1.772f * (U - 128.0f);

    output[idx * 3 + 0] = static_cast<uint8_t>(fminf(fmaxf(R, 0.0f), 255.0f));
    output[idx * 3 + 1] = static_cast<uint8_t>(fminf(fmaxf(G, 0.0f), 255.0f));
    output[idx * 3 + 2] = static_cast<uint8_t>(fminf(fmaxf(B, 0.0f), 255.0f));
}
