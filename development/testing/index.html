    Testing Guide¶

    Comprehensive testing strategy for Isaac ROS Jetbot NanoOWL to ensure reliability and quality.

    🎯 Testing Philosophy¶

    Test Pyramid:

    Text Only
               /\
          /  \  E2E Tests (10%)
         /____\
        /      \  Integration Tests (30%)
       /________\
      /          \  Unit Tests (60%)
     /____________\

    Coverage Goals: - Unit Tests: 80%+ coverage - Integration Tests: All critical paths - E2E Tests: Major user workflows - Performance Tests: Baseline benchmarks

    🧪 Unit Testing¶

    Python Unit Tests (pytest)¶

    Python
    # tests/python/test_object_detector.py
import pytest
import numpy as np
from jetbot_detection import ObjectDetector

@pytest.fixture
def detector():
    """Fixture providing detector instance."""
    return ObjectDetector(model_path='models/yolov8n.engine')

@pytest.fixture
def sample_image():
    """Fixture providing test image."""
    return np.random.randint(0, 255, (640, 640, 3), dtype=np.uint8)

class TestObjectDetector:
    """Test suite for ObjectDetector class."""

    def test_initialization(self, detector):
        """Test detector initializes correctly."""
        assert detector is not None
        assert detector.model_path.exists()

    def test_detection_output_format(self, detector, sample_image):
        """Test detection returns correct format."""
        detections = detector.detect(sample_image)

        assert isinstance(detections, list)
        if detections:
            assert 'class' in detections[0]
            assert 'confidence' in detections[0]
            assert 'bbox' in detections[0]

    @pytest.mark.parametrize("confidence", [0.3, 0.5, 0.7, 0.9])
    def test_confidence_thresholds(self, detector, sample_image, confidence):
        """Test different confidence thresholds."""
        detector.confidence_threshold = confidence
        detections = detector.detect(sample_image)

        for det in detections:
            assert det['confidence'] >= confidence

    def test_empty_image_handling(self, detector):
        """Test handling of empty/invalid images."""
        with pytest.raises(ValueError):
            detector.detect(None)

        with pytest.raises(ValueError):
            detector.detect(np.array([]))

    @pytest.mark.slow
    def test_performance_benchmark(self, detector, sample_image, benchmark):
        """Benchmark detection performance."""
        result = benchmark(detector.detect, sample_image)
        # Should complete in < 50ms on Jetson Orin Nano
        assert benchmark.stats['mean'] < 0.05

    Run Python tests:

    Bash
    # All tests
pytest tests/python/

# With coverage
pytest tests/python/ --cov=jetbot_detection --cov-report=html

# Specific test
pytest tests/python/test_object_detector.py::TestObjectDetector::test_initialization

# Verbose output
pytest tests/python/ -v

# Stop on first failure
pytest tests/python/ -x

# Run marked tests
pytest tests/python/ -m slow

    C++ Unit Tests (gtest)¶

    C++
    // tests/cpp/test_slam_node.cpp
#include <gtest/gtest.h>
#include <rclcpp/rclcpp.hpp>
#include "jetbot_slam/slam_node.hpp"

class SlamNodeTest : public ::testing::Test {
protected:
    void SetUp() override {
        rclcpp::init(0, nullptr);
        node_ = std::make_shared<jetbot_slam::SlamNode>();
    }

    void TearDown() override {
        node_.reset();
        rclcpp::shutdown();
    }

    std::shared_ptr<jetbot_slam::SlamNode> node_;
};

TEST_F(SlamNodeTest, Initialization) {
    EXPECT_NE(node_, nullptr);
    EXPECT_STREQ(node_->get_name(), "slam_node");
}

TEST_F(SlamNodeTest, MapCreation) {
    // Create test scan
    auto scan = std::make_shared<sensor_msgs::msg::LaserScan>();
    scan->ranges.resize(360, 1.0);

    node_->processScan(scan);

    auto map = node_->getMap();
    EXPECT_GT(map.info.width, 0);
    EXPECT_GT(map.info.height, 0);
}

TEST_F(SlamNodeTest, LoopClosureDetection) {
    // Test loop closure
    EXPECT_TRUE(node_->enableLoopClosure(true));

    // Simulate repeated scans at same location
    auto scan = createTestScan();
    for (int i = 0; i < 10; ++i) {
        node_->processScan(scan);
    }

    EXPECT_TRUE(node_->hasDetectedLoopClosure());
}

TEST_F(SlamNodeTest, PerformanceBenchmark) {
    auto scan = createTestScan();

    auto start = std::chrono::high_resolution_clock::now();
    for (int i = 0; i < 100; ++i) {
        node_->processScan(scan);
    }
    auto end = std::chrono::high_resolution_clock::now();

    auto duration = std::chrono::duration_cast<std::chrono::milliseconds>(end - start);
    double avg_ms = duration.count() / 100.0;

    // Should average < 30ms per scan
    EXPECT_LT(avg_ms, 30.0);
}

    🔗 Integration Testing¶

    ROS2 Launch Testing¶

    Python
    # tests/integration/test_detection_pipeline.py
import pytest
import unittest
from launch import LaunchDescription
from launch_ros.actions import Node
from launch_testing.actions import ReadyToTest
import rclpy
from sensor_msgs.msg import Image
from vision_msgs.msg import Detection2DArray

@pytest.mark.launch_test
def generate_test_description():
    """Generate launch description for testing."""
    return LaunchDescription([
        Node(
            package='v4l2_camera',
            executable='v4l2_camera_node',
            name='camera'
        ),
        Node(
            package='jetbot_detection',
            executable='detection_node',
            name='detector'
        ),
        ReadyToTest()
    ])

class TestDetectionPipeline(unittest.TestCase):
    """Integration test for detection pipeline."""

    @classmethod
    def setUpClass(cls):
        rclpy.init()

    @classmethod
    def tearDownClass(cls):
        rclpy.shutdown()

    def setUp(self):
        self.node = rclpy.create_node('test_node')

    def tearDown(self):
        self.node.destroy_node()

    def test_camera_to_detection(self, launch_service, proc_info, proc_output):
        """Test complete pipeline from camera to detections."""
        received_image = False
        received_detection = False

        def image_callback(msg):
            nonlocal received_image
            received_image = True

        def detection_callback(msg):
            nonlocal received_detection
            received_detection = True
            self.assertGreater(len(msg.detections), 0)

        # Subscribe to topics
        image_sub = self.node.create_subscription(
            Image, '/camera/image_raw', image_callback, 10
        )
        detection_sub = self.node.create_subscription(
            Detection2DArray, '/detections', detection_callback, 10
        )

        # Wait for messages
        timeout = 10.0  # seconds
        start_time = self.node.get_clock().now()

        while not (received_image and received_detection):
            rclpy.spin_once(self.node, timeout_sec=0.1)
            if (self.node.get_clock().now() - start_time).nanoseconds / 1e9 > timeout:
                break

        self.assertTrue(received_image, "Did not receive camera image")
        self.assertTrue(received_detection, "Did not receive detections")

    Navigation Test¶

    Python
    # tests/e2e/test_autonomous_navigation.py
import pytest
import rclpy
from rclpy.action import ActionClient
from nav2_msgs.action import NavigateToPose
from geometry_msgs.msg import PoseStamped

class TestAutonomousNavigation:
    """E2E test for autonomous navigation."""

    @pytest.fixture(autouse=True)
    def setup(self):
        rclpy.init()
        self.node = rclpy.create_node('test_nav')
        self.nav_client = ActionClient(self.node, NavigateToPose, 'navigate_to_pose')
        yield
        self.node.destroy_node()
        rclpy.shutdown()

    def test_navigate_to_goal(self):
        """Test robot navigates to goal successfully."""
        # Wait for navigation server
        assert self.nav_client.wait_for_server(timeout_sec=10.0)

        # Create goal
        goal = NavigateToPose.Goal()
        goal.pose.header.frame_id = 'map'
        goal.pose.pose.position.x = 2.0
        goal.pose.pose.position.y = 1.0
        goal.pose.pose.orientation.w = 1.0

        # Send goal
        future = self.nav_client.send_goal_async(goal)
        rclpy.spin_until_future_complete(self.node, future, timeout_sec=60.0)

        goal_handle = future.result()
        assert goal_handle.accepted, "Goal was rejected"

        # Wait for result
        result_future = goal_handle.get_result_async()
        rclpy.spin_until_future_complete(self.node, result_future, timeout_sec=120.0)

        result = result_future.result()
        assert result.status == 4, f"Navigation failed with status {result.status}"

    Benchmark Suite¶

    Python
    # tests/performance/benchmark_detection.py
import time
import numpy as np
import pytest
from jetbot_detection import ObjectDetector

class TestPerformance:
    """Performance benchmarks for detection."""

    @pytest.fixture
    def detector(self):
        return ObjectDetector()

    @pytest.fixture
    def test_images(self):
        """Generate test images of various sizes."""
        sizes = [(416, 416), (640, 640), (1280, 720), (1920, 1080)]
        return {
            f"{w}x{h}": np.random.randint(0, 255, (h, w, 3), dtype=np.uint8)
            for w, h in sizes
        }

    def test_throughput(self, detector, test_images):
        """Measure detection throughput (FPS)."""
        image = test_images['640x640']
        iterations = 100

        start = time.time()
        for _ in range(iterations):
            detector.detect(image)
        duration = time.time() - start

        fps = iterations / duration
        print(f"Throughput: {fps:.2f} FPS")

        # Should achieve 30+ FPS on Jetson Orin Nano
        assert fps >= 30.0, f"Expected >= 30 FPS, got {fps:.2f}"

    def test_latency(self, detector, test_images):
        """Measure detection latency."""
        image = test_images['640x640']
        latencies = []

        # Warm-up
        for _ in range(10):
            detector.detect(image)

        # Measure
        for _ in range(100):
            start = time.perf_counter()
            detector.detect(image)
            latency = (time.perf_counter() - start) * 1000  # ms
            latencies.append(latency)

        avg_latency = np.mean(latencies)
        p95_latency = np.percentile(latencies, 95)

        print(f"Average latency: {avg_latency:.2f}ms")
        print(f"P95 latency: {p95_latency:.2f}ms")

        assert avg_latency < 35.0, f"Expected < 35ms, got {avg_latency:.2f}ms"
        assert p95_latency < 50.0, f"Expected P95 < 50ms, got {p95_latency:.2f}ms"

    @pytest.mark.parametrize("resolution", ["416x416", "640x640", "1280x720"])
    def test_resolution_scaling(self, detector, test_images, resolution):
        """Test performance across resolutions."""
        image = test_images[resolution]

        start = time.time()
        iterations = 50
        for _ in range(iterations):
            detector.detect(image)
        duration = time.time() - start

        fps = iterations / duration
        print(f"{resolution}: {fps:.2f} FPS")

    🤖 Continuous Testing¶

    GitHub Actions Workflow¶

    YAML
    # .github/workflows/test.yml
name: Tests

on: [push, pull_request]

jobs:
  test:
    runs-on: ubuntu-latest

    steps:
      - uses: actions/checkout@v3

      - name: Setup ROS2
        uses: ros-tooling/setup-ros@v0.6
        with:
          required-ros-distributions: humble

      - name: Install dependencies
        run: |
          rosdep update
          rosdep install --from-paths src --ignore-src -y

      - name: Build
        run: colcon build --symlink-install

      - name: Test
        run: colcon test --return-code-on-test-failure

      - name: Upload test results
        uses: actions/upload-artifact@v3
        if: always()
        with:
          name: test-results
          path: build/*/test_results/*.xml
