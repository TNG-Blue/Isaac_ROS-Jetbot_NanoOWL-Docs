    Complete Installation Guide¶

    Comprehensive installation instructions for Isaac ROS Jetbot NanoOWL on NVIDIA Jetson platforms.

    📋 System Requirements¶

    Minimum Requirements¶

    Component Specification
    Platform NVIDIA Jetson Nano (4GB)
    Storage 64GB microSD (Class 10)
    Power 5V/2A USB-C
    Memory 4GB RAM
    OS Ubuntu 20.04 + JetPack 4.6+

    Recommended Requirements¶

    Component Specification
    Platform NVIDIA Jetson Orin Nano Super (8GB)
    Storage 128GB NVMe SSD or microSD (UHS-I)
    Power 5V/4A USB-C or barrel jack
    Memory 8GB RAM
    OS Ubuntu 22.04 + JetPack 5.1.2+

    Hardware Compatibility

    Also supports: Jetson Xavier NX, Xavier AGX, AGX Orin. Performance scales with hardware capabilities.

    🔧 Method 1: Docker Installation (Recommended)¶

    Docker provides the fastest and most reliable installation with zero dependency conflicts.

    Prerequisites¶

    Bash
    # Update system
sudo apt update && sudo apt upgrade -y

# Install Docker (if not present)
curl -fsSL https://get.docker.com -o get-docker.sh
sudo sh get-docker.sh

# Add user to docker group
sudo usermod -aG docker $USER
newgrp docker

# Install Docker Compose
sudo apt install docker-compose -y

    Install Isaac ROS Jetbot¶

    Bash
    # Clone repository
cd ~
git clone --recursive https://github.com/TNG-Blue/Isaac_ROS-Jetbot_NanoOWL.git
cd Isaac_ROS-Jetbot_NanoOWL

# Build and start containers
docker-compose up -d --build

# Verify containers are running
docker-compose ps

    Configuration¶

    Edit docker-compose.yml to customize:

    YAML
    # docker-compose.yml
version: '3.8'

services:
  jetbot-core:
    image: tngblue/jetbot-nanoowl:latest
    runtime: nvidia  # GPU access
    privileged: true
    network_mode: host
    volumes:
      - ./config:/app/config
      - ./models:/app/models
      - /dev:/dev  # Hardware access
    environment:
      - ROS_DOMAIN_ID=0
      - DISPLAY=${DISPLAY}
      - NVIDIA_VISIBLE_DEVICES=all
      - NVIDIA_DRIVER_CAPABILITIES=all

    🛠️ Method 2: Native Installation¶

    Full native installation for maximum performance and development flexibility.

    Step 1: Install JetPack¶

    Bash
    # For Jetson Orin Nano (JetPack 5.1.2)
# Use NVIDIA SDK Manager from host PC
# Or flash with pre-built image from:
# https://developer.nvidia.com/embedded/jetpack

# Verify installation
cat /etc/nv_tegra_release
# Expected: R35 (release), REVISION: 4.1

    Step 2: Install ROS2 Humble¶

    Bash
    # Set locale
sudo apt update && sudo apt install locales
sudo locale-gen en_US en_US.UTF-8
sudo update-locale LC_ALL=en_US.UTF-8 LANG=en_US.UTF-8
export LANG=en_US.UTF-8

# Add ROS2 repository
sudo apt install software-properties-common
sudo add-apt-repository universe
sudo apt update && sudo apt install curl -y

sudo curl -sSL https://raw.githubusercontent.com/ros/rosdistro/master/ros.key \
  -o /usr/share/keyrings/ros-archive-keyring.gpg

echo "deb [arch=$(dpkg --print-architecture) signed-by=/usr/share/keyrings/ros-archive-keyring.gpg] \
  http://packages.ros.org/ros2/ubuntu $(. /etc/os-release && echo $UBUNTU_CODENAME) main" | \
  sudo tee /etc/apt/sources.list.d/ros2.list > /dev/null

# Install ROS2 Humble
sudo apt update
sudo apt install ros-humble-desktop -y

# Install development tools
sudo apt install \
  python3-colcon-common-extensions \
  python3-rosdep \
  python3-vcstool \
  build-essential \
  -y

# Initialize rosdep
sudo rosdep init
rosdep update

    Step 3: Install Isaac ROS¶

    Bash
    # Install Isaac ROS dependencies
sudo apt install \
  ros-humble-isaac-ros-nitros \
  ros-humble-isaac-ros-common \
  ros-humble-isaac-ros-dnn-inference \
  -y

# Install CUDA dependencies
sudo apt install \
  cuda-toolkit-11-8 \
  nvidia-tensorrt \
  python3-libnvinfer \
  python3-libnvinfer-dev \
  -y

    Step 4: Build Workspace¶

    Bash
    # Create workspace
mkdir -p ~/ros2_ws/src
cd ~/ros2_ws/src

# Clone main repository
git clone --recursive https://github.com/TNG-Blue/Isaac_ROS-Jetbot_NanoOWL.git

# Clone additional dependencies
vcs import < Isaac_ROS-Jetbot_NanoOWL/dependencies.repos

# Install Python dependencies
cd Isaac_ROS-Jetbot_NanoOWL
pip3 install -r requirements.txt

# Install ROS dependencies
cd ~/ros2_ws
rosdep install --from-paths src --ignore-src -r -y

# Build packages
colcon build --symlink-install --parallel-workers 2 \
  --cmake-args -DCMAKE_BUILD_TYPE=Release

# This takes 20-30 minutes on Jetson Orin Nano

    Step 5: Setup Environment¶

    Bash
    # Source workspace
source ~/ros2_ws/install/setup.bash

# Add to bashrc
echo "source ~/ros2_ws/install/setup.bash" >> ~/.bashrc
echo "export ROS_DOMAIN_ID=0" >> ~/.bashrc

# Setup GPU environment
echo "export CUDA_HOME=/usr/local/cuda" >> ~/.bashrc
echo "export PATH=\$CUDA_HOME/bin:\$PATH" >> ~/.bashrc
echo "export LD_LIBRARY_PATH=\$CUDA_HOME/lib64:\$LD_LIBRARY_PATH" >> ~/.bashrc

source ~/.bashrc

    ☸️ Method 3: Kubernetes Deployment¶

    Production-ready deployment with Kubernetes orchestration.

    Prerequisites¶

    Bash
    # Install K3s (lightweight Kubernetes)
curl -sfL https://get.k3s.io | sh -

# Verify installation
sudo k3s kubectl get nodes

# Install Helm
curl https://raw.githubusercontent.com/helm/helm/main/scripts/get-helm-3 | bash

    Deploy with Helm¶

    Bash
    # Add Helm repository
helm repo add tng-blue https://tng-blue.github.io/charts
helm repo update

# Create namespace
kubectl create namespace robotics

# Install Jetbot chart
helm install jetbot-nanoowl tng-blue/jetbot-nanoowl \
  --namespace robotics \
  --set image.tag=latest \
  --set nodeSelector.nvidia\.com/gpu=true \
  --set resources.limits.nvidia\.com/gpu=1

    Custom Configuration¶

    Create values.yaml:

    YAML
    # values.yaml
replicaCount: 1

image:
  repository: tngblue/jetbot-nanoowl
  tag: "latest"
  pullPolicy: IfNotPresent

resources:
  limits:
    nvidia.com/gpu: 1
    memory: 4Gi
  requests:
    memory: 2Gi

nodeSelector:
  nvidia.com/gpu: "true"

config:
  ros:
    domainId: 0
  detection:
    modelPath: /models/yolov8n.engine
    confidenceThreshold: 0.5
  navigation:
    maxSpeed: 0.5
    safetyDistance: 0.5

    Install with custom values:

    Bash
    helm install jetbot-nanoowl tng-blue/jetbot-nanoowl \
  --namespace robotics \
  -f values.yaml

    📦 Additional Components¶

    Install Optional Packages¶

    Bash
    # LIDAR support
sudo apt install ros-humble-rplidar-ros -y

# Stereo vision
sudo apt install ros-humble-stereo-image-proc -y

# GPS integration
sudo apt install ros-humble-nmea-navsat-driver -y

# Web interface
sudo apt install ros-humble-rosbridge-server -y

# Monitoring tools
sudo apt install ros-humble-rqt-* -y

    Download AI Models¶

    Bash
    cd ~/ros2_ws/src/Isaac_ROS-Jetbot_NanoOWL

# Download YOLOv8 models
./scripts/download_models.sh

# Models downloaded to: models/
# - yolov8n.engine (Nano)
# - yolov8s.engine (Small)
# - yolov8m.engine (Medium)

    Install Web Dashboard¶

    Bash
    # Install Node.js
curl -fsSL https://deb.nodesource.com/setup_18.x | sudo -E bash -
sudo apt install nodejs -y

# Build web interface
cd ~/ros2_ws/src/Isaac_ROS-Jetbot_NanoOWL/web
npm install
npm run build

# Start web server
npm start

    🔒 Security Configuration¶

    Setup Firewall¶

    Bash
    # Install UFW
sudo apt install ufw -y

# Allow SSH
sudo ufw allow 22/tcp

# Allow ROS2 DDS ports
sudo ufw allow 7400:7500/udp
sudo ufw allow 7400:7500/tcp

# Allow web interface
sudo ufw allow 8080/tcp

# Enable firewall
sudo ufw enable

    Setup User Permissions¶

    Bash
    # Add user to required groups
sudo usermod -aG video $USER
sudo usermod -aG i2c $USER
sudo usermod -aG gpio $USER
sudo usermod -aG dialout $USER

# Apply changes
newgrp video

    🎛️ Performance Tuning¶

    Power Mode Configuration¶

    Bash
    # Jetson Orin Nano - Maximum performance
sudo nvpmodel -m 0  # 25W mode
sudo jetson_clocks  # Lock clocks to maximum

# View available modes
sudo nvpmodel -q

# Monitor power usage
sudo tegrastats

    Storage Optimization¶

    Bash
    # Clean unnecessary packages
sudo apt autoremove -y
sudo apt clean

# Remove build artifacts
cd ~/ros2_ws
rm -rf build/ log/

# Rebuild with optimizations
colcon build --symlink-install --parallel-workers 2 \
  --cmake-args -DCMAKE_BUILD_TYPE=Release

    🔄 Update & Maintenance¶

    Update Software¶

    Bash
    # Update system packages
sudo apt update && sudo apt upgrade -y

# Update ROS2 packages
sudo apt install --only-upgrade ros-humble-*

# Rebuild workspace
cd ~/ros2_ws
colcon build --symlink-install

    Backup Configuration¶

    Bash
    # Backup config directory
tar -czf jetbot-config-$(date +%Y%m%d).tar.gz \
  ~/ros2_ws/src/Isaac_ROS-Jetbot_NanoOWL/config

# Backup models
tar -czf jetbot-models-$(date +%Y%m%d).tar.gz \
  ~/ros2_ws/src/Isaac_ROS-Jetbot_NanoOWL/models

    ❓ Troubleshooting¶

    Build fails with memory error
    Bash
    # Reduce parallel workers
colcon build --symlink-install --parallel-workers 1

# Or increase swap space
sudo fallocate -l 4G /swapfile
sudo chmod 600 /swapfile
sudo mkswap /swapfile
sudo swapon /swapfile
    CUDA not found
    Bash
    # Add CUDA to PATH
export CUDA_HOME=/usr/local/cuda
export PATH=$CUDA_HOME/bin:$PATH
export LD_LIBRARY_PATH=$CUDA_HOME/lib64:$LD_LIBRARY_PATH

# Verify
nvcc --version
    ROS2 packages not found
    Bash
    # Source environment
source /opt/ros/humble/setup.bash
source ~/ros2_ws/install/setup.bash

# Verify
printenv | grep ROS
