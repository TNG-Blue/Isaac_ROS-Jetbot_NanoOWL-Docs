    Sensors Tutorial

    Sensor Setup¶

    Verify and configure all onboard sensors for optimal performance.

    🎯 Learning Objectives¶

    By the end of this tutorial, you will: - ✅ Verify IMU (Inertial Measurement Unit) calibration - ✅ Check encoder alignment and accuracy - ✅ Test LIDAR/distance sensors (if equipped) - ✅ Validate sensor fusion pipeline - ✅ Document sensor specifications

    Estimated Time: 40-50 minutes
    Difficulty Level: Beginner
    Prerequisites: Camera Calibration completed

    📋 Sensor Overview¶

    Your Jetbot includes several sensors:

    Primary Sensors:
  Camera: 1280×720 @ 30 FPS (calibrated)
  IMU: 6-axis (accel + gyro) @ 200 Hz
  Encoders: Left & right motor feedback
  Battery Monitor: Voltage & current sensing

Optional Sensors:
  LIDAR: 360° distance scanning
  Thermal: Infrared imaging
  GPS: Global positioning (if equipped)
  Compass: Magnetic heading

    🔩 Part 1: IMU Calibration¶

    1.1 Understand IMU¶

    6-Axis IMU contains:
  Accelerometer (3-axis):
    - Measures acceleration in X, Y, Z
    - Affected by: gravity, vibrations, mounting offset

  Gyroscope (3-axis):
    - Measures angular velocity
    - Affected by: bias, temperature drift, mounting tilt

    1.2 Access IMU Data¶

    # Check IMU node
ros2 node list | grep imu

# View raw IMU data
ros2 topic echo /imu/data --once

# Expected output format:
# linear_acceleration:
#   x: 0.05  # Small offset from gravity
#   y: -0.02
#   z: 9.81  # Gravity (9.81 m/s²)
# angular_velocity:
#   x: 0.001 # Should be near zero when still
#   y: 0.002
#   z: 0.001

    1.3 IMU Bias Calibration¶

    # Create calibration script
python3 << 'EOF'
import rclpy
from sensor_msgs.msg import Imu
import numpy as np
import time

class IMUCalibration:
    def __init__(self):
        rclpy.init()
        self.node = rclpy.create_node('imu_calibration')
        self.subscription = self.node.create_subscription(
            Imu, '/imu/data', self.imu_callback, 100)

        self.accel_readings = []
        self.gyro_readings = []
        self.collecting = True
        self.start_time = time.time()

    def imu_callback(self, msg):
        if not self.collecting:
            return

        # Store readings
        self.accel_readings.append([
            msg.linear_acceleration.x,
            msg.linear_acceleration.y,
            msg.linear_acceleration.z
        ])

        self.gyro_readings.append([
            msg.angular_velocity.x,
            msg.angular_velocity.y,
            msg.angular_velocity.z
        ])

        # Collect for 5 seconds
        if time.time() - self.start_time > 5:
            self.collecting = False
            self.calculate_bias()
            rclpy.shutdown()

    def calculate_bias(self):
        accel_array = np.array(self.accel_readings)
        gyro_array = np.array(self.gyro_readings)

        # Calculate averages (biases)
        accel_bias = np.mean(accel_array, axis=0)
        gyro_bias = np.mean(gyro_array, axis=0)

        print("\n=== IMU Calibration Results ===")
        print("\nAccelerometer Bias (offset from gravity):")
        print(f"  X: {accel_bias[0]:.4f} m/s²")
        print(f"  Y: {accel_bias[1]:.4f} m/s²")
        print(f"  Z: {accel_bias[2]:.4f} m/s² (should be ~9.81)")

        print("\nGyroscope Bias (should be near zero):")
        print(f"  X: {gyro_bias[0]:.4f} rad/s")
        print(f"  Y: {gyro_bias[1]:.4f} rad/s")
        print(f"  Z: {gyro_bias[2]:.4f} rad/s")

        # Diagnostic
        print("\n=== Diagnostics ===")
        accel_mag = np.linalg.norm(accel_bias)
        if 9.5 < accel_mag < 10.1:
            print("✓ Accelerometer gravity magnitude OK")
        else:
            print(f"✗ Warning: Gravity magnitude {accel_mag:.2f} (expected ~9.81)")

        if np.linalg.norm(gyro_bias) < 0.05:
            print("✓ Gyroscope bias acceptable")
        else:
            print("✗ Warning: Gyroscope bias too high")

if __name__ == '__main__':
    print("IMU Calibration: Keep robot STILL for 5 seconds...")
    print("Place robot on level surface, don't move it.")
    time.sleep(2)

    calib = IMUCalibration()
    rclpy.spin(calib.node)
EOF

# Run calibration
python3 imu_calibration.py

    2.2 Encoder Alignment Test¶

    # Create test script
python3 << 'EOF'
import rclpy
from geometry_msgs.msg import Twist
from jetbot_interfaces.msg import MotorState
import time

class EncoderTest:
    def __init__(self):
        rclpy.init()
        self.node = rclpy.create_node('encoder_test')

        self.motor_pub = self.node.create_publisher(Twist, '/cmd_vel', 10)
        self.motor_sub = self.node.create_subscription(
            MotorState, '/motor/state', self.motor_callback, 10)

        self.motor_state = None

    def motor_callback(self, msg):
        self.motor_state = msg

    def test_forward(self):
        print("\n=== Testing Forward Motion ===")
        twist = Twist()
        twist.linear.x = 0.2  # 0.2 m/s forward

        print("Sending forward command for 5 seconds...")
        start_pos_left = 0
        start_pos_right = 0

        for i in range(50):  # 5 seconds at 100 Hz
            self.motor_pub.publish(twist)
            if self.motor_state:
                if i == 0:
                    start_pos_left = self.motor_state.left_position
                    start_pos_right = self.motor_state.right_position
                if i == 49:
                    dist_left = self.motor_state.left_position - start_pos_left
                    dist_right = self.motor_state.right_position - start_pos_right

                    print(f"Left distance: {dist_left:.3f} m")
                    print(f"Right distance: {dist_right:.3f} m")
                    print(f"Difference: {abs(dist_left - dist_right):.3f} m")

                    if abs(dist_left - dist_right) < 0.05:  # < 5cm difference
                        print("✓ Encoders well aligned")
                    else:
                        print("✗ Encoders misaligned - may need adjustment")
            time.sleep(0.1)

        # Stop
        twist.linear.x = 0.0
        self.motor_pub.publish(twist)

if __name__ == '__main__':
    test = EncoderTest()
    test.test_forward()
    rclpy.shutdown()
EOF

python3 encoder_test.py

    2.3 Encoder Calibration¶

    If Encoders Misaligned:

1. Check Physical:
   - Motor mounts secure?
   - Encoder wheels clean?
   - No slip on wheel?

2. Adjust in Software:
   - Edit: ~/ros2_ws/src/jetbot_control/config/control_params.yaml
   - Motor constants:
     left_multiplier: 1.0
     right_multiplier: 1.0  # Adjust to balance
   - Recompile: colcon build

3. Retest after adjustment

    3.2 LIDAR Calibration¶

    # Create LIDAR calibration script
python3 << 'EOF'
import rclpy
from sensor_msgs.msg import LaserScan
import numpy as np

class LIDARCalibration:
    def __init__(self):
        rclpy.init()
        self.node = rclpy.create_node('lidar_calibration')
        self.subscription = self.node.create_subscription(
            LaserScan, '/scan', self.scan_callback, 10)
        self.scan_count = 0

    def scan_callback(self, msg):
        self.scan_count += 1

        # Filter valid ranges
        ranges = np.array(msg.ranges)
        valid_ranges = ranges[(ranges > msg.range_min) & (ranges < msg.range_max)]

        if self.scan_count % 10 == 0:  # Every 10th scan
            print(f"\n=== LIDAR Scan {self.scan_count} ===")
            print(f"Ranges: {len(valid_ranges)}/{len(ranges)} valid")
            print(f"Min distance: {np.min(valid_ranges):.3f} m")
            print(f"Max distance: {np.max(valid_ranges):.3f} m")
            print(f"Mean distance: {np.mean(valid_ranges):.3f} m")

            # Check for bad scans
            if len(valid_ranges) < 100:
                print("✗ Warning: Few valid ranges - check LIDAR connection")

        if self.scan_count >= 100:  # Stop after 100 scans
            rclpy.shutdown()

if __name__ == '__main__':
    print("LIDAR Calibration: Collecting 100 scans...")
    calib = LIDARCalibration()
    rclpy.spin(calib.node)
EOF

python3 lidar_calibration.py

# voltage: 7.8 V

# percentage: 65%

    4.2 Voltage Verification¶

    Battery Voltage Levels (2S Li-Po):

100%: 8.40V - Fully charged
 75%: 7.80V - Good
 50%: 7.30V - Acceptable
 25%: 6.80V - Low
  0%: 6.00V - Minimum (stop use)

Current Draw Expectations:
  Idle:     0.1A
  Movement: 0.5-1.0A
  Full load: 1.5-2.0A
  Max safe: 3.0A

    ✅ Part 5: Comprehensive Sensor Test¶

    5.1 Run Full Diagnostic¶

    # Create comprehensive test
python3 << 'EOF'
import rclpy
import time
from geometry_msgs.msg import Twist
from sensor_msgs.msg import Image, Imu
import numpy as np

class SensorTest:
    def __init__(self):
        rclpy.init()
        self.node = rclpy.create_node('sensor_test')

        # Subscribe to all sensors
        self.camera_sub = self.node.create_subscription(Image, '/camera/image_raw', 
                                                        self.camera_callback, 1)
        self.imu_sub = self.node.create_subscription(Imu, '/imu/data', 
                                                     self.imu_callback, 10)

        self.camera_ok = False
        self.imu_ok = False

    def camera_callback(self, msg):
        self.camera_ok = True

    def imu_callback(self, msg):
        self.imu_ok = True

    def run_test(self):
        print("\n=== SENSOR DIAGNOSTIC TEST ===\n")

        # Check ROS2 nodes
        print("1. Checking ROS2 Nodes...")
        topics = self.node.get_topic_names_and_types()
        print(f"   Active topics: {len(topics)}")

        # Wait for sensor data
        print("\n2. Waiting for sensor data (10 seconds)...")
        for i in range(100):
            rclpy.spin_once(self.node, timeout_ns=100_000_000)
            if i % 20 == 0:
                print(f"   ...{i/10:.1f}s")

        # Report results
        print("\n3. Results:")
        print(f"   Camera: {'✓' if self.camera_ok else '✗'}")
        print(f"   IMU: {'✓' if self.imu_ok else '✗'}")

        # Final status
        if self.camera_ok and self.imu_ok:
            print("\n✓ All sensors operational!")
        else:
            print("\n✗ Some sensors not responding")

        rclpy.shutdown()

if __name__ == '__main__':
    test = SensorTest()
    test.run_test()
EOF

python3 sensor_test.py

    📊 Sensor Specification Reference¶

    Camera (USB3):
  Resolution: 1280×720
  FPS: 30
  FOV: 90° horizontal
  Latency: ~33ms

IMU (6-axis):
  Accel Range: ±16G
  Gyro Range: ±2000 deg/s
  Sampling: 200 Hz
  Latency: ~10ms

Encoders:
  Type: Optical quadrature
  Resolution: 48 CPR per motor
  Max RPM: 300
  Accuracy: ±2% over distance

LIDAR (optional):
  Scanning Range: 0.15m - 12m
  Field of View: 360°
  Scanning Frequency: 10 Hz
  Angular Resolution: 0.36° per scan

Battery:
  Type: 2S Li-Po
  Nominal: 7.4V
  Capacity: 5000mAh
  Chemistry: LiPo

    ✅ Verification Checklist¶

    • IMU bias < 0.05 rad/s
    • Gravity magnitude 9.5-10.1 m/s²
    • Encoders aligned (< 5cm difference)
    • Camera streaming at 30 FPS
    • LIDAR (if equipped) returning valid ranges
    • Battery voltage > 7.4V
    • No sensor error messages
    • All topics publishing consistently
