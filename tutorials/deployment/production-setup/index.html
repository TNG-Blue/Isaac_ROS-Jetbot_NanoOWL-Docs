    Deployment Tutorial

    Production Setup¶

    Prepare your system for production deployment.

    🎯 Learning Objectives¶

    By the end of this tutorial, you will: - ✅ Harden system for production use - ✅ Optimize software for performance - ✅ Setup system logging and diagnostics - ✅ Configure automatic startup - ✅ Test system reliability

    Estimated Time: 2 hours
    Difficulty Level: Professional
    Prerequisites: Multi-Robot Setup completed

    🔧 Step 1: System Hardening¶

    1.1 Update System Software¶

    # Update all packages
sudo apt update
sudo apt upgrade -y

# Install security updates
sudo apt autoremove -y
sudo apt autoclean -y

# Install firewall
sudo apt install -y ufw

# Configure firewall
sudo ufw enable
sudo ufw allow 22/tcp     # SSH
sudo ufw allow 8888/tcp   # ROS2
sudo ufw allow 5353/udp   # mDNS

# Verify
sudo ufw status

    1.2 Configure System Resources¶

    # Disable unnecessary services
sudo systemctl disable bluetooth
sudo systemctl disable avahi-daemon  # mDNS broadcast
sudo systemctl disable ModemManager

# Enable swap (helps with memory pressure)
sudo fallocate -l 2G /swapfile
sudo chmod 600 /swapfile
sudo mkswap /swapfile
sudo swapon /swapfile

# Verify swap
free -h

    1.3 Setup Watchdog¶

    # Install watchdog service
sudo apt install -y watchdog

# Configure watchdog
sudo tee /etc/watchdog.conf > /dev/null << 'EOF'
watchdog-device = /dev/watchdog
max-load-1 = 24
max-load-5 = 18
max-load-15 = 12
realtime = yes
priority = 1
EOF

# Enable watchdog service
sudo systemctl enable watchdog
sudo systemctl start watchdog

    📊 Step 2: Performance Optimization¶

    2.1 Optimize Boot Parameters¶

    # Edit boot parameters
sudo nano /boot/extlinux/extlinux.conf

# Add performance optimizations:
# APPEND ... cgroup_enable=cpuset cgroup_enable=memory swapaccount=1 \
#   nvpmodel_t194_p3776=5 maxcpus=8 iommu.passthrough=1

# Apply
sudo update-extlinux

    2.2 Set CPU Governor¶

    # Install cpufrequtils
sudo apt install -y cpufrequtils

# Set performance governor
echo "GOVERNOR=performance" | sudo tee /etc/default/cpufrequtils

# Apply
sudo systemctl restart cpufrequtils

# Verify
cat /sys/devices/system/cpu/cpu0/cpufreq/scaling_governor

    2.3 Optimize ROS2 Middleware¶

    # Create optimized DDS configuration
cat > ~/.ros/dds_config.xml << 'EOF'
<?xml version="1.0" encoding="UTF-8"?>
<dds>
  <qos_profile name="optimization_profile">
    <participant>
      <rtps>
        <network>
          <send_socket_buffer_size>65536</send_socket_buffer_size>
          <listen_socket_buffer_size=65536</listen_socket_buffer_size>
        </network>
      </rtps>
    </participant>
  </qos_profile>
</dds>
EOF

# Set environment
export RMW_IMPLEMENTATION=rmw_cyclonedds_cpp
export CYCLONEDDS_URI=file://$HOME/.ros/dds_config.xml

    📝 Step 3: Logging Configuration¶

    3.1 Setup Structured Logging¶

    # Create log directory
mkdir -p /var/log/jetbot

# Create log rotation config
cat | sudo tee /etc/logrotate.d/jetbot-ros2 << 'EOF'
/var/log/jetbot/*.log {
    daily
    rotate 7
    compress
    delaycompress
    missingok
    notifempty
    create 0640 jetbot jetbot
    sharedscripts
}
EOF

    3.2 Configure ROS2 Logging¶

    # Create logging configuration
cat > ~/ros2_ws/log_config.yaml << 'EOF'
version: 1
disable_existing_loggers: false

formatters:
  standard:
    format: '[%(levelname)s][%(name)s] %(message)s'
  detailed:
    format: '%(asctime)s [%(levelname)s][%(name)s] %(message)s'

handlers:
  console:
    class: logging.StreamHandler
    level: WARNING
    formatter: standard
    stream: ext://sys.stdout

  file:
    class: logging.handlers.RotatingFileHandler
    level: DEBUG
    formatter: detailed
    filename: /var/log/jetbot/ros2.log
    maxBytes: 10485760  # 10MB
    backupCount: 5

root:
  level: INFO
  handlers: [console, file]
EOF

# Use in launch files
# Set environment variable RCUTILS_LOGGING_CONFIG_FILE

    🚀 Step 4: Automatic Startup¶

    4.1 Create Systemd Service¶

    # Create ROS2 service
sudo cat > /etc/systemd/system/jetbot-ros2.service << 'EOF'
[Unit]
Description=Isaac ROS Jetbot NanoOWL
After=network-online.target
Wants=network-online.target

[Service]
Type=simple
User=jetbot
WorkingDirectory=/home/jetbot/ros2_ws
Environment="ROS_DOMAIN_ID=0"
Environment="ROS_DISTRO=humble"
Environment="LD_LIBRARY_PATH=/opt/ros/humble/lib"
EnvironmentFile=/etc/environment

ExecStartPre=/bin/bash -c "source /opt/ros/humble/setup.bash && source install/setup.bash"
ExecStart=/bin/bash -c "source /opt/ros/humble/setup.bash && source install/setup.bash && ros2 launch jetbot_bringup main.launch.py"

Restart=always
RestartSec=10
StandardOutput=journal
StandardError=journal

[Install]
WantedBy=multi-user.target
EOF

# Enable service
sudo systemctl daemon-reload
sudo systemctl enable jetbot-ros2
sudo systemctl start jetbot-ros2

# Check status
sudo systemctl status jetbot-ros2

    4.2 Monitor Service Logs¶

    # View service logs
journalctl -u jetbot-ros2 -f

# View last 100 lines
journalctl -u jetbot-ros2 -n 100

# View since last hour
journalctl -u jetbot-ros2 --since "1 hour ago"

    🧪 Step 5: System Testing¶

    5.1 Run Reliability Tests¶

    # Create test suite
python3 << 'EOF'
import subprocess
import time
import sys

class ProductionTest:
    def __init__(self):
        self.tests_passed = 0
        self.tests_failed = 0

    def test_ros2_connectivity(self):
        """Test ROS2 network"""
        print("Testing ROS2 connectivity...")
        result = subprocess.run(['ros2', 'node', 'list'], capture_output=True)
        if result.returncode == 0:
            nodes = result.stdout.decode().strip().split('\n')
            print(f"  ✓ Found {len(nodes)} nodes")
            self.tests_passed += 1
        else:
            print("  ✗ ROS2 not responding")
            self.tests_failed += 1

    def test_sensor_fusion(self):
        """Test all sensors"""
        print("Testing sensor fusion...")
        result = subprocess.run(
            ['ros2', 'topic', 'hz', '/odom'],
            capture_output=True,
            timeout=5
        )
        if result.returncode == 0:
            print("  ✓ Sensor fusion active")
            self.tests_passed += 1
        else:
            print("  ✗ Sensor fusion failed")
            self.tests_failed += 1

    def test_motor_control(self):
        """Test motor response"""
        print("Testing motor control...")
        # Send command and verify response
        result = subprocess.run(
            ['ros2', 'service', 'list'],
            capture_output=True
        )
        if b'motor' in result.stdout:
            print("  ✓ Motor control available")
            self.tests_passed += 1
        else:
            print("  ✗ Motor control unavailable")
            self.tests_failed += 1

    def test_cpu_usage(self):
        """Test system resources"""
        print("Testing system resources...")
        result = subprocess.run(['free', '-h'], capture_output=True)
        output = result.stdout.decode()
        print(f"  {output.split(chr(10))[0]}")
        self.tests_passed += 1

    def run_all(self):
        print("\n=== Production Readiness Tests ===\n")
        self.test_ros2_connectivity()
        self.test_sensor_fusion()
        self.test_motor_control()
        self.test_cpu_usage()

        print(f"\n=== Test Results ===")
        print(f"Passed: {self.tests_passed}")
        print(f"Failed: {self.tests_failed}")

        if self.tests_failed == 0:
            print("\n✓ System ready for production!")
            return 0
        else:
            print("\n✗ Fix failures before deployment")
            return 1

if __name__ == '__main__':
    test = ProductionTest()
    sys.exit(test.run_all())
EOF

python3 production_test.py

    5.2 Stress Test¶

    # Create stress test
python3 << 'EOF'
import rclpy
from geometry_msgs.msg import Twist
import time
import threading

def stress_test():
    rclpy.init()
    node = rclpy.create_node('stress_test')
    pub = node.create_publisher(Twist, '/cmd_vel', 10)

    print("Starting stress test (30 seconds)...")
    start = time.time()
    count = 0

    while time.time() - start < 30:
        twist = Twist()
        twist.linear.x = 0.2
        twist.angular.z = 0.1
        pub.publish(twist)
        count += 1
        time.sleep(0.01)  # 100 Hz

    # Stop
    twist = Twist()
    pub.publish(twist)

    print(f"✓ Published {count} messages over 30 seconds")
    print(f"✓ Average rate: {count/30:.1f} Hz")

    rclpy.shutdown()

if __name__ == '__main__':
    stress_test()
EOF

python3 stress_test.py

    6.1 Create Monitoring Dashboard¶

sudo apt install -y prometheus-node-exporter

# Start monitoring
sudo systemctl start prometheus-node-exporter
sudo systemctl start grafana-server

    6.2 Setup Health Checks¶

    # Create health check script
cat > /usr/local/bin/jetbot-health-check.sh << 'EOF'
#!/bin/bash

echo "=== Jetbot Health Check ==="

# Check ROS2
if ros2 node list &>/dev/null; then
    echo "✓ ROS2 OK"
else
    echo "✗ ROS2 FAILED"
fi

# Check disk
DISK_USAGE=$(df / | awk 'NR==2 {print $5}' | sed 's/%//')
echo "✓ Disk: ${DISK_USAGE}% used"

# Check memory
MEM_USAGE=$(free | awk 'NR==2 {printf "%.0f", $3/$2 * 100.0}')
echo "✓ Memory: ${MEM_USAGE}% used"

# Check temperature
TEMP=$(cat /sys/class/thermal/thermal_zone0/temp)
TEMP_C=$((TEMP / 1000))
echo "✓ Temperature: ${TEMP_C}°C"

# Check services
for service in jetbot-ros2; do
    if systemctl is-active --quiet $service; then
        echo "✓ Service $service: Running"
    else
        echo "✗ Service $service: STOPPED"
    fi
done

echo ""
EOF

chmod +x /usr/local/bin/jetbot-health-check.sh

# Run health check
/usr/local/bin/jetbot-health-check.sh
