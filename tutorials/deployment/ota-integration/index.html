    OTA Integration¶

    Setup Over-The-Air firmware and software updates.

    🎯 Learning Objectives¶

    By the end of this tutorial, you will: - ✅ Setup OTA update infrastructure - ✅ Create firmware packages - ✅ Configure automatic updates - ✅ Implement rollback procedures - ✅ Monitor update deployments

    Estimated Time: 1.5 hours
    Difficulty Level: Professional
    Prerequisites: Production Setup completed

    🏗️ OTA Architecture¶

    YAML
    Update Server (Cloud):
  ├─ Firmware repository
  ├─ Version manifest
  └─ Update scheduler

Jetbot Devices:
  ├─ Update checker
  ├─ Download manager
  ├─ Verification system
  ├─ Installation
  └─ Rollback mechanism

    📦 Step 1: Setup Update Server¶

    1.1 Create Update Repository¶

    Bash
    # Create repository structure
mkdir -p ~/update_server/firmware/{stable,beta,experimental}
mkdir -p ~/update_server/software/{stable,beta,experimental}
mkdir -p ~/update_server/manifests

# Create version manifest
cat > ~/update_server/manifests/releases.json << 'EOF'
{
  "firmware": {

    1.2 Create Update Packages¶

    Bash
    # Build firmware package
cat > ~/create_firmware_package.sh << 'EOF'
#!/bin/bash

VERSION=${1:-"2.1.0"}
FIRMWARE_DIR="/tmp/firmware_build"

mkdir -p $FIRMWARE_DIR

# Copy firmware files
cp ~/firmware.bin $FIRMWARE_DIR/

# Create manifest
cat > $FIRMWARE_DIR/manifest.txt << 'MANIFEST'
VERSION=2.1.0
BUILD_DATE=$(date)
TARGET=jetbot_nanoowl
ARCHITECTURE=aarch64
MANIFEST

# Create package
tar -czf ~/update_server/firmware/stable/firmware_${VERSION}.tar.gz -C $FIRMWARE_DIR .

# Calculate checksum
sha256sum ~/update_server/firmware/stable/firmware_${VERSION}.tar.gz > \
  ~/update_server/firmware/stable/firmware_${VERSION}.sha256

echo "✓ Firmware package created: firmware_${VERSION}.tar.gz"
EOF

chmod +x ~/create_firmware_package.sh
./create_firmware_package.sh 2.1.0

    🔄 Step 2: OTA Update Client¶

    2.1 Create Update Checker¶

    Bash
    # Create update client
cat > ~/ros2_ws/src/jetbot_ota/src/ota_client.py << 'EOF'
import rclpy
from rclpy.node import Node
import requests
import os
import hashlib
import json
import threading
from pathlib import Path

class OTAClient(Node):
    def __init__(self):
        super().__init__('ota_client')

        # Configuration
        self.update_server = 'https://updates.example.com'
        self.manifest_url = f'{self.update_server}/manifests/releases.json'
        self.update_dir = os.path.expanduser('~/ota_updates')
        self.firmware_dir = '/boot/firmware'
        self.backup_dir = os.path.expanduser('~/firmware_backups')

        os.makedirs(self.update_dir, exist_ok=True)
        os.makedirs(self.backup_dir, exist_ok=True)

        # Current versions
        self.firmware_version = self.get_current_firmware_version()
        self.software_version = self.get_current_software_version()

        # Timer for periodic checks
        self.timer = self.create_timer(3600.0, self.check_for_updates)  # Check hourly

        self.get_logger().info(f"OTA Client initialized")
        self.get_logger().info(f"Current firmware: {self.firmware_version}")
        self.get_logger().info(f"Current software: {self.software_version}")

    def check_for_updates(self):
        """Check for available updates"""
        try:
            response = requests.get(self.manifest_url, timeout=10)
            manifest = response.json()

            # Check firmware updates
            latest_fw = manifest['firmware']['stable']
            if latest_fw > self.firmware_version:
                self.get_logger().warn(f"Firmware update available: {latest_fw}")
                self.schedule_update('firmware', latest_fw)

            # Check software updates
            latest_sw = manifest['software']['current']
            if latest_sw > self.software_version:
                self.get_logger().warn(f"Software update available: {latest_sw}")
                self.schedule_update('software', latest_sw)

        except Exception as e:
            self.get_logger().error(f"Failed to check updates: {e}")

    def schedule_update(self, update_type, version):
        """Schedule update in background"""
        thread = threading.Thread(
            target=self.perform_update,
            args=(update_type, version),
            daemon=True
        )
        thread.start()

    def perform_update(self, update_type, version):
        """Perform actual update"""
        try:
            if update_type == 'firmware':
                self.update_firmware(version)
            elif update_type == 'software':
                self.update_software(version)
        except Exception as e:
            self.get_logger().error(f"Update failed: {e}")
            self.rollback(update_type)

    def update_firmware(self, version):
        """Update firmware"""
        self.get_logger().info(f"Starting firmware update to {version}...")

        # Download
        url = f'{self.update_server}/firmware/stable/firmware_{version}.tar.gz'
        filepath = os.path.join(self.update_dir, f'firmware_{version}.tar.gz')

        if not self.download_file(url, filepath):
            raise Exception("Download failed")

        # Verify checksum
        if not self.verify_checksum(filepath, f'{filepath}.sha256'):
            raise Exception("Checksum verification failed")

        # Backup current
        self.backup_current_firmware()

        # Install
        self.install_firmware(filepath, version)

        self.get_logger().info(f"✓ Firmware updated to {version}")
        self.firmware_version = version

    def update_software(self, version):
        """Update software packages"""
        self.get_logger().info(f"Starting software update to {version}...")

        # Get manifest for this version
        response = requests.get(self.manifest_url)
        manifest = response.json()

        packages = manifest['software']['releases'][version]['packages']

        for package in packages:
            self.get_logger().info(f"Updating {package['name']}...")

            # Download package
            filepath = os.path.join(self.update_dir, f"{package['name']}_{version}.tar.gz")
            if not self.download_file(package['url'], filepath):
                raise Exception(f"Failed to download {package['name']}")

            # Extract and install
            os.system(f"tar -xzf {filepath} -C ~/ros2_ws/src/")

        # Rebuild
        os.system(f"cd ~/ros2_ws && colcon build --symlink-install")

        self.get_logger().info(f"✓ Software updated to {version}")
        self.software_version = version

    def download_file(self, url, filepath):
        """Download file with progress"""
        try:
            response = requests.get(url, stream=True, timeout=300)
            total_size = int(response.headers.get('content-length', 0))
            downloaded = 0

            with open(filepath, 'wb') as f:
                for chunk in response.iter_content(chunk_size=8192):
                    f.write(chunk)
                    downloaded += len(chunk)
                    if total_size:
                        percent = (downloaded / total_size) * 100
                        self.get_logger().info(f"Download: {percent:.1f}%")

            return True
        except Exception as e:
            self.get_logger().error(f"Download error: {e}")
            return False

    def verify_checksum(self, filepath, checksum_file):
        """Verify file checksum"""
        try:
            with open(checksum_file, 'r') as f:
                expected = f.read().split()[0]

            sha256_hash = hashlib.sha256()
            with open(filepath, 'rb') as f:
                for chunk in iter(lambda: f.read(4096), b''):
                    sha256_hash.update(chunk)

            actual = sha256_hash.hexdigest()

            if actual == expected:
                self.get_logger().info("✓ Checksum verified")
                return True
            else:
                self.get_logger().error("✗ Checksum mismatch")
                return False
        except Exception as e:
            self.get_logger().error(f"Checksum verification failed: {e}")
            return False

    def backup_current_firmware(self):
        """Backup current firmware"""
        import shutil
        backup_file = os.path.join(
            self.backup_dir,
            f'firmware_backup_{self.firmware_version}.bin'
        )
        shutil.copy2('/boot/firmware/firmware.bin', backup_file)
        self.get_logger().info(f"Firmware backed up to {backup_file}")

    def install_firmware(self, filepath, version):
        """Install firmware"""
        import subprocess
        try:
            # Extract
            extract_dir = os.path.join(self.update_dir, f'firmware_{version}')
            os.system(f"tar -xzf {filepath} -C {extract_dir}")

            # Install (requires sudo)
            firmware_bin = os.path.join(extract_dir, 'firmware.bin')
            os.system(f"sudo cp {firmware_bin} /boot/firmware/firmware.bin")

            # Reboot required
            self.get_logger().warn("Reboot required for firmware update")
        except Exception as e:
            raise Exception(f"Firmware installation failed: {e}")

    def rollback(self, update_type):
        """Rollback to previous version"""
        self.get_logger().error(f"Rolling back {update_type}...")

        if update_type == 'firmware':
            # Restore from backup
            backups = sorted(Path(self.backup_dir).glob('firmware_backup_*.bin'))
            if backups:
                os.system(f"sudo cp {backups[-1]} /boot/firmware/firmware.bin")
                self.get_logger().info("✓ Firmware rolled back")

    def get_current_firmware_version(self):
        """Get current firmware version"""
        try:
            with open('/boot/firmware/version.txt', 'r') as f:
                return f.read().strip()
        except:
            return "unknown"

    def get_current_software_version(self):
        """Get current software version"""
        try:
            import pkg_resources
            return pkg_resources.get_distribution('jetbot-core').version
        except:
            return "unknown"

def main():
    rclpy.init()
    client = OTAClient()
    rclpy.spin(client)
    rclpy.shutdown()

if __name__ == '__main__':
    main()
EOF

    📋 Step 4: Update Policies¶

    4.1 Create Update Schedule¶

    YAML
    Update Schedule:

Automatic Updates (off-peak):
  ├─ Check: Daily at 2:00 AM
  ├─ Download: If new version available
  ├─ Install: User confirmation required
  └─ Reboot: Within 24 hours

Manual Updates:
  ├─ User can trigger anytime
  ├─ With confirmation dialog
  └─ Immediate installation

    4.2 Version Management¶

    YAML
    Version Strategy:

Stable (2.1.0):
  - Production-ready
  - Thoroughly tested
  - Critical bug fixes

Beta (2.2.0-beta1):
  - Pre-release testing
  - Opt-in for early adopters
  - Breaking changes possible

Experimental:
  - Daily builds
  - For development only
  - Not recommended for production

    📊 Step 5: Monitoring Deployments¶

    5.1 Track Update Status¶

    Bash
    # Create deployment monitor
python3 << 'EOF'
import os
import json
from datetime import datetime

class UpdateMonitor:
    def __init__(self):
        self.log_file = '/var/log/jetbot/ota_updates.log'

    def log_update(self, robot_id, update_type, version, status):
        """Log update event"""
        event = {
            'timestamp': datetime.now().isoformat(),
            'robot_id': robot_id,
            'update_type': update_type,
            'version': version,
            'status': status  # started, completed, failed, rolled_back
        }

        with open(self.log_file, 'a') as f:
            f.write(json.dumps(event) + '\n')

    def get_deployment_status(self):
        """Get deployment statistics"""
        events = []
        with open(self.log_file, 'r') as f:
            for line in f:
                events.append(json.loads(line))

        completed = sum(1 for e in events if e['status'] == 'completed')
        failed = sum(1 for e in events if e['status'] == 'failed')
        in_progress = sum(1 for e in events if e['status'] == 'started' 
                          and not any(e2['robot_id'] == e['robot_id'] 
                                    and e2['status'] in ['completed', 'failed']
                                    and e2['timestamp'] > e['timestamp'] 
                                    for e2 in events))

        print(f"\n=== Deployment Status ===")
        print(f"Completed: {completed}")
        print(f"In Progress: {in_progress}")
        print(f"Failed: {failed}")

        return {
            'completed': completed,
            'in_progress': in_progress,
            'failed': failed
        }

monitor = UpdateMonitor()
monitor.get_deployment_status()
EOF
