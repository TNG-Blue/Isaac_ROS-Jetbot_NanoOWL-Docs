    Custom Models¶

    Train custom YOLOv8 detection models for your specific use case.

    🎯 Learning Objectives¶

    By the end of this tutorial, you will: - ✅ Prepare custom dataset for training - ✅ Train YOLOv8 model on your data - ✅ Quantize model for edge deployment - ✅ Deploy and test on Jetbot - ✅ Measure performance metrics

    Estimated Time: 2-3 hours
    Difficulty Level: Advanced
    Prerequisites: Sensor Setup completed

    📊 Dataset Preparation¶

    Step 1: Collect Images¶

    # Create dataset directory
mkdir -p ~/custom_dataset/{images,labels}

# Option A: Record with ROS2
ros2 bag record /camera/image_raw -o ~/camera_recording

# Extract images from bag
python3 << 'EOF'
import rosbag2
import cv2
from cv_bridge import CvBridge
import os

bridge = CvBridge()
bag_path = os.path.expanduser('~/camera_recording')
output_dir = os.path.expanduser('~/custom_dataset/images')

reader = rosbag2.SequentialReader()
reader.open(bag_path)

counter = 0
while reader.has_next():
    msg_type, data, _ = reader.read_next()
    if msg_type == 'sensor_msgs/Image':
        cv_image = bridge.imgmsg_to_cv2(data, desired_encoding='bgr8')
        filename = os.path.join(output_dir, f'image_{counter:06d}.jpg')
        cv2.imwrite(filename, cv_image)
        counter += 1

print(f"Extracted {counter} images")
EOF

# Option B: Manual capture
python3 << 'EOF'
import cv2
cap = cv2.VideoCapture(0)
counter = 0
output_dir = os.path.expanduser('~/custom_dataset/images')

while True:
    ret, frame = cap.read()
    if not ret:
        break

    cv2.imshow('Press SPACE to capture, Q to quit', frame)
    key = cv2.waitKey(1) & 0xFF

    if key == ord(' '):
        filename = os.path.join(output_dir, f'image_{counter:06d}.jpg')
        cv2.imwrite(filename, frame)
        counter += 1
        print(f"Captured: {counter}")

    elif key == ord('q'):
        break

cap.release()
cv2.destroyAllWindows()
print(f"Total captured: {counter}")
EOF

    Step 2: Annotate Images¶

    # Install annotation tool
pip3 install labelImg

# Launch labeling tool
labelImg ~/custom_dataset/images

# Annotation process:
# 1. Open directory: ~/custom_dataset/images
# 2. Change format to YOLO
# 3. Use keyboard:
#    - W: Draw box
#    - D: Next image
#    - A: Previous image
# 4. Save labels to ~/custom_dataset/labels

    Step 3: Create Dataset Structure¶

    # Organize into train/val/test splits
mkdir -p ~/custom_dataset/{train,val,test}/{images,labels}

# Split dataset (80% train, 10% val, 10% test)
python3 << 'EOF'
import os
import shutil
import random

source_images = os.path.expanduser('~/custom_dataset/images')
source_labels = os.path.expanduser('~/custom_dataset/labels')

# Get all images
images = sorted([f for f in os.listdir(source_images) if f.endswith('.jpg')])
random.shuffle(images)

# Calculate split
total = len(images)
train_count = int(total * 0.8)
val_count = int(total * 0.1)

print(f"Total: {total}, Train: {train_count}, Val: {val_count}, Test: {total-train_count-val_count}")

# Train split
for i, img in enumerate(images[:train_count]):
    shutil.copy(
        os.path.join(source_images, img),
        os.path.expanduser(f'~/custom_dataset/train/images/{img}')
    )
    label_file = img.replace('.jpg', '.txt')
    shutil.copy(
        os.path.join(source_labels, label_file),
        os.path.expanduser(f'~/custom_dataset/train/labels/{label_file}')
    )

# Val split
for img in images[train_count:train_count+val_count]:
    shutil.copy(
        os.path.join(source_images, img),
        os.path.expanduser(f'~/custom_dataset/val/images/{img}')
    )
    label_file = img.replace('.jpg', '.txt')
    shutil.copy(
        os.path.join(source_labels, label_file),
        os.path.expanduser(f'~/custom_dataset/val/labels/{label_file}')
    )

# Test split
for img in images[train_count+val_count:]:
    shutil.copy(
        os.path.join(source_images, img),
        os.path.expanduser(f'~/custom_dataset/test/images/{img}')
    )

print("✓ Dataset split complete")
EOF

    🧠 Model Training¶

    Step 1: Install YOLOv8¶

    # Install ultralytics
pip3 install ultralytics torch torchvision

# Verify installation
python3 -c "from ultralytics import YOLO; print('✓ YOLOv8 installed')"

    Step 2: Create Training Script¶

    # Create training script
python3 << 'EOF'
from ultralytics import YOLO
import os

# Create dataset.yaml
dataset_yaml = """
path: /home/jetbot/custom_dataset
train: train/images
val: val/images
test: test/images

nc: 2  # Number of classes (change as needed)
names: ['object1', 'object2']  # Class names
"""

yaml_path = os.path.expanduser('~/custom_dataset/dataset.yaml')
with open(yaml_path, 'w') as f:
    f.write(dataset_yaml)

# Load model (nano for Jetbot)
model = YOLO('yolov8n.pt')

# Train
results = model.train(
    data=yaml_path,
    epochs=100,
    imgsz=640,
    patience=20,  # Early stopping
    device=0,  # GPU device
    batch=16,
    optimizer='SGD',
    lr0=0.01,
    lrf=0.01,
    momentum=0.937,
    weight_decay=0.0005,
    warmup_epochs=3,
    warmup_momentum=0.8,
    warmup_bias_lr=0.1,
    close_mosaic=10,
    project=os.path.expanduser('~/yolov8_training'),
    name='custom_model_v1',
    exist_ok=False,
    pretrained=True,
    amp=True,  # Mixed precision
    mosaic=1.0,
    flipud=0.5,
    fliplr=0.5,
    hsv_h=0.015,
    hsv_s=0.7,
    hsv_v=0.4,
)

print("\n✓ Training complete!")
print(f"Best model: {results.save_dir}/weights/best.pt")
EOF

# Run training
python3 train_model.py

    Step 3: Monitor Training¶

    # In another terminal, monitor GPU usage
watch -n 1 'nvidia-smi'

# Monitor training progress
tensorboard --logdir ~/yolov8_training

# Expected metrics:
# - Loss decreases over epochs
# - mAP (mean Average Precision) increases
# - Validation accuracy improves

    📦 Model Quantization¶

    Step 1: Export Model¶

    # Export to ONNX
python3 << 'EOF'
from ultralytics import YOLO
import os

model_path = os.path.expanduser('~/yolov8_training/custom_model_v1/weights/best.pt')
model = YOLO(model_path)

# Export to ONNX
export_path = model.export(format='onnx', half=False, imgsz=640)
print(f"✓ Exported to: {export_path}")

# Export to TensorRT (for GPU optimization)
export_path_trt = model.export(format='engine', half=True, imgsz=640, device=0)
print(f"✓ Exported TensorRT: {export_path_trt}")
EOF

    Step 2: Quantize for Edge¶

    # Quantize to INT8 using TensorRT
python3 << 'EOF'
from ultralytics import YOLO
import os

model_path = os.path.expanduser('~/yolov8_training/custom_model_v1/weights/best.pt')
model = YOLO(model_path)

# Export to INT8 quantized model
export_path = model.export(
    format='engine',
    half=True,  # FP16
    int8=True,  # INT8 quantization
    imgsz=640,
    device=0,
    data=os.path.expanduser('~/custom_dataset/dataset.yaml')
)

print(f"✓ Quantized model: {export_path}")

# Check model size
original_size = os.path.getsize(model_path) / (1024**2)
quantized_size = os.path.getsize(export_path) / (1024**2)

print(f"\nModel Size:")
print(f"  Original: {original_size:.2f} MB")
print(f"  Quantized: {quantized_size:.2f} MB")
print(f"  Reduction: {(1-quantized_size/original_size)*100:.1f}%")
EOF

    🧪 Testing & Evaluation¶

    Step 1: Evaluate Model¶

    # Test model performance
python3 << 'EOF'
from ultralytics import YOLO
import os

model_path = os.path.expanduser('~/yolov8_training/custom_model_v1/weights/best.pt')
model = YOLO(model_path)

# Validate
metrics = model.val(
    data=os.path.expanduser('~/custom_dataset/dataset.yaml'),
    imgsz=640,
    conf=0.5,
    device=0
)

print("\n=== Validation Metrics ===")
print(f"mAP50: {metrics.box.map50:.3f}")
print(f"mAP50-95: {metrics.box.map:.3f}")
print(f"Precision: {metrics.box.p.mean():.3f}")
print(f"Recall: {metrics.box.r.mean():.3f}")
EOF

    Step 2: Test on Image¶

    # Test on single image
python3 << 'EOF'
from ultralytics import YOLO
import cv2

model_path = os.path.expanduser('~/yolov8_training/custom_model_v1/weights/best.pt')
model = YOLO(model_path)

# Run inference
results = model.predict(
    source='/path/to/test/image.jpg',
    conf=0.5,
    device=0,
    visualize=True
)

# Display results
for result in results:
    print(f"Detections: {len(result.boxes)}")
    for box in result.boxes:
        print(f"  Class: {result.names[int(box.cls)]}, Conf: {box.conf:.2f}")

    # Save result
    result.save(filename='prediction.jpg')
EOF

    🚀 Deploy to Jetbot¶

    Step 1: Copy Model¶

    # Copy quantized model to Jetbot
scp ~/yolov8_training/custom_model_v1/weights/best.engine jetbot@jetbot.local:~/models/

# Or copy ONNX model
scp ~/yolov8_training/custom_model_v1/weights/best.onnx jetbot@jetbot.local:~/models/

    Step 2: Update Detection Node¶

    # Edit detection configuration on Jetbot
cat > ~/ros2_ws/src/jetbot_detection/config/detection_params.yaml << 'EOF'
detection_node:
  ros__parameters:
    model_path: "/home/jetbot/models/best.engine"
    model_type: "yolov8"
    confidence_threshold: 0.5
    device: "cuda"
    input_topic: "/camera/image_raw"
    output_topic: "/detection/objects"
EOF

# Rebuild detection node
colcon build --packages-select jetbot_detection

# Launch with new model
ros2 launch jetbot_detection detection.launch.py

    Step 3: Performance Measurement¶

    # Measure inference speed
ros2 topic hz /detection/objects
# Expected: 20-35 FPS depending on model size

# Check resource usage
ros2 topic echo /diagnostics | grep -E "cpu|memory|gpu"

    📊 Performance Optimization¶

    Tips for Better Accuracy¶

    Training Improvements:
  ✓ More data (1000+ images)
  ✓ Diverse lighting conditions
  ✓ Various angles and distances
  ✓ Augmentation (flip, rotate, blur)
  ✓ More epochs (train longer)
  ✓ Larger batch size if memory allows

Inference Improvements:
  ✓ Use quantized model (INT8)
  ✓ Reduce input resolution (320 instead of 640)
  ✓ Batch inference (multiple images)
  ✓ Enable CUDA acceleration
  ✓ Half precision (FP16)

    ✅ Verification Checklist¶

    • Dataset prepared (train/val/test splits)
    • Training completed without errors
    • Validation metrics acceptable (mAP > 0.6)
    • Model quantized successfully
    • Model size < 50MB (for Jetbot)
    • Inference speed > 15 FPS
    • Model deployed to Jetbot
    • Detection working in ROS2 pipeline
