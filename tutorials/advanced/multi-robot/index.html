    Multi-Robot Setup¶

    Configure and operate multiple Jetbots in coordinated missions.

    🎯 Learning Objectives¶

    By the end of this tutorial, you will: - ✅ Configure multiple robots on ROS2 network - ✅ Setup robot identification and namespacing - ✅ Implement inter-robot communication - ✅ Coordinate multi-robot missions - ✅ Monitor fleet health

    Estimated Time: 1.5-2 hours
    Difficulty Level: Advanced
    Prerequisites: SLAM Configuration completed

    🌐 Network Architecture¶

    YAML
    Multi-Robot Network:

┌─────────────────────────────────┐
│    Wireless Network (WiFi)      │
│   Domain: 192.168.1.0/24        │
└────────┬──────────┬─────────────┘
         │          │
    ┌────▼────┐  ┌──▼─────┐
    │ Jetbot1 │  │ Jetbot2 │
    │ :11311  │  │ :11312  │
    └─────────┘  └─────────┘
         │          │
         └────┬─────┘
              │
       ┌──────▼──────┐
       │ Fleet Master │
       │   (Control)  │
       └──────────────┘

    🔧 Step 1: Robot Identification¶

    1.1 Assign Robot Names¶

    Bash
    # On Jetbot 1
ssh jetbot1@jetbot1.local

# Set hostname
hostnamectl set-hostname jetbot1
# Edit /etc/hosts to add: 127.0.0.1 jetbot1

# On Jetbot 2
ssh jetbot2@jetbot2.local

hostnamectl set-hostname jetbot2
# Edit /etc/hosts to add: 127.0.0.1 jetbot2

    1.2 Configure ROS2 Domain IDs¶

    Bash
    # Each robot needs unique DDS settings
# Or same domain ID with namespacing

# On Jetbot 1
cat >> ~/.bashrc << 'EOF'
export ROS_DOMAIN_ID=0
export ROS_NAMESPACE=jetbot1
EOF

# On Jetbot 2
cat >> ~/.bashrc << 'EOF'
export ROS_DOMAIN_ID=0
export ROS_NAMESPACE=jetbot2
EOF

source ~/.bashrc

    1.3 Verify Network¶

    Bash
    # On Jetbot 1
ping jetbot2.local
# Should respond

# On Jetbot 2
ping jetbot1.local
# Should respond

# Verify ROS2 network
ros2 node list
# Should see nodes from other robot with namespace prefix

    🚀 Step 2: Launch Multi-Robot System¶

    2.1 Create Fleet Launch File¶

    Bash
    # Create multi-robot launcher
cat > ~/ros2_ws/src/jetbot_bringup/launch/fleet.launch.py << 'EOF'
import os
from launch import LaunchDescription
from launch.actions import IncludeLaunchDescription
from launch.launch_description_sources import PythonLaunchDescriptionSource
from launch_ros.substitutions import FindPackageShare

def generate_launch_description():
    return LaunchDescription([
        # Robot 1
        IncludeLaunchDescription(
            PythonLaunchDescriptionSource(
                os.path.join(
                    FindPackageShare('jetbot_bringup'),
                    'launch',
                    'main.launch.py'
                )
            ),
            launch_arguments={
                'robot_name': 'jetbot1',
                'robot_id': '1',
            }.items()
        ),

        # Robot 2
        IncludeLaunchDescription(
            PythonLaunchDescriptionSource(
                os.path.join(
                    FindPackageShare('jetbot_bringup'),
                    'launch',
                    'main.launch.py'
                )
            ),
            launch_arguments={
                'robot_name': 'jetbot2',
                'robot_id': '2',
            }.items()
        ),

        # Fleet coordinator
        IncludeLaunchDescription(
            PythonLaunchDescriptionSource(
                os.path.join(
                    FindPackageShare('jetbot_bringup'),
                    'launch',
                    'fleet_coordinator.launch.py'
                )
            )
        ),
    ])
EOF

    2.2 Update Main Launch File¶

    Bash
    # Edit ~/ros2_ws/src/jetbot_bringup/launch/main.launch.py

# Add parameters
from launch.substitutions import LaunchConfiguration

robot_name = LaunchConfiguration('robot_name', default='jetbot1')
robot_id = LaunchConfiguration('robot_id', default='1')

# Apply namespace
declare_robot_name_cmd = DeclareLaunchArgument(
    'robot_name',
    default_value='jetbot1',
    description='Robot name for namespace'
)

# Update nodes with namespace
Node(
    package='jetbot_camera',
    executable='camera_node',
    name='camera',
    namespace=robot_name,
    parameters=[...]
)

    📡 Step 3: Inter-Robot Communication¶

    3.1 Create Fleet Communication Node¶

    Bash
    # Create coordinator
cat > ~/ros2_ws/src/jetbot_bringup/src/fleet_coordinator.py << 'EOF'
import rclpy
from rclpy.node import Node
from geometry_msgs.msg import Pose, PoseArray
from std_msgs.msg import String
import json

class FleetCoordinator(Node):
    def __init__(self):
        super().__init__('fleet_coordinator')

        # Subscribers for each robot pose
        self.pose_sub1 = self.create_subscription(
            Pose, '/jetbot1/pose', self.pose_callback1, 10)
        self.pose_sub2 = self.create_subscription(
            Pose, '/jetbot2/pose', self.pose_callback2, 10)

        # Publisher for fleet state
        self.fleet_state_pub = self.create_publisher(
            String, '/fleet/state', 10)

        # Robot positions
        self.robot_poses = {
            'jetbot1': None,
            'jetbot2': None
        }

        # Timer for periodic updates
        self.timer = self.create_timer(1.0, self.update_fleet_state)

    def pose_callback1(self, msg):
        self.robot_poses['jetbot1'] = msg

    def pose_callback2(self, msg):
        self.robot_poses['jetbot2'] = msg

    def update_fleet_state(self):
        # Create fleet state message
        fleet_state = {
            'timestamp': self.get_clock().now().to_msg(),
            'robots': {}
        }

        for name, pose in self.robot_poses.items():
            if pose:
                fleet_state['robots'][name] = {
                    'x': float(pose.position.x),
                    'y': float(pose.position.y),
                    'z': float(pose.position.z),
                }

        # Publish
        msg = String()
        msg.data = json.dumps(fleet_state, default=str)
        self.fleet_state_pub.publish(msg)

        self.get_logger().debug(f"Fleet state: {fleet_state}")

def main():
    rclpy.init()
    coordinator = FleetCoordinator()
    rclpy.spin(coordinator)
    rclpy.shutdown()

if __name__ == '__main__':
    main()
EOF

# Make executable
chmod +x ~/ros2_ws/src/jetbot_bringup/src/fleet_coordinator.py

    🎮 Step 4: Coordinated Missions¶

    4.1 Create Multi-Robot Mission¶

    Bash
    # Create mission planner
cat > ~/multi_robot_mission.py << 'EOF'
import rclpy
from geometry_msgs.msg import Twist
import time

class MultiRobotMission:
    def __init__(self):
        rclpy.init()
        self.node = rclpy.create_node('multi_robot_mission')

        # Publishers for each robot
        self.pub1 = self.node.create_publisher(Twist, '/jetbot1/cmd_vel', 10)
        self.pub2 = self.node.create_publisher(Twist, '/jetbot2/cmd_vel', 10)

    def move_robot(self, pub, linear_x, angular_z, duration):
        """Move robot for specified duration"""
        twist = Twist()
        twist.linear.x = linear_x
        twist.angular.z = angular_z

        for _ in range(int(duration * 10)):
            pub.publish(twist)
            time.sleep(0.1)

        # Stop
        twist.linear.x = 0.0
        twist.angular.z = 0.0
        pub.publish(twist)

    def parallel_patrol(self):
        """Both robots patrol in parallel"""
        print("Starting parallel patrol mission...")

        # Robot 1 moves forward
        self.move_robot(self.pub1, 0.2, 0.0, 3.0)

        # Robot 2 moves forward
        self.move_robot(self.pub2, 0.2, 0.0, 3.0)

        print("✓ Parallel patrol complete")

    def formation_movement(self):
        """Robots maintain formation"""
        print("Formation movement: Line formation...")

        # Start simultaneously
        for i in range(30):
            twist1 = Twist()
            twist1.linear.x = 0.2
            self.pub1.publish(twist1)

            twist2 = Twist()
            twist2.linear.x = 0.2
            self.pub2.publish(twist2)

            time.sleep(0.1)

        # Stop
        twist = Twist()
        self.pub1.publish(twist)
        self.pub2.publish(twist)

        print("✓ Formation complete")

    def leapfrog_pattern(self):
        """Robots alternate movement"""
        print("Leapfrog pattern...")

        # Robot 1 moves
        self.move_robot(self.pub1, 0.2, 0.0, 2.0)

        # Robot 2 moves
        self.move_robot(self.pub2, 0.2, 0.0, 2.0)

        # Robot 1 moves again
        self.move_robot(self.pub1, 0.2, 0.0, 2.0)

        print("✓ Leapfrog pattern complete")

def main():
    mission = MultiRobotMission()

    # Run missions
    mission.parallel_patrol()
    time.sleep(1.0)

    mission.formation_movement()
    time.sleep(1.0)

    mission.leapfrog_pattern()

    print("\n✓ All missions complete!")
    rclpy.shutdown()

if __name__ == '__main__':
    main()
EOF

# Run mission
python3 multi_robot_mission.py

    📊 Step 5: Fleet Monitoring¶

    5.1 Create Fleet Monitor¶

    Bash
    # Monitor dashboard
python3 << 'EOF'
import rclpy
from geometry_msgs.msg import Pose
import json

class FleetMonitor:
    def __init__(self):
        rclpy.init()
        self.node = rclpy.create_node('fleet_monitor')

        # Subscribe to each robot
        self.robots = {
            'jetbot1': None,
            'jetbot2': None
        }

        for robot_name in self.robots:
            self.node.create_subscription(
                Pose,
                f'/{robot_name}/pose',
                lambda msg, name=robot_name: self.update_robot(name, msg),
                10
            )

        # Timer for periodic display
        self.timer = self.node.create_timer(1.0, self.display_status)

    def update_robot(self, name, pose):
        self.robots[name] = pose

    def display_status(self):
        print("\n=== Fleet Status ===")
        for name, pose in self.robots.items():
            if pose:
                print(f"{name}: x={pose.position.x:.2f}, y={pose.position.y:.2f}, z={pose.position.z:.2f}")
            else:
                print(f"{name}: No data")

if __name__ == '__main__':
    monitor = FleetMonitor()
    rclpy.spin(monitor.node)
EOF

    ✅ Verification Checklist¶

    • All robots visible on network
    • ROS2 can communicate between robots
    • Each robot has unique namespace
    • Fleet coordinator running
    • Inter-robot topics visible
    • Multi-robot missions execute
    • Fleet monitor shows all robots
    • Synchronization works

    🐛 Troubleshooting¶

    Issue: Robots Can't See Each Other¶

    YAML
    Check:
  1. Network connectivity: ping jetbot2.local
  2. ROS_DOMAIN_ID: Should be same
  3. Firewall: May block multicast
  4. DDS settings: Check ~/.ros/dds_config.xml

    Issue: Topics Not Syncing¶

    YAML
    Solutions:
  1. Verify namespace is applied
  2. Check topic names: ros2 topic list
  3. Ensure subscribers active
  4. Check message types match
