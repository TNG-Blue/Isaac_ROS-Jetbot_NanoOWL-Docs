    Microcontroller Integration¶

    Documentation for Arduino Portenta H7 and OpenMV RT1062 integration with Isaac ROS Jetbot NanoOWL.

    🎯 Overview¶

    The Isaac ROS Jetbot NanoOWL platform uses dedicated microcontrollers for real-time motor control, sensor interfacing, and low-level hardware management.

    Microcontroller Roles: - 🖥️ Arduino Portenta H7: Primary motor control, I2C hub, real-time tasks - 📷 OpenMV RT1062: Vision pre-processing, QR/barcode scanning, blob detection - 🔄 Communication: UART, I2C, CAN bus, USB to Jetson Orin Nano Super

    🖥️ Arduino Portenta H7¶

    Overview¶

    The Arduino Portenta H7 serves as the central microcontroller for real-time control tasks and hardware interfacing.

    Key Features: 

    Processors:
  Main Core: STM32H747XI (Cortex-M7 @ 480 MHz)
  Secondary Core: Cortex-M4 @ 240 MHz (optional)

Memory:
  Flash: 2 MB
  RAM: 1 MB (shared between cores)
  QSPI Flash: 16 MB external

Interfaces:
  USB: USB-C (OTG, Host, Device)
  Ethernet: 10/100 Mbps
  CAN: 2x CAN-FD ports
  UART: 10x serial ports
  I2C: 4x buses
  SPI: 6x buses
  PWM: 22 channels
  ADC: 16x 16-bit channels

WiFi/Bluetooth: Via Vision Shield or separate module
Power: 5V via USB-C or VIN (6-27V)

    Pin Mapping¶

    Arduino Portenta H7 Pinout
──────────────────────────────────────

Power Pins:
  VIN    : 6-27V input (buck converter to 5V/3.3V)
  5V     : 5V output (500mA max)
  3V3    : 3.3V output (800mA max)
  GND    : Ground

Digital I/O:
  D0-D6  : GPIO (3.3V logic)

PWM Outputs:
  PWM0-PWM9 : 16-bit resolution, up to 200 kHz

Analog Inputs:
  A0-A7  : 16-bit ADC (0-3.3V)

Communication:
  TX/RX  : UART0 (Serial1)
  SDA/SCL: I2C0 (Wire)
  MISO/MOSI/SCK/CS : SPI0
  CANRX/CANTX : CAN bus

High-Speed:
  LVDS+/- : Display interface (optional)
  CAM_D+/- : MIPI camera (optional)

    /**
 * @file main_m7.cpp
 * @brief Cortex-M7 main core - Real-time motor control
 */

#include <Arduino.h>
#include <RPC.h>  // Inter-core communication

// Motor control pins
const int MOTOR_LEFT_PWM = PWM0;
const int MOTOR_LEFT_DIR = D0;
const int MOTOR_RIGHT_PWM = PWM1;
const int MOTOR_RIGHT_DIR = D1;

// Encoder pins
const int ENCODER_LEFT_A = D2;
const int ENCODER_LEFT_B = D3;
const int ENCODER_RIGHT_A = D4;
const int ENCODER_RIGHT_B = D5;

// PID controller
struct PIDController {
    float Kp = 2.0;
    float Ki = 0.5;
    float Kd = 0.1;

    float setpoint = 0.0;
    float integral = 0.0;
    float prev_error = 0.0;

    float compute(float input, float dt) {
        float error = setpoint - input;
        integral += error * dt;
        float derivative = (error - prev_error) / dt;
        prev_error = error;

        return Kp * error + Ki * integral + Kd * derivative;
    }

    void reset() {
        integral = 0.0;
        prev_error = 0.0;
    }
};

PIDController pid_left, pid_right;

// Motor state
volatile long encoder_left = 0;
volatile long encoder_right = 0;
float velocity_left = 0.0;  // m/s
float velocity_right = 0.0;

// Encoder ISRs
void encoder_left_isr() {
    if (digitalRead(ENCODER_LEFT_A) == digitalRead(ENCODER_LEFT_B))
        encoder_left++;
    else
        encoder_left--;
}

void encoder_right_isr() {
    if (digitalRead(ENCODER_RIGHT_A) == digitalRead(ENCODER_RIGHT_B))
        encoder_right++;
    else
        encoder_right--;
}

void setup() {
    // Initialize serial for Jetson communication
    Serial1.begin(115200);

    // Motor pins
    pinMode(MOTOR_LEFT_PWM, OUTPUT);
    pinMode(MOTOR_LEFT_DIR, OUTPUT);
    pinMode(MOTOR_RIGHT_PWM, OUTPUT);
    pinMode(MOTOR_RIGHT_DIR, OUTPUT);

    // Encoder pins
    pinMode(ENCODER_LEFT_A, INPUT_PULLUP);
    pinMode(ENCODER_LEFT_B, INPUT_PULLUP);
    pinMode(ENCODER_RIGHT_A, INPUT_PULLUP);
    pinMode(ENCODER_RIGHT_B, INPUT_PULLUP);

    // Attach interrupts
    attachInterrupt(digitalPinToInterrupt(ENCODER_LEFT_A), encoder_left_isr, CHANGE);
    attachInterrupt(digitalPinToInterrupt(ENCODER_RIGHT_A), encoder_right_isr, CHANGE);

    // Start M4 core (if used)
    RPC.begin();
    bootM4();
}

void loop() {
    static unsigned long last_time = 0;
    unsigned long current_time = micros();
    float dt = (current_time - last_time) / 1000000.0;  // seconds

    if (dt >= 0.01) {  // 100 Hz control loop
        // Calculate velocities from encoders
        const float WHEEL_RADIUS = 0.033;  // meters
        const int ENCODER_CPR = 360;  // counts per revolution

        float dist_per_count = (2 * PI * WHEEL_RADIUS) / ENCODER_CPR;
        velocity_left = (encoder_left * dist_per_count) / dt;
        velocity_right = (encoder_right * dist_per_count) / dt;

        // Reset encoders
        encoder_left = 0;
        encoder_right = 0;

        // PID control
        float left_output = pid_left.compute(velocity_left, dt);
        float right_output = pid_right.compute(velocity_right, dt);

        // Apply motor commands
        setMotor(MOTOR_LEFT_PWM, MOTOR_LEFT_DIR, left_output);
        setMotor(MOTOR_RIGHT_PWM, MOTOR_RIGHT_DIR, right_output);

        // Send telemetry to Jetson
        sendTelemetry();

        last_time = current_time;
    }

    // Process commands from Jetson
    processCommands();
}

void setMotor(int pwm_pin, int dir_pin, float output) {
    // Clamp output to [-255, 255]
    output = constrain(output, -255, 255);

    // Set direction
    if (output >= 0) {
        digitalWrite(dir_pin, HIGH);
    } else {
        digitalWrite(dir_pin, LOW);
        output = -output;
    }

    // Set PWM
    analogWrite(pwm_pin, (int)output);
}

void processCommands() {
    if (Serial1.available()) {
        String cmd = Serial1.readStringUntil('\n');

        // Parse command: "VEL left_vel right_vel"
        if (cmd.startsWith("VEL")) {
            int space1 = cmd.indexOf(' ', 4);
            int space2 = cmd.indexOf(' ', space1 + 1);

            float left_vel = cmd.substring(4, space1).toFloat();
            float right_vel = cmd.substring(space1 + 1, space2).toFloat();

            pid_left.setpoint = left_vel;
            pid_right.setpoint = right_vel;
        }
        // Parse command: "PID Kp Ki Kd"
        else if (cmd.startsWith("PID")) {
            // Update PID parameters
            // (implementation omitted for brevity)
        }
        // Emergency stop
        else if (cmd.startsWith("STOP")) {
            pid_left.setpoint = 0.0;
            pid_right.setpoint = 0.0;
            pid_left.reset();
            pid_right.reset();

            setMotor(MOTOR_LEFT_PWM, MOTOR_LEFT_DIR, 0);
            setMotor(MOTOR_RIGHT_PWM, MOTOR_RIGHT_DIR, 0);
        }
    }
}

void sendTelemetry() {
    // Send: "TEL velocity_left velocity_right encoder_left encoder_right"
    Serial1.print("TEL ");
    Serial1.print(velocity_left, 4);
    Serial1.print(" ");
    Serial1.print(velocity_right, 4);
    Serial1.print(" ");
    Serial1.print(encoder_left);
    Serial1.print(" ");
    Serial1.println(encoder_right);
}

    M4 Core (Optional - Sensor Processing)¶

    /**
 * @file main_m4.cpp
 * @brief Cortex-M4 secondary core - Sensor data acquisition
 */

#include <Arduino.h>
#include <RPC.h>

// I2C sensors
#include <Wire.h>

// IMU
const int IMU_ADDR = 0x68;

void setup() {
    Wire.begin();
    RPC.begin();

    // Initialize IMU
    Wire.beginTransmission(IMU_ADDR);
    Wire.write(0x6B);  // PWR_MGMT_1
    Wire.write(0x00);  // Wake up
    Wire.endTransmission();
}

void loop() {
    // Read IMU data at 200 Hz
    static unsigned long last_time = 0;
    if (millis() - last_time >= 5) {  // 200 Hz
        readIMU();
        last_time = millis();
    }
}

void readIMU() {
    // Read accelerometer and gyroscope
    Wire.beginTransmission(IMU_ADDR);
    Wire.write(0x3B);  // ACCEL_XOUT_H
    Wire.endTransmission(false);
    Wire.requestFrom(IMU_ADDR, 14, true);

    int16_t accel_x = Wire.read() << 8 | Wire.read();
    int16_t accel_y = Wire.read() << 8 | Wire.read();
    int16_t accel_z = Wire.read() << 8 | Wire.read();
    int16_t temp = Wire.read() << 8 | Wire.read();
    int16_t gyro_x = Wire.read() << 8 | Wire.read();
    int16_t gyro_y = Wire.read() << 8 | Wire.read();
    int16_t gyro_z = Wire.read() << 8 | Wire.read();

    // Send to M7 core via RPC
    RPC.call("imu_data", accel_x, accel_y, accel_z, gyro_x, gyro_y, gyro_z);
}

    Serial Protocol (Jetson ↔ Portenta)¶

    Baud Rate: 115200
Data Bits: 8
Parity: None
Stop Bits: 1
Flow Control: None

Command Format (Jetson → Portenta):
  VEL <left_vel> <right_vel>     # Set motor velocities (m/s)
  PID <Kp> <Ki> <Kd>              # Update PID gains
  STOP                            # Emergency stop
  RESET                           # Reset system
  CONFIG <param> <value>          # Set configuration parameter

Response Format (Portenta → Jetson):
  TEL <vel_l> <vel_r> <enc_l> <enc_r>   # Telemetry data
  ACK <command>                           # Command acknowledged
  ERR <error_code> <message>              # Error message
  STATUS <battery> <temp> <current>       # System status

    ROS2 Bridge Node¶

    #!/usr/bin/env python3
"""
ROS2 node for communicating with Arduino Portenta H7
"""

import rclpy
from rclpy.node import Node
from geometry_msgs.msg import Twist
from nav_msgs.msg import Odometry
from sensor_msgs.msg import JointState
import serial
import threading

class PortentaBridgeNode(Node):
    def __init__(self):
        super().__init__('portenta_bridge_node')

        # Parameters
        self.declare_parameter('serial_port', '/dev/ttyACM0')
        self.declare_parameter('baud_rate', 115200)
        self.declare_parameter('wheel_separation', 0.15)  # meters
        self.declare_parameter('wheel_radius', 0.033)    # meters

        port = self.get_parameter('serial_port').value
        baud = self.get_parameter('baud_rate').value

        # Serial connection
        self.serial = serial.Serial(port, baud, timeout=0.1)
        self.serial.reset_input_buffer()

        # ROS2 subscriptions
        self.cmd_vel_sub = self.create_subscription(
            Twist,
            '/cmd_vel',
            self.cmd_vel_callback,
            10
        )

        # ROS2 publishers
        self.odom_pub = self.create_publisher(Odometry, '/odom', 10)
        self.joint_state_pub = self.create_publisher(JointState, '/joint_states', 10)

        # Start serial read thread
        self.read_thread = threading.Thread(target=self.read_serial, daemon=True)
        self.read_thread.start()

        self.get_logger().info('Portenta bridge node started')

    def cmd_vel_callback(self, msg):
        """Convert Twist message to differential drive velocities"""
        linear = msg.linear.x
        angular = msg.angular.z

        wheel_sep = self.get_parameter('wheel_separation').value

        # Differential drive kinematics
        left_vel = linear - (angular * wheel_sep / 2.0)
        right_vel = linear + (angular * wheel_sep / 2.0)

        # Send command to Portenta
        cmd = f"VEL {left_vel:.4f} {right_vel:.4f}\n"
        self.serial.write(cmd.encode())

    def read_serial(self):
        """Thread for reading serial data from Portenta"""
        while rclpy.ok():
            try:
                if self.serial.in_waiting:
                    line = self.serial.readline().decode('utf-8').strip()
                    self.process_telemetry(line)
            except Exception as e:
                self.get_logger().error(f'Serial read error: {e}')

    def process_telemetry(self, line):
        """Process telemetry data from Portenta"""
        parts = line.split()

        if parts[0] == 'TEL' and len(parts) == 5:
            # Parse telemetry: TEL vel_l vel_r enc_l enc_r
            vel_left = float(parts[1])
            vel_right = float(parts[2])
            enc_left = int(parts[3])
            enc_right = int(parts[4])

            # Publish odometry
            self.publish_odometry(vel_left, vel_right)

            # Publish joint states
            self.publish_joint_states(vel_left, vel_right)

        elif parts[0] == 'STATUS' and len(parts) == 4:
            # Parse status: STATUS battery temp current
            battery = float(parts[1])
            temp = float(parts[2])
            current = float(parts[3])

            self.get_logger().info(
                f'Status - Battery: {battery}V, Temp: {temp}°C, Current: {current}A'
            )

        elif parts[0] == 'ERR':
            # Error message
            self.get_logger().error(f'Portenta error: {" ".join(parts[1:])}')

    def publish_odometry(self, vel_left, vel_right):
        """Publish odometry message"""
        # Calculate linear and angular velocities
        wheel_sep = self.get_parameter('wheel_separation').value

        linear = (vel_left + vel_right) / 2.0
        angular = (vel_right - vel_left) / wheel_sep

        # Create and publish Odometry message
        msg = Odometry()
        msg.header.stamp = self.get_clock().now().to_msg()
        msg.header.frame_id = 'odom'
        msg.child_frame_id = 'base_link'

        msg.twist.twist.linear.x = linear
        msg.twist.twist.angular.z = angular

        self.odom_pub.publish(msg)

    def publish_joint_states(self, vel_left, vel_right):
        """Publish joint states message"""
        msg = JointState()
        msg.header.stamp = self.get_clock().now().to_msg()
        msg.name = ['left_wheel_joint', 'right_wheel_joint']
        msg.velocity = [vel_left, vel_right]

        self.joint_state_pub.publish(msg)

def main(args=None):
    rclpy.init(args=args)
    node = PortentaBridgeNode()

    try:
        rclpy.spin(node)
    except KeyboardInterrupt:
        pass
    finally:
        node.serial.close()
        node.destroy_node()
        rclpy.shutdown()

if __name__ == '__main__':
    main()

    📷 OpenMV RT1062¶

    Overview¶

    The OpenMV RT1062 provides dedicated computer vision capabilities for lightweight image processing tasks.

    Key Features: 

    Processor: NXP i.MX RT1062 (Cortex-M7 @ 600 MHz)
Memory:
  SDRAM: 32 MB
  Flash: 8 MB

Camera:
  Sensor: OV5640 (5 MP)
  Resolution: 2592 x 1944 (max)
  Frame Rate: 15 FPS @ 1080p
  Interface: Parallel camera interface

Storage: MicroSD card slot
USB: USB 2.0 (device mode)
GPIO: 10 I/O pins (3.3V logic)
Power: 5V via USB or VIN

    MicroPython Programming¶

    """
OpenMV RT1062 - QR Code Scanner
Streams QR code data to Jetson via UART
"""

import sensor
import image
import time
import pyb

# Initialize camera
sensor.reset()
sensor.set_pixformat(sensor.RGB565)
sensor.set_framesize(sensor.QVGA)  # 320x240
sensor.skip_frames(time=2000)
sensor.set_auto_gain(False)
sensor.set_auto_whitebal(False)

# Initialize UART for Jetson communication
uart = pyb.UART(3, 115200, timeout_char=1000)

# LED indicator
led = pyb.LED(1)  # Red LED

print("OpenMV RT1062 - QR Code Scanner")
print("Streaming to Jetson via UART3...")

clock = time.clock()

while True:
    clock.tick()
    img = sensor.snapshot()

    # Detect QR codes
    qr_codes = img.find_qrcodes()

    if qr_codes:
        for qr in qr_codes:
            # Draw rectangle around QR code
            img.draw_rectangle(qr.rect(), color=(255, 0, 0))

            # Send data to Jetson
            data = qr.payload()
            uart.write(f"QR:{data}\n")

            # Blink LED
            led.on()
            pyb.delay(100)
            led.off()

            print(f"QR Code detected: {data}")

    # Display FPS
    fps = clock.fps()
    img.draw_string(5, 5, f"FPS: {fps:.1f}", color=(255, 255, 255), scale=2)

    Blob Detection (Color Tracking)¶

    """
OpenMV RT1062 - Color Blob Detection
Tracks colored objects and sends centroid to Jetson
"""

import sensor
import image
import time
import pyb

# Color thresholds (L, A, B)
# Use Tools -> Machine Vision -> Threshold Editor in OpenMV IDE
red_threshold = (30, 100, 15, 127, 15, 127)
green_threshold = (30, 100, -64, -8, -32, 32)
blue_threshold = (0, 30, 0, 64, -128, 0)

# Initialize
sensor.reset()
sensor.set_pixformat(sensor.RGB565)
sensor.set_framesize(sensor.QVGA)
sensor.skip_frames(time=2000)
sensor.set_auto_gain(False)
sensor.set_auto_whitebal(False)

uart = pyb.UART(3, 115200)

clock = time.clock()

while True:
    clock.tick()
    img = sensor.snapshot()

    # Find red blobs
    blobs = img.find_blobs(
        [red_threshold],
        pixels_threshold=200,
        area_threshold=200,
        merge=True
    )

    if blobs:
        # Find largest blob
        largest = max(blobs, key=lambda b: b.pixels())

        # Draw crosshair
        img.draw_rectangle(largest.rect(), color=(255, 0, 0))
        img.draw_cross(largest.cx(), largest.cy(), color=(255, 0, 0))

        # Send centroid to Jetson
        cx = largest.cx()
        cy = largest.cy()
        area = largest.pixels()

        uart.write(f"BLOB:{cx},{cy},{area}\n")

        print(f"Blob at ({cx}, {cy}), area={area}")

    fps = clock.fps()
    img.draw_string(5, 5, f"FPS: {fps:.1f}", color=(255, 255, 255))

    Line Following¶

    """
OpenMV RT1062 - Line Following
Detects line and calculates steering angle
"""

import sensor
import image
import time
import pyb
import math

# Initialize
sensor.reset()
sensor.set_pixformat(sensor.GRAYSCALE)
sensor.set_framesize(sensor.QQVGA)  # 160x120 for speed
sensor.skip_frames(time=2000)

uart = pyb.UART(3, 115200)

# Region of interest (ROI)
ROI = (0, 100, 160, 20)  # Bottom 20 rows

clock = time.clock()

while True:
    clock.tick()
    img = sensor.snapshot()

    # Find line
    line = img.get_regression(
        [(100, 100)],  # Black threshold
        roi=ROI,
        robust=True
    )

    if line:
        # Draw line
        img.draw_line(line.line(), color=127)

        # Calculate steering angle
        # line.theta() returns angle in degrees (0-180)
        # 90° = straight ahead
        angle = line.theta()

        # Convert to steering command (-1.0 to 1.0)
        # Negative = turn left, Positive = turn right
        steering = (angle - 90) / 90.0

        # Send to Jetson
        uart.write(f"LINE:{steering:.3f}\n")

        print(f"Line angle: {angle:.1f}°, Steering: {steering:.3f}")
    else:
        # No line detected
        uart.write("LINE:LOST\n")

    fps = clock.fps()
    img.draw_string(5, 5, f"FPS: {fps:.1f}")

    ROS2 Bridge Node (OpenMV)¶

    #!/usr/bin/env python3
"""
ROS2 node for OpenMV RT1062 communication
"""

import rclpy
from rclpy.node import Node
from geometry_msgs.msg import Point
from std_msgs.msg import String
import serial

class OpenMVBridgeNode(Node):
    def __init__(self):
        super().__init__('openmv_bridge_node')

        # Parameters
        self.declare_parameter('serial_port', '/dev/ttyACM1')
        self.declare_parameter('baud_rate', 115200)

        port = self.get_parameter('serial_port').value
        baud = self.get_parameter('baud_rate').value

        # Serial connection
        self.serial = serial.Serial(port, baud, timeout=0.1)

        # Publishers
        self.qr_pub = self.create_publisher(String, '/openmv/qr_code', 10)
        self.blob_pub = self.create_publisher(Point, '/openmv/blob_centroid', 10)
        self.line_pub = self.create_publisher(String, '/openmv/line_steering', 10)

        # Timer for reading serial
        self.create_timer(0.01, self.read_serial)  # 100 Hz

        self.get_logger().info('OpenMV bridge node started')

    def read_serial(self):
        """Read and process data from OpenMV"""
        if self.serial.in_waiting:
            try:
                line = self.serial.readline().decode('utf-8').strip()
                self.process_data(line)
            except Exception as e:
                self.get_logger().error(f'Serial error: {e}')

    def process_data(self, line):
        """Process data from OpenMV"""
        parts = line.split(':')

        if len(parts) == 2:
            msg_type, data = parts

            if msg_type == 'QR':
                # QR code detected
                msg = String()
                msg.data = data
                self.qr_pub.publish(msg)
                self.get_logger().info(f'QR Code: {data}')

            elif msg_type == 'BLOB':
                # Blob centroid: cx,cy,area
                coords = data.split(',')
                if len(coords) == 3:
                    msg = Point()
                    msg.x = float(coords[0])
                    msg.y = float(coords[1])
                    msg.z = float(coords[2])  # area
                    self.blob_pub.publish(msg)

            elif msg_type == 'LINE':
                # Line steering angle
                msg = String()
                msg.data = data
                self.line_pub.publish(msg)

def main(args=None):
    rclpy.init(args=args)
    node = OpenMVBridgeNode()

    try:
        rclpy.spin(node)
    except KeyboardInterrupt:
        pass
    finally:
        node.serial.close()
        node.destroy_node()
        rclpy.shutdown()

if __name__ == '__main__':
    main()

    🛠️ Troubleshooting¶

    Common Issues¶

    Serial Connection Fails¶

    # Check device
ls /dev/ttyACM*

# Verify permissions
sudo usermod -aG dialout $USER
sudo chmod 666 /dev/ttyACM0

# Create udev rule
sudo nano /etc/udev/rules.d/99-usb-serial.rules
# Add:
SUBSYSTEM=="tty", ATTRS{idVendor}=="2341", ATTRS{idProduct}=="025b", SYMLINK+="portenta"

sudo udevadm control --reload-rules

    Firmware Upload Issues¶

    # Put Portenta in bootloader mode:
# 1. Double-tap reset button
# 2. LED should pulse (DFU mode)
