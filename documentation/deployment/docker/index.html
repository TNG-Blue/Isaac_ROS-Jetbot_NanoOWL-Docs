    Docker Deployment¶

    Deploy Isaac ROS Jetbot NanoOWL using Docker containers.

    📚 Overview¶

    Docker enables containerized deployment of the Isaac ROS Jetbot NanoOWL system, providing consistent environments across development, testing, and production.

    Benefits: - ✅ Reproducible environments - ✅ Easy distribution - ✅ Isolated dependencies - ✅ Multi-service composition - ✅ CI/CD integration - ✅ Rapid deployment

    🏗️ Docker Architecture¶

    Available Images¶

    YAML
    Docker Images:
├── python:latest
│   └── Base Python 3.10 with ROS2 dependencies
├── jetbot_nanoowl:latest
│   └── Complete Isaac ROS Jetbot system
├── object_detection_isaac_ros:latest
│   └── YOLOv8 + TensorRT detection module
├── lidar_slam:latest
│   └── Visual-inertial SLAM system
├── stereo:latest
│   └── Stereo vision processing
├── server_ota:latest
│   └── OTA update server
├── oled:latest
│   └── OLED display controller
└── isaac_ros:latest
    └── Base Isaac ROS environment

    🚀 Quick Start with Docker¶

    Method 1: Pre-built Images¶

    Bash
    # Pull pre-built image
docker pull tngblue/jetbot_nanoowl:latest

# Run container
docker run -it --rm \
  --runtime nvidia \
  --gpus all \
  -v /dev/video0:/dev/video0 \
  -v $(pwd)/data:/home/jetbot/data \
  tngblue/jetbot_nanoowl:latest

# Inside container
ros2 launch jetbot_bringup main.launch.py

    Method 2: Build from Source¶

    Bash
    # Clone repository
git clone https://github.com/TNG-Blue/Isaac_ROS-Jetbot_NanoOWL.git
cd Isaac_ROS-Jetbot_NanoOWL/Docker

# Build image
./scripts/build_all.sh

# Or build specific image
docker build -t jetbot_nanoowl:latest -f images/jetbot_nanoowl/Dockerfile .

# Run container
docker run -it --gpus all jetbot_nanoowl:latest

    📋 Dockerfile Examples¶

    Main Jetbot Image¶

    Docker
    # From: Docker/images/jetbot_nanoowl/Dockerfile
FROM nvcr.io/nvidia/l4t-cuda:12.2-devel-ubuntu22.04

# Install ROS2 and dependencies
RUN apt-get update && apt-get install -y \
    ros-humble-desktop \
    ros-humble-nav2-bringup \
    python3-pip \
    libopencv-dev

# Install Python packages
RUN pip install \
    ultralytics \
    tensorrt \
    pycuda

# Copy source code
COPY . /home/jetbot/ros2_ws

# Build ROS2
WORKDIR /home/jetbot/ros2_ws
RUN . /opt/ros/humble/setup.sh && \
    colcon build --symlink-install

# Entrypoint
COPY entrypoint.sh /
ENTRYPOINT ["/entrypoint.sh"]
CMD ["ros2", "launch", "jetbot_bringup", "main.launch.py"]

    Multi-Stage Object Detection¶

    Docker
    # Build stage
FROM nvcr.io/nvidia/l4t-cuda:12.2-devel-ubuntu22.04 as builder

WORKDIR /build
COPY . .

RUN apt-get update && \
    apt-get install -y ros-humble-desktop && \
    . /opt/ros/humble/setup.sh && \
    colcon build

# Runtime stage
FROM nvcr.io/nvidia/l4t-cuda:12.2-runtime-ubuntu22.04

RUN apt-get update && \
    apt-get install -y ros-humble-desktop

COPY --from=builder /build/install /opt/ros/jetbot/install

CMD ["bash", "-c", \
     "source /opt/ros/jetbot/install/setup.bash && \
      ros2 launch jetbot_detection detection.launch.py"]

    🐳 Docker Compose¶

    Full Stack Deployment¶

    YAML
    # File: Docker/compose/docker-compose.yml
version: '3.8'

services:
  # Motor Controller
  motor_controller:
    image: jetbot_nanoowl:latest
    container_name: motor_controller
    runtime: nvidia
    environment:
      - NVIDIA_VISIBLE_DEVICES=all
      - ROS_DOMAIN_ID=0
    devices:
      - /dev/ttyUSB0:/dev/ttyUSB0  # Serial port
      - /dev/video0:/dev/video0     # Camera
    volumes:
      - ./data:/home/jetbot/data
      - ./config:/home/jetbot/config
    networks:
      - jetbot_network
    command: ros2 launch jetbot_control motor_controller.launch.py

  # Object Detection
  detection:
    image: object_detection_isaac_ros:latest
    container_name: detection
    runtime: nvidia
    environment:
      - NVIDIA_VISIBLE_DEVICES=all
      - ROS_DOMAIN_ID=0
    depends_on:
      - motor_controller
    volumes:
      - ./models:/home/jetbot/models
      - ./data:/home/jetbot/data
    networks:
      - jetbot_network
    command: ros2 launch jetbot_detection detection.launch.py

  # SLAM & Navigation
  slam:
    image: lidar_slam:latest
    container_name: slam
    runtime: nvidia
    environment:
      - NVIDIA_VISIBLE_DEVICES=all
      - ROS_DOMAIN_ID=0
    depends_on:
      - motor_controller
    volumes:
      - ./maps:/home/jetbot/maps
      - ./data:/home/jetbot/data
    networks:
      - jetbot_network
    command: ros2 launch jetbot_slam slam.launch.py

  # OTA Update Server
  ota_server:
    image: server_ota:latest
    container_name: ota_server
    ports:
      - "8080:8080"
    volumes:
      - ./updates:/app/updates
      - ./logs:/app/logs
    networks:
      - jetbot_network
    environment:
      - PORT=8080
      - UPDATE_DIR=/app/updates

  # Monitoring
  monitoring:
    image: jetbot_nanoowl:latest
    container_name: monitoring
    environment:
      - ROS_DOMAIN_ID=0
    depends_on:
      - motor_controller
    ports:
      - "9090:9090"
    networks:
      - jetbot_network
    command: bash -c "source install/setup.bash && ros2 run jetbot_monitoring monitoring_node"

  # OLED Display (optional)
  oled:
    image: oled:latest
    container_name: oled
    devices:
      - /dev/i2c-1:/dev/i2c-1
    environment:
      - ROS_DOMAIN_ID=0
    depends_on:
      - motor_controller
    networks:
      - jetbot_network
    command: ros2 run jetbot_display oled_node

networks:
  jetbot_network:
    driver: bridge

volumes:
  data:
    driver: local
  models:
    driver: local
  maps:
    driver: local

    📦 Build Scripts¶

    Build All Images¶

    Bash
    #!/bin/bash
# File: Docker/scripts/build_all.sh

set -e

REGISTRY=${1:-tngblue}
VERSION=${2:-latest}

echo "Building all Docker images..."

# Base Python image
echo "Building python:latest..."
docker build -t ${REGISTRY}/python:${VERSION} -f images/python/Dockerfile .

# Isaac ROS base
echo "Building isaac_ros:latest..."
docker build -t ${REGISTRY}/isaac_ros:${VERSION} -f images/isaac_ros/Dockerfile .

# Main Jetbot image
echo "Building jetbot_nanoowl:latest..."
docker build -t ${REGISTRY}/jetbot_nanoowl:${VERSION} -f images/jetbot_nanoowl/Dockerfile .

# Object Detection
echo "Building object_detection_isaac_ros:latest..."
docker build -t ${REGISTRY}/object_detection_isaac_ros:${VERSION} \
  -f images/object_detection_isaac_ros/Dockerfile .

# SLAM
echo "Building lidar_slam:latest..."
docker build -t ${REGISTRY}/lidar_slam:${VERSION} -f images/lidar_slam/Dockerfile .

# Stereo Vision
echo "Building stereo:latest..."
docker build -t ${REGISTRY}/stereo:${VERSION} -f images/stereo/Dockerfile .

# OTA Server
echo "Building server_ota:latest..."
docker build -t ${REGISTRY}/server_ota:${VERSION} -f images/server_ota/Dockerfile .

# OLED Display
echo "Building oled:latest..."
docker build -t ${REGISTRY}/oled:${VERSION} -f images/oled/Dockerfile .

echo "✓ All images built successfully!"

    🚀 Deployment Scenarios¶

    Scenario 1: Single Container¶

    Bash
    # Start complete system in one container
docker run -d --name jetbot \
  --runtime nvidia \
  --gpus all \
  -e ROS_DOMAIN_ID=0 \
  -v /dev/video0:/dev/video0 \
  -v /dev/ttyUSB0:/dev/ttyUSB0 \
  -v $(pwd)/data:/home/jetbot/data \
  -p 9090:9090 \
  tngblue/jetbot_nanoowl:latest

# Monitor logs
docker logs -f jetbot

# Stop container
docker stop jetbot

    Scenario 2: Docker Compose Multi-Container¶

    Bash
    # Start all services
cd Docker/compose
docker-compose up -d

# View services
docker-compose ps

# View logs
docker-compose logs -f

# Stop services
docker-compose down

    Scenario 3: Development Environment¶

    Bash
    # Start container with mounted source
docker run -it --rm \
  --runtime nvidia \
  --gpus all \
  -v $(pwd):/home/jetbot/development \
  -v /dev/video0:/dev/video0 \
  -w /home/jetbot/development \
  tngblue/jetbot_nanoowl:latest \
  bash

# Inside container
source install/setup.bash
colcon build --symlink-install
ros2 launch jetbot_bringup main.launch.py

    🔧 Configuration¶

    Environment Variables¶

    Bash
    # Set in docker-compose.yml or docker run
ROS_DOMAIN_ID=0              # ROS2 domain
ROS_LOCALHOST_ONLY=0         # Allow network communication
NVIDIA_VISIBLE_DEVICES=all   # GPU access
CUDA_VISIBLE_DEVICES=0       # Specific GPU

    Volume Mounts¶

    YAML
    volumes:
  - ./config:/home/jetbot/config          # Configuration files
  - ./data:/home/jetbot/data              # Output data
  - ./models:/home/jetbot/models          # AI models
  - ./maps:/home/jetbot/maps              # SLAM maps
  - /dev/video0:/dev/video0               # Camera device
  - /dev/ttyUSB0:/dev/ttyUSB0             # Serial port
  - /dev/i2c-1:/dev/i2c-1                 # I2C bus

    🔍 Monitoring Containers¶

    Check Container Status¶

    Bash
    # List running containers
docker ps

# View detailed info
docker inspect jetbot

# Check resource usage
docker stats jetbot

# View logs
docker logs jetbot
docker logs -f jetbot --tail 100

    Access Container Shell¶

    Bash
    # Execute bash in running container
docker exec -it jetbot bash

# Run ROS2 commands
docker exec -it jetbot ros2 topic list
docker exec -it jetbot ros2 node list

    📊 Performance Optimization¶

    GPU Optimization¶

    YAML
    # docker-compose.yml
services:
  jetbot:
    runtime: nvidia
    environment:
      - NVIDIA_VISIBLE_DEVICES=all
      - NVIDIA_DRIVER_CAPABILITIES=compute,utility
    deploy:
      resources:
        reservations:
          devices:
            - driver: nvidia
              count: 1
              capabilities: [gpu]

    Memory Limits¶

    Bash
    # Limit memory usage
docker run --memory=6g --memory-swap=6g jetbot_nanoowl:latest

# Or in docker-compose.yml
services:
  jetbot:
    deploy:
      resources:
        limits:
          memory: 6G
        reservations:
          memory: 4G

    🐛 Troubleshooting¶

    GPU Access Issues¶

    Bash
    # Check NVIDIA Docker integration
docker run --rm --runtime=nvidia nvidia/cuda:12.2-runtime nvidia-smi

# If failed, install nvidia-docker
distribution=$(. /etc/os-release;echo $ID$VERSION_ID)
curl -s -L https://nvidia.github.io/nvidia-docker/gpgkey | sudo apt-key add -
curl -s -L https://nvidia.github.io/nvidia-docker/$distribution/nvidia-docker.list | \
  sudo tee /etc/apt/sources.list.d/nvidia-docker.list

sudo apt-get update
sudo apt-get install -y nvidia-docker2
sudo systemctl restart docker

    ROS2 Network Issues¶

    Bash
    # Check ROS2 network
docker exec jetbot ros2 node list

# Verify domain ID
docker exec jetbot env | grep ROS_DOMAIN_ID

# Enable network communication
docker network inspect jetbot_network

    Container Won't Start¶

    Bash
    # Check image exists
docker images | grep jetbot_nanoowl

# Build image if missing
docker build -t jetbot_nanoowl:latest -f Docker/images/jetbot_nanoowl/Dockerfile .

# Check logs for errors
docker logs <container-id>

    🔐 Security¶

    Image Security¶

    Docker
    # Non-root user
RUN useradd -m -s /bin/bash jetbot
USER jetbot

# Read-only filesystem where possible
RUN chmod 555 /app

    Registry Authentication¶

    Bash
    # Login to Docker Hub
docker login

# Tag and push image
docker tag jetbot_nanoowl:latest myregistry/jetbot_nanoowl:v1.0
docker push myregistry/jetbot_nanoowl:v1.0

# Pull from private registry
docker login myregistry
docker pull myregistry/jetbot_nanoowl:v1.0
