    OTA Updates Deployment¶

    Over-The-Air update infrastructure for Isaac ROS Jetbot NanoOWL deployments.

    📚 Overview¶

    OTA (Over-The-Air) updates enable remote firmware and software updates for deployed robot fleets without physical access.

    Benefits: - ✅ Remote updates - ✅ Scheduled deployment - ✅ Automatic rollback - ✅ Fleet management - ✅ Zero downtime - ✅ Version control

    🚀 Quick Start¶

    Update Server Installation¶

    Bash
    # Deploy update server container
docker pull tngblue/server_ota:latest

docker run -d --name ota-server \
  -p 8080:8080 \
  -v $(pwd)/updates:/app/updates \
  -v $(pwd)/logs:/app/logs \
  -e PORT=8080 \
  tngblue/server_ota:latest

# Verify server
curl http://localhost:8080/api/v1/status

    Robot OTA Client Setup¶

    Bash
    # Install OTA client on Jetbot
pip install jetbot-ota-client

# Configure OTA settings
cat > ~/.jetbot/ota-config.yaml << 'EOF'
update_server: "http://update-server.example.com:8080"
check_interval: 3600  # seconds
auto_update: false
rollback_timeout: 300  # seconds
EOF

# Start OTA client daemon
systemctl start jetbot-ota-client

    📦 Update Management¶

    Create Firmware Update¶

    Bash
    # Build firmware image
cd ~/firmware
make clean
make build TARGET=jetbot_nanoowl

# Package firmware
tar -czf firmware_v2.1.0.tar.gz -C build/ .

# Upload to update server
curl -X POST http://update-server:8080/api/v1/firmware/upload \
  -F "file=@firmware_v2.1.0.tar.gz" \
  -F "version=2.1.0" \
  -F "target=jetbot_nanoowl" \
  -H "Authorization: Bearer $API_KEY"

    Create Software Update¶

    Bash
    # Build ROS2 packages
cd ~/ros2_ws
colcon build --packages-select jetbot_detection jetbot_slam

# Create deployment package
tar -czf software_v1.3.0.tar.gz -C install .

# Upload to update server
curl -X POST http://update-server:8080/api/v1/software/upload \
  -F "file=@software_v1.3.0.tar.gz" \
  -F "version=1.3.0" \
  -F "packages=jetbot_detection,jetbot_slam" \
  -H "Authorization: Bearer $API_KEY"

    🎯 Deployment Strategies¶

    Strategy 1: Immediate Deployment¶

    YAML
    # Update manifest
api_version: "1.0"
update_type: "firmware"
version: "2.1.0"
target: "jetbot_nanoowl"
deployment:
  strategy: "immediate"
  retry_count: 3
  retry_interval: 60
robots:
  - "jetbot-1"
  - "jetbot-2"
  - "jetbot-3"

    Strategy 2: Staged Rollout¶

    YAML
    api_version: "1.0"
update_type: "software"
version: "1.3.0"
deployment:
  strategy: "staged"
  stages:
    - devices: 1
      duration: 300  # 5 minutes
    - devices: 5
      duration: 600  # 10 minutes
    - devices: "all"
  rollback_condition: "error_rate > 10%"
  health_check_interval: 30

    Strategy 3: Scheduled Deployment¶

    YAML
    api_version: "1.0"
update_type: "software"
version: "1.3.0"
deployment:
  strategy: "scheduled"
  schedule:
    - time: "02:00"  # 2 AM UTC
      devices: "all"
      timezone: "UTC"
  notification_advance: 24  # hours
  maintenance_window: 1800   # seconds

    🤖 Robot OTA Client¶

    Client Configuration¶

    Python
    # File: ~/ros2_ws/src/jetbot_ota/config.py
import yaml

class OTAConfig:
    def __init__(self, config_file):
        with open(config_file, 'r') as f:
            self.config = yaml.safe_load(f)

    @property
    def update_server(self):
        return self.config['update_server']

    @property
    def check_interval(self):
        return self.config.get('check_interval', 3600)

    @property
    def auto_update(self):
        return self.config.get('auto_update', False)

    @property
    def rollback_timeout(self):
        return self.config.get('rollback_timeout', 300)

    OTA Client Implementation¶

    Python
    # File: ~/ros2_ws/src/jetbot_ota/ota_client.py
import rclpy
from rclpy.node import Node
import requests
import subprocess
import time
from pathlib import Path

class OTAClient(Node):
    def __init__(self):
        super().__init__('ota_client')

        self.config = OTAConfig('/home/jetbot/.jetbot/ota-config.yaml')
        self.robot_id = self.get_robot_id()
        self.current_version = self.get_current_version()

        # Timer for checking updates
        self.timer = self.create_timer(
            self.config.check_interval,
            self.check_for_updates
        )

        self.get_logger().info(f"OTA Client initialized. Robot: {self.robot_id}")

    def check_for_updates(self):
        """Check update server for new versions"""
        try:
            response = requests.get(
                f"{self.config.update_server}/api/v1/latest-update",
                params={"robot_id": self.robot_id},
                timeout=10
            )

            if response.status_code == 200:
                update_info = response.json()

                if update_info['version'] > self.current_version:
                    self.get_logger().info(
                        f"Update available: {update_info['version']}"
                    )

                    if self.config.auto_update:
                        self.download_and_install(update_info)
                    else:
                        self.notify_update_available(update_info)

        except Exception as e:
            self.get_logger().error(f"Failed to check updates: {e}")

    def download_and_install(self, update_info):
        """Download and install update"""
        try:
            # Download
            self.get_logger().info(f"Downloading update {update_info['version']}")
            self.download_update(update_info)

            # Verify
            if not self.verify_update(update_info):
                raise Exception("Checksum verification failed")

            # Backup current
            self.backup_current_version()

            # Install
            self.get_logger().info("Installing update...")
            self.install_update(update_info)

            # Verify installation
            if self.verify_installation(update_info):
                self.get_logger().info("Update installed successfully")
                self.current_version = update_info['version']
                self.report_update_status("completed", update_info)
            else:
                raise Exception("Installation verification failed")

        except Exception as e:
            self.get_logger().error(f"Update failed: {e}")
            self.rollback_update()
            self.report_update_status("failed", update_info)

    def download_update(self, update_info):
        """Download update file"""
        url = update_info['download_url']
        output_file = Path(f"/tmp/{update_info['filename']}")

        response = requests.get(url, stream=True)
        total_size = int(response.headers.get('content-length', 0))

        with open(output_file, 'wb') as f:
            downloaded = 0
            for chunk in response.iter_content(chunk_size=8192):
                f.write(chunk)
                downloaded += len(chunk)
                progress = (downloaded / total_size) * 100
                self.get_logger().debug(f"Download progress: {progress:.1f}%")

    def verify_update(self, update_info):
        """Verify update checksum"""
        import hashlib

        file_path = Path(f"/tmp/{update_info['filename']}")
        sha256_hash = hashlib.sha256()

        with open(file_path, 'rb') as f:
            for chunk in iter(lambda: f.read(4096), b''):
                sha256_hash.update(chunk)

        actual_checksum = sha256_hash.hexdigest()
        expected_checksum = update_info['checksum'].replace('sha256:', '')

        return actual_checksum == expected_checksum

    def install_update(self, update_info):
        """Install update"""
        subprocess.run([
            'sudo', 'systemctl', 'stop', 'jetbot-ros2'
        ])

        # Install logic depends on update type
        if update_info['type'] == 'firmware':
            self.install_firmware(update_info)
        elif update_info['type'] == 'software':
            self.install_software(update_info)

        subprocess.run([
            'sudo', 'systemctl', 'start', 'jetbot-ros2'
        ])

    def rollback_update(self):
        """Rollback to previous version"""
        self.get_logger().warn("Rolling back update...")
        subprocess.run(['sudo', '/usr/local/bin/rollback-update.sh'])

    def report_update_status(self, status, update_info):
        """Report update status to server"""
        try:
            requests.post(
                f"{self.config.update_server}/api/v1/update-status",
                json={
                    "robot_id": self.robot_id,
                    "update_id": update_info['id'],
                    "status": status,
                    "timestamp": time.time()
                }
            )
        except Exception as e:
            self.get_logger().error(f"Failed to report status: {e}")

def main():
    rclpy.init()
    client = OTAClient()
    rclpy.spin(client)
    rclpy.shutdown()

if __name__ == '__main__':
    main()

    🔄 Rollback Procedures¶

    Manual Rollback¶

    Bash
    # Check available backups
ls -la /home/jetbot/backups/

# Rollback to previous version
sudo /usr/local/bin/rollback-update.sh --version=2.0.0

# Verify
jetbot-cli status

    Automatic Rollback¶

    YAML
    # Enable automatic rollback on failure
api_version: "1.0"
deployment:
  auto_rollback: true
  rollback_conditions:
    - "error_rate > 20%"
    - "cpu_usage > 95%"
    - "motor_failure_detected"
  health_check_interval: 60

    🐛 Troubleshooting¶

    Update Failed¶

    Bash
    # Check update logs
tail -f /var/log/jetbot/ota-client.log

# Check disk space
df -h

# Check network connectivity
ping update-server

# Check update file integrity
sha256sum /tmp/update.tar.gz

    Rollback Needed¶

    Bash
    # Manually trigger rollback
sudo /usr/local/bin/rollback-update.sh

# Verify rollback
jetbot-cli version

# Report failure
curl -X POST http://update-server:8080/api/v1/update-status \
  -d '{"robot_id": "jetbot-1", "status": "rolled_back"}'
