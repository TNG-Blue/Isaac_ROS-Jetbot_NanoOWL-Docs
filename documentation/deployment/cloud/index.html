    Cloud Integration Deployment¶

    Deploy Isaac ROS Jetbot NanoOWL on cloud platforms.

    📚 Overview¶

    Cloud integration enables remote management, data analytics, and processing for Isaac ROS Jetbot fleets across distributed locations.

    Benefits: - ✅ Remote monitoring - ✅ Cloud processing - ✅ Centralized management - ✅ Data analytics - ✅ Scalable infrastructure - ✅ Global deployment

    🚀 AWS Deployment¶

    Prerequisites¶

    # Install AWS CLI
curl "https://awscli.amazonaws.com/awscli-exe-linux-x86_64.zip" -o "awscliv2.zip"
unzip awscliv2.zip
sudo ./aws/install

# Configure AWS credentials
aws configure
# AWS Access Key ID: [your-key]
# AWS Secret Access Key: [your-secret]
# Default region: us-east-1

    EC2 Deployment¶

    # Create EC2 instance for control plane
aws ec2 run-instances \
  --image-id ami-0c55b159cbfafe1f0 \
  --instance-type t3.medium \
  --key-name jetbot-key \
  --security-group-ids sg-0123456789abcdef0 \
  --tag-specifications 'ResourceType=instance,Tags=[{Key=Name,Value=jetbot-control}]'

# Get instance IP
INSTANCE_IP=$(aws ec2 describe-instances \
  --filters "Name=tag:Name,Values=jetbot-control" \
  --query 'Reservations[0].Instances[0].PublicIpAddress' \
  --output text)

# SSH to instance
ssh -i jetbot-key.pem ec2-user@$INSTANCE_IP

    RDS Database¶

    # Create RDS instance for telemetry
aws rds create-db-instance \
  --db-instance-identifier jetbot-db \
  --db-instance-class db.t3.micro \
  --engine mysql \
  --master-username admin \
  --master-user-password SecurePassword123 \
  --allocated-storage 20 \
  --backup-retention-period 7

# Get endpoint
aws rds describe-db-instances \
  --db-instance-identifier jetbot-db \
  --query 'DBInstances[0].Endpoint.Address' \
  --output text

    S3 Bucket¶

    # Create S3 bucket for data storage
aws s3 mb s3://jetbot-fleet-data-$(date +%s)

# Enable versioning
aws s3api put-bucket-versioning \
  --bucket jetbot-fleet-data-1234567890 \
  --versioning-configuration Status=Enabled

# Create lifecycle policy
aws s3api put-bucket-lifecycle-configuration \
  --bucket jetbot-fleet-data-1234567890 \
  --lifecycle-configuration file://lifecycle.json

    AWS CloudFormation Template¶

    # File: aws-deployment.yaml
AWSTemplateFormatVersion: '2010-09-09'
Description: 'Isaac ROS Jetbot NanoOWL - AWS CloudFormation Stack'

Parameters:
  InstanceType:
    Type: String
    Default: t3.medium
  DBInstanceClass:
    Type: String
    Default: db.t3.micro

Resources:
  # VPC
  JetbotVPC:
    Type: AWS::EC2::VPC
    Properties:
      CidrBlock: 10.0.0.0/16
      EnableDnsHostnames: true
      EnableDnsSupport: true

  # Subnet
  JetbotSubnet:
    Type: AWS::EC2::Subnet
    Properties:
      VpcId: !Ref JetbotVPC
      CidrBlock: 10.0.1.0/24
      AvailabilityZone: us-east-1a

  # EC2 Instance
  JetbotControlPlane:
    Type: AWS::EC2::Instance
    Properties:
      ImageId: ami-0c55b159cbfafe1f0
      InstanceType: !Ref InstanceType
      SubnetId: !Ref JetbotSubnet
      Tags:
        - Key: Name
          Value: jetbot-control

  # RDS Database
  JetbotDatabase:
    Type: AWS::RDS::DBInstance
    Properties:
      DBInstanceIdentifier: jetbot-db
      DBInstanceClass: !Ref DBInstanceClass
      Engine: mysql
      MasterUsername: admin
      MasterUserPassword: !Sub '{{resolve:secretsmanager:jetbot-db-password:SecretString:password}}'
      AllocatedStorage: 20
      BackupRetentionPeriod: 7

  # S3 Bucket
  JetbotBucket:
    Type: AWS::S3::Bucket
    Properties:
      BucketName: !Sub 'jetbot-fleet-data-${AWS::AccountId}'
      VersioningConfiguration:
        Status: Enabled

Outputs:
  ControlPlaneIP:
    Value: !GetAtt JetbotControlPlane.PublicIp
  DatabaseEndpoint:
    Value: !GetAtt JetbotDatabase.Endpoint.Address
  S3BucketName:
    Value: !Ref JetbotBucket

    🌐 Google Cloud Deployment¶

    Prerequisites¶

    # Install Google Cloud SDK
curl https://sdk.cloud.google.com | bash

# Initialize
gcloud init

# Set project
gcloud config set project my-project-id

    Compute Engine Deployment¶

    # Create VM instance
gcloud compute instances create jetbot-control \
  --zone=us-central1-a \
  --machine-type=n1-standard-2 \
  --image-family=ubuntu-2204-lts \
  --image-project=ubuntu-os-cloud

# SSH to instance
gcloud compute ssh jetbot-control --zone=us-central1-a

    Firestore Database¶

    # Create Firestore database
gcloud firestore databases create --location=us-central1

# Add collection
gcloud firestore collections create robots
gcloud firestore documents create robots/jetbot-1 \
  --data='{"status":"ready","battery":85}'

    Cloud Storage¶

    # Create storage bucket
gsutil mb gs://jetbot-fleet-data

# Configure lifecycle
gsutil lifecycle set lifecycle.json gs://jetbot-fleet-data

    Pub/Sub Messaging¶

    # Create topic
gcloud pubsub topics create jetbot-telemetry

# Create subscription
gcloud pubsub subscriptions create jetbot-telemetry-sub \
  --topic=jetbot-telemetry \
  --push-endpoint=https://api.example.com/webhook

curl -sL https://aka.ms/InstallAzureCLIDeb | sudo bash

az account set --subscription "subscription-id"

    # Create resource group

  --name jetbot-rg \
  --location eastus

az vm create \
  --resource-group jetbot-rg \
  --name jetbot-control \
  --image UbuntuLTS \
  --size Standard_B2s \
  --public-ip-address-allocation static

    Cosmos DB¶

    # Create Cosmos DB account
az cosmosdb create \
  --name jetbot-db \
  --resource-group jetbot-rg \
  --locations regionName=eastus \
  --default-consistency-level Session

# Create database
az cosmosdb database create \
  --account-name jetbot-db \
  --resource-group jetbot-rg \
  --name jetbot

    Blob Storage¶

    # Create storage account
az storage account create \
  --name jetbotdata \
  --resource-group jetbot-rg \
  --location eastus

# Create blob container
az storage container create \
  --account-name jetbotdata \
  --name fleet-data

    🔌 Data Sync Integration¶

    AWS Kinesis Streaming¶

    # File: ~/ros2_ws/src/jetbot_cloud/cloud_sync.py
import boto3
import json
import rclpy
from rclpy.node import Node
from sensor_msgs.msg import Imu

class CloudSync(Node):
    def __init__(self):
        super().__init__('cloud_sync')

        # Initialize AWS clients
        self.kinesis_client = boto3.client('kinesis', region_name='us-east-1')
        self.s3_client = boto3.client('s3')

        # Subscribe to sensor data
        self.imu_sub = self.create_subscription(
            Imu, '/imu/data', self.imu_callback, 10)

        self.robot_id = 'jetbot-1'
        self.stream_name = 'jetbot-telemetry'

    def imu_callback(self, msg):
        """Stream IMU data to Kinesis"""
        data = {
            'robot_id': self.robot_id,
            'timestamp': self.get_clock().now().to_msg().sec,
            'imu': {
                'acceleration_x': float(msg.linear_acceleration.x),
                'acceleration_y': float(msg.linear_acceleration.y),
                'acceleration_z': float(msg.linear_acceleration.z),
                'angular_velocity_x': float(msg.angular_velocity.x),
                'angular_velocity_y': float(msg.angular_velocity.y),
                'angular_velocity_z': float(msg.angular_velocity.z),
            }
        }

        try:
            self.kinesis_client.put_record(
                StreamName=self.stream_name,
                Data=json.dumps(data),
                PartitionKey=self.robot_id
            )
        except Exception as e:
            self.get_logger().error(f"Failed to sync data: {e}")

    Google Pub/Sub Integration¶

    # File: pubsub_sync.py
from google.cloud import pubsub_v1
import json

class PubSubSync:
    def __init__(self, project_id, topic_name):
        self.publisher = pubsub_v1.PublisherClient()
        self.topic_path = self.publisher.topic_path(project_id, topic_name)

    def publish_telemetry(self, data):
        """Publish telemetry to Pub/Sub"""
        message_json = json.dumps(data)
        message_bytes = message_json.encode('utf-8')

        publish_future = self.publisher.publish(
            self.topic_path, message_bytes)

        message_id = publish_future.result()
        print(f"Published message ID: {message_id}")

    🔐 Security¶

    AWS Security Groups¶

    # Create security group
aws ec2 create-security-group \
  --group-name jetbot-sg \
  --description "Jetbot security group"

# Allow SSH
aws ec2 authorize-security-group-ingress \
  --group-id sg-0123456789abcdef0 \
  --protocol tcp \
  --port 22 \
  --cidr 0.0.0.0/0

# Allow API access
aws ec2 authorize-security-group-ingress \
  --group-id sg-0123456789abcdef0 \
  --protocol tcp \
  --port 8080 \
  --cidr 0.0.0.0/0

    Secrets Management¶

    # Store API key in AWS Secrets Manager
aws secretsmanager create-secret \
  --name jetbot-api-key \
  --secret-string "your-secret-key"

# Retrieve secret
aws secretsmanager get-secret-value \
  --secret-id jetbot-api-key \
  --query SecretString \
  --output text
