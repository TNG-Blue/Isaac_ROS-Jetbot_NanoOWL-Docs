    Object Detection Module¶

    Real-time object detection using YOLOv8 and NanoOWL with TensorRT optimization.

    🎯 Overview¶

    The Object Detection module provides state-of-the-art real-time object detection capabilities optimized for NVIDIA Jetson hardware.

    Key Features: - YOLOv8 Nano/Small/Medium models - NanoOWL vision-language model - TensorRT INT8/FP16 optimization - 35+ FPS at 1280×720 resolution - 80+ COCO object categories - Custom model training support

    🚀 Quick Start¶

    Installation¶

    # Install dependencies
sudo apt install ros-humble-vision-msgs

# Clone and build
cd ~/ros2_ws/src
git clone https://github.com/TNG-Blue/Isaac_ROS-Jetbot_NanoOWL.git
cd ~/ros2_ws
colcon build --packages-select jetbot_detection
source install/setup.bash

    Running Detection¶

    # Launch YOLOv8 detection
ros2 launch jetbot_detection yolov8.launch.py

# With custom parameters
ros2 launch jetbot_detection yolov8.launch.py \
  confidence:=0.6 \
  model:=yolov8n \
  device:=cuda

    🔧 Configuration¶

    Parameters¶

    # config/detection_params.yaml
detection_node:
  ros__parameters:
    # Model settings
    model_type: "yolov8"  # yolov8, nanoowl
    model_path: "/models/yolov8n.engine"
    confidence_threshold: 0.5
    nms_threshold: 0.4

    # Input settings
    input_topic: "/camera/image_raw"
    camera_info_topic: "/camera/camera_info"
    image_width: 1280
    image_height: 720

    # Output settings
    publish_annotated: true
    publish_visualization: true
    annotated_topic: "/detection/annotated_image"
    detections_topic: "/detection/objects"

    # Performance settings
    device: "cuda"  # cuda, cpu
    use_tensorrt: true
    precision: "fp16"  # fp32, fp16, int8
    max_batch_size: 1

    # Classes
    class_names_file: "/config/coco_classes.txt"
    filter_classes: []  # Empty = all classes

    📊 API Reference¶

    ROS2 Topics¶

    Subscriptions¶

    /camera/image_raw:
  Type: sensor_msgs/Image
  QoS: BEST_EFFORT, KEEP_LAST(5)
  Description: Input camera stream

/camera/camera_info:
  Type: sensor_msgs/CameraInfo
  QoS: RELIABLE, KEEP_LAST(1)
  Description: Camera calibration info

    Publications¶

    /detection/objects:
  Type: vision_msgs/Detection2DArray
  QoS: RELIABLE, KEEP_LAST(10)
  Description: Detected objects with bounding boxes

/detection/annotated_image:
  Type: sensor_msgs/Image
  QoS: BEST_EFFORT, KEEP_LAST(1)
  Description: Visualized detections

/detection/performance:
  Type: std_msgs/Float32MultiArray
  QoS: RELIABLE, KEEP_LAST(1)
  Description: [fps, latency_ms, memory_mb]

    ROS2 Services¶

    /detection/set_confidence:
  Type: std_srvs/SetFloat
  Description: Update confidence threshold

/detection/reload_model:
  Type: std_srvs/Trigger
  Description: Reload detection model

/detection/get_classes:
  Type: std_srvs/GetStringArray
  Description: Get list of detected classes

    💻 C++ API¶

    Detector Interface¶

    #include <jetbot_detection/detector.hpp>

namespace jetbot {
namespace detection {

class Detector {
public:
    virtual ~Detector() = default;

    // Initialize detector
    virtual bool initialize(const std::string& model_path) = 0;

    // Perform detection
    virtual std::vector<Detection> detect(
        const cv::Mat& image,
        float confidence_threshold = 0.5f
    ) = 0;

    // Get class names
    virtual std::vector<std::string> getClassNames() const = 0;

    // Get model info
    virtual ModelInfo getModelInfo() const = 0;
};

struct Detection {
    int class_id;
    std::string class_name;
    float confidence;
    cv::Rect bbox;
    cv::Point2f center;
};

struct ModelInfo {
    std::string type;
    std::string path;
    std::string precision;
    int input_width;
    int input_height;
    int num_classes;
};

} // namespace detection
} // namespace jetbot

    YOLOv8 Example¶

    #include <jetbot_detection/yolov8_detector.hpp>

int main() {
    // Create detector
    auto detector = std::make_unique<jetbot::detection::YOLOv8Detector>();

    // Initialize with model
    if (!detector->initialize("/models/yolov8n.engine")) {
        std::cerr << "Failed to initialize detector" << std::endl;
        return -1;
    }

    // Load image
    cv::Mat image = cv::imread("test.jpg");

    // Perform detection
    auto detections = detector->detect(image, 0.5f);

    // Process results
    for (const auto& det : detections) {
        std::cout << "Detected: " << det.class_name 
                  << " (" << det.confidence << ")" 
                  << " at [" << det.bbox.x << "," << det.bbox.y 
                  << "," << det.bbox.width << "," << det.bbox.height << "]"
                  << std::endl;
    }

    return 0;
}

    🐍 Python API¶

    #!/usr/bin/env python3
import rclpy
from rclpy.node import Node
from sensor_msgs.msg import Image
from vision_msgs.msg import Detection2DArray
from cv_bridge import CvBridge

class DetectionSubscriber(Node):
    def __init__(self):
        super().__init__('detection_subscriber')

        self.subscription = self.create_subscription(
            Detection2DArray,
            '/detection/objects',
            self.detection_callback,
            10
        )

        self.image_subscription = self.create_subscription(
            Image,
            '/detection/annotated_image',
            self.image_callback,
            10
        )

        self.bridge = CvBridge()

    def detection_callback(self, msg):
        self.get_logger().info(f'Received {len(msg.detections)} detections')

        for detection in msg.detections:
            for result in detection.results:
                class_id = result.hypothesis.class_id
                score = result.hypothesis.score
                bbox = detection.bbox

                self.get_logger().info(
                    f'  {class_id}: {score:.2f} at '
                    f'[{bbox.center.position.x:.0f}, '
                    f'{bbox.center.position.y:.0f}]'
                )

    def image_callback(self, msg):
        # Convert to OpenCV image
        cv_image = self.bridge.imgmsg_to_cv2(msg, desired_encoding='bgr8')

        # Display or save
        # cv2.imshow('Detections', cv_image)
        # cv2.waitKey(1)

def main():
    rclpy.init()
    node = DetectionSubscriber()

    try:
        rclpy.spin(node)
    except KeyboardInterrupt:
        pass

    node.destroy_node()
    rclpy.shutdown()

if __name__ == '__main__':
    main()

    Custom Model Training¶

    # train_custom_model.py
from ultralytics import YOLO

# Load pretrained model
model = YOLO('yolov8n.pt')

# Train on custom dataset
results = model.train(
    data='custom_dataset.yaml',
    epochs=100,
    imgsz=640,
    batch=16,
    device=0,  # GPU
    project='runs/train',
    name='custom_yolov8'
)

# Export to ONNX
model.export(format='onnx', dynamic=False, simplify=True)

# Convert to TensorRT
# trtexec --onnx=yolov8n.onnx \
#         --saveEngine=yolov8n.engine \
#         --fp16 \
#         --workspace=4096

    ⚡ Performance Optimization¶

    TensorRT Optimization¶

    # Export ONNX model
python3 export_onnx.py --model yolov8n.pt --simplify

# Convert to TensorRT (FP16)
trtexec --onnx=yolov8n.onnx \
        --saveEngine=yolov8n_fp16.engine \
        --fp16 \
        --workspace=4096 \
        --minShapes=input:1x3x640x640 \
        --optShapes=input:1x3x640x640 \
        --maxShapes=input:1x3x640x640

# Convert to TensorRT (INT8 with calibration)
trtexec --onnx=yolov8n.onnx \
        --saveEngine=yolov8n_int8.engine \
        --int8 \
        --calib=calibration_cache.bin \
        --workspace=4096
