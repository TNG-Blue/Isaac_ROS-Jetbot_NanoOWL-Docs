    API Reference¶

    Complete function and class documentation for the CUDA Vision Library.

    📦 Core Classes¶

    CUDAProcessor¶

    Main processing class for GPU-accelerated image operations.

    Constructor¶

    CUDAProcessor::CUDAProcessor(
    int device_id = 0,
    const ImageSize& size = {1280, 720}
)

    Parameters: - device_id (int): NVIDIA GPU device ID (0-based index) - size (ImageSize): Input image dimensions {width, height}

    Example: 

    jetbot::CUDAProcessor processor(0, {1280, 720});

    Methods¶

    processFrame()¶

    Process a single image frame.

    void processFrame(
    const uint8_t* input_gpu,
    uint8_t* output_gpu,
    ImageFormat format = ImageFormat::RGBA
)

    Parameters: - input_gpu (const uint8_t*): GPU buffer containing input image - output_gpu (uint8_t*): GPU buffer for output (pre-allocated) - format (ImageFormat): Input image format (RGBA, BGRx, NV12)

    Example: 

    processor.processFrame(input_buffer, output_buffer, ImageFormat::RGBA);

    allocateGPUMemory()¶

    Pre-allocate GPU memory for batch processing.

    GPUMemory allocateGPUMemory(
    size_t num_frames,
    ImageFormat format = ImageFormat::RGBA
)

    Returns: GPUMemory object managing allocated resources

    Example: 

    auto gpu_mem = processor.allocateGPUMemory(10, ImageFormat::RGBA);

    enableStreaming()¶

    Enable asynchronous stream processing.

    void enableStreaming(int num_streams = 4)

    Parameters: - num_streams (int): Number of CUDA streams for async processing

    Example: 

    processor.enableStreaming(4);

    getCapabilities()¶

    Query processor capabilities.

    DeviceCapabilities getCapabilities() const

    Returns: DeviceCapabilities struct with GPU info

    Example: 

    auto caps = processor.getCapabilities();
std::cout << "Max threads: " << caps.max_threads_per_block << "\n";

    🖼️ Image Processing Functions¶

    Image Format Conversion¶

    grayscaleConvert()¶

    Convert color image to grayscale.

    void grayscaleConvert(
    const uint8_t* input_gpu,
    uint8_t* output_gpu,
    int width, int height,
    ImageFormat input_format = ImageFormat::RGBA,
    cudaStream_t stream = nullptr
)

    Parameters: - input_gpu: Color image on GPU - output_gpu: Grayscale output (pre-allocated) - width, height: Image dimensions - input_format: Input color format - stream: CUDA stream for async execution

    Example: 

    jetbot::grayscaleConvert(input, output, 1280, 720, 
                         jetbot::ImageFormat::RGBA);

    colorspaceConvert()¶

    Convert between color spaces (RGB ↔ HSV, RGB ↔ YUV).

    void colorspaceConvert(
    const uint8_t* input_gpu,
    uint8_t* output_gpu,
    int width, int height,
    ColorFormat from_format,
    ColorFormat to_format,
    cudaStream_t stream = nullptr
)

    Supported Conversions: - RGB ↔ HSV - RGB ↔ YUV - RGBA → NV12

    Example: 

    jetbot::colorspaceConvert(rgb_buffer, hsv_buffer, 1280, 720,
                          jetbot::ColorFormat::RGB, 
                          jetbot::ColorFormat::HSV);

    Filtering and Edge Detection¶

    gaussianBlur()¶

    Apply Gaussian blur to image.

    void gaussianBlur(
    const uint8_t* input_gpu,
    uint8_t* output_gpu,
    int width, int height,
    int kernel_size,
    float sigma = 1.0f,
    cudaStream_t stream = nullptr
)

    Parameters: - kernel_size: Kernel size (3, 5, 7, 9, 11) - sigma: Standard deviation for Gaussian kernel

    Example: 

    jetbot::gaussianBlur(input, output, 1280, 720, 5, 1.5f);

    cannyEdgeDetection()¶

    Detect edges using Canny algorithm.

    void cannyEdgeDetection(
    const uint8_t* input_gpu,
    uint8_t* output_gpu,
    int width, int height,
    float low_threshold = 50.0f,
    float high_threshold = 150.0f,
    cudaStream_t stream = nullptr
)

    Parameters: - low_threshold: Lower threshold for edge linking - high_threshold: Upper threshold for strong edges

    Example: 

    jetbot::cannyEdgeDetection(input, output, 1280, 720, 50.0f, 150.0f);

    sobelFilter()¶

    Apply Sobel filter for gradient computation.

    void sobelFilter(
    const uint8_t* input_gpu,
    float* output_gx, float* output_gy,
    int width, int height,
    cudaStream_t stream = nullptr
)

    Returns: Gradients in X and Y directions

    Example: 

    float *gx, *gy;
cudaMalloc(&gx, size); cudaMalloc(&gy, size);
jetbot::sobelFilter(input, gx, gy, 1280, 720);

    🔍 Feature Extraction¶

    SIFT Features¶

    extractSIFTFeatures()¶

    Extract Scale-Invariant Feature Transform keypoints.

    void extractSIFTFeatures(
    const uint8_t* input_gpu,
    std::vector<Keypoint>& keypoints,
    std::vector<Descriptor>& descriptors,
    int width, int height,
    float threshold = 0.03f,
    cudaStream_t stream = nullptr
)

    Returns: - keypoints: Vector of detected keypoints - descriptors: 128-dimension descriptor for each keypoint

    Example: 

    std::vector<jetbot::Keypoint> kps;
std::vector<jetbot::Descriptor> descs;
jetbot::extractSIFTFeatures(input, kps, descs, 640, 480, 0.03f);

    Harris Corner Detection¶

    detectHarrisCorners()¶

    Detect corners using Harris corner detector.

    void detectHarrisCorners(
    const uint8_t* input_gpu,
    std::vector<Corner>& corners,
    int width, int height,
    float threshold = 0.01f,
    int max_corners = 500,
    cudaStream_t stream = nullptr
)

    Parameters: - threshold: Corner response threshold - max_corners: Maximum number of corners to return

    Example: 

    std::vector<jetbot::Corner> corners;
jetbot::detectHarrisCorners(input, corners, 1280, 720, 0.01f, 500);

    🐍 Python API¶

    Import¶

    import jetbot_cuda_vision as jcv

    CUDAProcessor Class¶

    # Create processor
processor = jcv.CUDAProcessor(gpu_id=0, width=1280, height=720)

# Process frame (numpy array)
output = processor.process_frame(input_array)

# Get capabilities
caps = processor.get_capabilities()
print(f"Max threads: {caps['max_threads_per_block']}")

    Image Processing Functions¶

    # Grayscale conversion
gray = jcv.grayscale_convert(input_image)

# Gaussian blur
blurred = jcv.gaussian_blur(input_image, kernel_size=5, sigma=1.5)

# Canny edge detection
edges = jcv.canny_edge_detection(input_image, low=50, high=150)

# Feature extraction
keypoints, descriptors = jcv.extract_sift_features(input_image)

# Harris corners
corners = jcv.detect_harris_corners(input_image, max_corners=500)

    Streaming¶

    # Create video processor
video = jcv.VideoProcessor(width=1280, height=720, fps=30)

# Define processing chain
chain = [
    {"type": "grayscale"},
    {"type": "gaussian_blur", "kernel_size": 5}
]
video.set_processing_chain(chain)

# Process frame
processed = video.process_frame(input_frame)

    ⚠️ Important Notes¶

    Memory Lifecycle¶

    • GPU Memory: Caller responsible for allocation/deallocation
    • Return Values: Vectors contain copies of GPU data automatically
    • Pinned Memory: Use for CPU-GPU transfers

    Thread Safety¶

    • Streams: Each stream can be used by single thread
    • Device: Device context must be set in thread
    • Multi-threading: Create separate processor per thread

    Performance Considerations¶

    • Batch Processing: Use enableStreaming() for throughput
    • Memory Alignment: Ensure 32-byte alignment for GPU buffers
    • Stream Ordering: Enqueue operations in single stream
