    Code Examples¶

    Practical examples demonstrating CUDA Vision Library usage.

    📌 Example 1: Basic Image Processing¶

    C++ Example¶

    Convert image to grayscale and apply Gaussian blur.

    C++
    #include <jetbot_cuda_vision/cuda_processor.hpp>
#include <jetbot_cuda_vision/image_processing.hpp>
#include <iostream>
#include <cuda_runtime.h>

int main() {
    const int WIDTH = 1280;
    const int HEIGHT = 720;
    const size_t IMAGE_SIZE = WIDTH * HEIGHT * 4; // RGBA

    // Allocate GPU memory
    uint8_t *input_gpu, *grayscale_gpu, *blurred_gpu;
    cudaMalloc(&input_gpu, IMAGE_SIZE);
    cudaMalloc(&grayscale_gpu, WIDTH * HEIGHT);
    cudaMalloc(&blurred_gpu, WIDTH * HEIGHT);

    // Load image (pseudocode)
    uint8_t *host_image = new uint8_t[IMAGE_SIZE];
    // ... load image data ...
    cudaMemcpy(input_gpu, host_image, IMAGE_SIZE, cudaMemcpyHostToDevice);

    // Create processor
    jetbot::CUDAProcessor processor(0, {WIDTH, HEIGHT});

    // Convert to grayscale
    jetbot::grayscaleConvert(input_gpu, grayscale_gpu, WIDTH, HEIGHT,
                             jetbot::ImageFormat::RGBA);

    // Apply Gaussian blur
    jetbot::gaussianBlur(grayscale_gpu, blurred_gpu, WIDTH, HEIGHT, 
                         5, 1.5f);

    // Copy result back
    uint8_t *host_output = new uint8_t[WIDTH * HEIGHT];
    cudaMemcpy(host_output, blurred_gpu, WIDTH * HEIGHT, 
               cudaMemcpyDeviceToHost);

    std::cout << "✅ Processing complete\n";

    // Cleanup
    cudaFree(input_gpu);
    cudaFree(grayscale_gpu);
    cudaFree(blurred_gpu);
    delete[] host_image;
    delete[] host_output;

    return 0;
}

    Python Equivalent¶

    Python
    import jetbot_cuda_vision as jcv
import numpy as np
from PIL import Image

# Load image
image = Image.open("sample.jpg")
image_array = np.array(image, dtype=np.uint8)

# Create processor
processor = jcv.CUDAProcessor(gpu_id=0, width=1280, height=720)

# Convert to grayscale
gray = jcv.grayscale_convert(image_array)

# Apply Gaussian blur
blurred = jcv.gaussian_blur(gray, kernel_size=5, sigma=1.5)

# Save result
result = Image.fromarray(blurred)
result.save("output.jpg")

print("✅ Processing complete")

    📌 Example 2: Real-time Stream Processing¶

    Process continuous video stream with CUDA.

    C++ Implementation¶

    C++
    #include <jetbot_cuda_vision/video_processor.hpp>
#include <opencv2/opencv.hpp>
#include <iostream>

int main() {
    const int WIDTH = 1280;
    const int HEIGHT = 720;
    const int FPS = 30;

    // Open video source
    cv::VideoCapture cap(0);
    cap.set(cv::CAP_PROP_FRAME_WIDTH, WIDTH);
    cap.set(cv::CAP_PROP_FRAME_HEIGHT, HEIGHT);
    cap.set(cv::CAP_PROP_FPS, FPS);

    if (!cap.isOpened()) {
        std::cerr << "Failed to open camera\n";
        return 1;
    }

    // Create video processor
    jetbot::VideoProcessor video_proc(WIDTH, HEIGHT, FPS, 10);

    // Define processing chain
    std::vector<jetbot::ProcessingOp> chain = {
        {jetbot::OpType::GRAYSCALE, {}},
        {jetbot::OpType::GAUSSIAN_BLUR, {{"kernel_size", 5}, {"sigma", 1.5}}},
        {jetbot::OpType::CANNY_EDGE, {{"low_threshold", 50}, {"high_threshold", 150}}}
    };
    video_proc.setProcessingChain(chain);

    // Allocate GPU buffers
    size_t gpu_size = WIDTH * HEIGHT * 4;
    uint8_t *input_gpu, *output_gpu;
    cudaMalloc(&input_gpu, gpu_size);
    cudaMalloc(&output_gpu, gpu_size);

    cv::Mat frame;
    int frame_count = 0;

    while (true) {
        // Capture frame
        cap >> frame;
        if (frame.empty()) break;

        // Convert BGR to RGBA
        cv::Mat rgba_frame;
        cv::cvtColor(frame, rgba_frame, cv::COLOR_BGR2RGBA);

        // Upload to GPU
        cudaMemcpy(input_gpu, rgba_frame.data, gpu_size, 
                   cudaMemcpyHostToDevice);

        // Process
        video_proc.enqueueFrame(input_gpu);

        if (video_proc.getProcessedFrame(output_gpu, 100)) {
            // Download result
            uint8_t *host_output = new uint8_t[gpu_size];
            cudaMemcpy(host_output, output_gpu, gpu_size, 
                       cudaMemcpyDeviceToHost);

            // Convert back to OpenCV format
            cv::Mat result(HEIGHT, WIDTH, CV_8UC4, host_output);

            // Display
            cv::imshow("CUDA Vision Stream", result);

            // Performance metrics
            frame_count++;
            if (frame_count % 30 == 0) {
                std::cout << "Processed frames: " << frame_count << "\n";
            }

            delete[] host_output;
        }

        // Exit on 'q' key
        if (cv::waitKey(1) == 'q') break;
    }

    // Cleanup
    cudaFree(input_gpu);
    cudaFree(output_gpu);
    cap.release();
    cv::destroyAllWindows();

    std::cout << "✅ Stream processing complete\n";
    return 0;
}

    📌 Example 3: Feature Detection and Matching¶

    Extract and match SIFT features.

    C++ Example¶

    C++
    #include <jetbot_cuda_vision/feature_extraction.hpp>
#include <opencv2/opencv.hpp>
#include <iostream>
#include <vector>

int main() {
    // Load images
    cv::Mat img1 = cv::imread("image1.jpg", cv::IMREAD_GRAYSCALE);
    cv::Mat img2 = cv::imread("image2.jpg", cv::IMREAD_GRAYSCALE);

    if (img1.empty() || img2.empty()) {
        std::cerr << "Failed to load images\n";
        return 1;
    }

    int width = img1.cols;
    int height = img1.rows;

    // Allocate GPU memory
    uint8_t *gpu_img1, *gpu_img2;
    cudaMalloc(&gpu_img1, width * height);
    cudaMalloc(&gpu_img2, width * height);

    cudaMemcpy(gpu_img1, img1.data, width * height, 
               cudaMemcpyHostToDevice);
    cudaMemcpy(gpu_img2, img2.data, width * height, 
               cudaMemcpyHostToDevice);

    // Extract features
    std::vector<jetbot::Keypoint> kps1, kps2;
    std::vector<jetbot::Descriptor> descs1, descs2;

    std::cout << "Extracting features from image 1...\n";
    jetbot::extractSIFTFeatures(gpu_img1, kps1, descs1, width, height, 0.03f);

    std::cout << "Extracting features from image 2...\n";
    jetbot::extractSIFTFeatures(gpu_img2, kps2, descs2, width, height, 0.03f);

    std::cout << "Image 1 keypoints: " << kps1.size() << "\n";
    std::cout << "Image 2 keypoints: " << kps2.size() << "\n";

    // Simple feature matching (brute force)
    std::vector<cv::DMatch> matches;
    const float MATCH_THRESHOLD = 0.7f;

    for (size_t i = 0; i < descs1.size(); i++) {
        float best_dist = std::numeric_limits<float>::max();
        float second_best_dist = std::numeric_limits<float>::max();
        int best_idx = -1;

        for (size_t j = 0; j < descs2.size(); j++) {
            float dist = jetbot::descriptorDistance(descs1[i], descs2[j]);

            if (dist < best_dist) {
                second_best_dist = best_dist;
                best_dist = dist;
                best_idx = j;
            } else if (dist < second_best_dist) {
                second_best_dist = dist;
            }
        }

        // Lowe's ratio test
        if (best_dist < MATCH_THRESHOLD * second_best_dist) {
            matches.push_back(cv::DMatch(i, best_idx, best_dist));
        }
    }

    std::cout << "Matches found: " << matches.size() << "\n";

    // Draw matches
    cv::Mat match_img;
    cv::drawMatches(img1, convertKeypoints(kps1), 
                    img2, convertKeypoints(kps2), 
                    matches, match_img);

    cv::imwrite("matches.jpg", match_img);
    std::cout << "✅ Matching complete - result saved to matches.jpg\n";

    // Cleanup
    cudaFree(gpu_img1);
    cudaFree(gpu_img2);

    return 0;
}

    📌 Example 4: ROS2 Integration¶

    Use CUDA Vision in ROS2 node.

    ROS2 Node Implementation¶

    C++
    #include <rclcpp/rclcpp.hpp>
#include <sensor_msgs/msg/image.hpp>
#include <cv_bridge/cv_bridge.h>
#include <jetbot_cuda_vision/cuda_processor.hpp>
#include <cuda_runtime.h>

class CUDAVisionNode : public rclcpp::Node {
private:
    rclcpp::Subscription<sensor_msgs::msg::Image>::SharedPtr image_sub_;
    rclcpp::Publisher<sensor_msgs::msg::Image>::SharedPtr image_pub_;
    jetbot::CUDAProcessor processor_;
    uint8_t *input_gpu_, *output_gpu_;

public:
    CUDAVisionNode() 
        : Node("cuda_vision_node"),
          processor_(0, {1280, 720}) {

        // Subscribe to input topic
        image_sub_ = this->create_subscription<sensor_msgs::msg::Image>(
            "/camera/image_raw", 10,
            std::bind(&CUDAVisionNode::imageCallback, this, 
                      std::placeholders::_1));

        // Publish processed image
        image_pub_ = this->create_publisher<sensor_msgs::msg::Image>(
            "/cuda_vision/processed", 10);

        // Allocate GPU memory
        size_t size = 1280 * 720 * 4;
        cudaMalloc(&input_gpu_, size);
        cudaMalloc(&output_gpu_, size);

        RCLCPP_INFO(this->get_logger(), "CUDA Vision Node initialized");
    }

    ~CUDAVisionNode() {
        cudaFree(input_gpu_);
        cudaFree(output_gpu_);
    }

    void imageCallback(const sensor_msgs::msg::Image::SharedPtr msg) {
        // Convert ROS image to OpenCV
        cv::Mat cv_image = cv_bridge::toCvShare(msg, "rgba8")->image;

        // Upload to GPU
        cudaMemcpy(input_gpu_, cv_image.data, 
                   cv_image.total() * cv_image.elemSize(),
                   cudaMemcpyHostToDevice);

        // Process
        jetbot::grayscaleConvert(input_gpu_, output_gpu_, 
                                 1280, 720, jetbot::ImageFormat::RGBA);

        // Download result
        uint8_t *host_output = new uint8_t[1280 * 720];
        cudaMemcpy(host_output, output_gpu_, 1280 * 720,
                   cudaMemcpyDeviceToHost);

        // Convert back to ROS image
        cv::Mat result(720, 1280, CV_8UC1, host_output);
        sensor_msgs::msg::Image::SharedPtr output_msg = 
            cv_bridge::CvImage(msg->header, "mono8", result).toImageMsg();

        image_pub_->publish(*output_msg);

        delete[] host_output;
    }
};

int main(int argc, char *argv[]) {
    rclcpp::init(argc, argv);
    rclcpp::spin(std::make_shared<CUDAVisionNode>());
    rclcpp::shutdown();
    return 0;
}

    Python ROS2 Node¶

    Python
    import rclpy
from rclpy.node import Node
from sensor_msgs.msg import Image
from cv_bridge import CvBridge
import jetbot_cuda_vision as jcv
import numpy as np

class CUDAVisionNode(Node):
    def __init__(self):
        super().__init__('cuda_vision_node')

        # Create processor
        self.processor = jcv.CUDAProcessor(gpu_id=0, width=1280, height=720)
        self.bridge = CvBridge()

        # Subscribe to camera
        self.subscription = self.create_subscription(
            Image, '/camera/image_raw', self.image_callback, 10)

        # Publish processed image
        self.publisher = self.create_publisher(Image, '/cuda_vision/processed', 10)

        self.get_logger().info('CUDA Vision Node initialized')

    def image_callback(self, msg):
        # Convert ROS image to OpenCV
        cv_image = self.bridge.imgmsg_to_cv2(msg, desired_encoding='rgba8')

        # Process with CUDA
        gray = jcv.grayscale_convert(cv_image)

        # Convert back to ROS message
        output_msg = self.bridge.cv2_to_imgmsg(gray, encoding='mono8')
        output_msg.header = msg.header

        self.publisher.publish(output_msg)

def main(args=None):
    rclpy.init(args=args)
    node = CUDAVisionNode()
    rclpy.spin(node)
    node.destroy_node()
    rclpy.shutdown()

if __name__ == '__main__':
    main()

    📌 Example 5: Multi-Stream Processing¶

    Process multiple frames in parallel.

    C++
    #include <jetbot_cuda_vision/cuda_processor.hpp>
#include <cuda_runtime.h>
#include <iostream>
#include <vector>
#include <chrono>

int main() {
    const int WIDTH = 640;
    const int HEIGHT = 480;
    const int NUM_STREAMS = 4;
    const int NUM_FRAMES = 100;

    // Create processor with streaming enabled
    jetbot::CUDAProcessor processor(0, {WIDTH, HEIGHT});
    processor.enableStreaming(NUM_STREAMS);

    // Allocate memory for multiple frames
    std::vector<uint8_t*> input_frames(NUM_STREAMS);
    std::vector<uint8_t*> output_frames(NUM_STREAMS);

    for (int i = 0; i < NUM_STREAMS; i++) {
        cudaMalloc(&input_frames[i], WIDTH * HEIGHT * 4);
        cudaMalloc(&output_frames[i], WIDTH * HEIGHT);
    }

    // Host memory for results
    std::vector<uint8_t*> host_results(NUM_STREAMS);
    for (int i = 0; i < NUM_STREAMS; i++) {
        cudaMallocHost(&host_results[i], WIDTH * HEIGHT);
    }

    auto start = std::chrono::high_resolution_clock::now();

    // Process frames
    for (int f = 0; f < NUM_FRAMES; f++) {
        int stream_idx = f % NUM_STREAMS;

        // Simulate input data
        // In real usage, load actual frame data

        // Enqueue processing
        jetbot::grayscaleConvert(input_frames[stream_idx], 
                                output_frames[stream_idx],
                                WIDTH, HEIGHT, jetbot::ImageFormat::RGBA);

        // Copy result (async)
        cudaMemcpyAsync(host_results[stream_idx], 
                       output_frames[stream_idx],
                       WIDTH * HEIGHT,
                       cudaMemcpyDeviceToHost);
    }

    // Synchronize all streams
    cudaDeviceSynchronize();

    auto end = std::chrono::high_resolution_clock::now();
    auto duration = std::chrono::duration_cast<std::chrono::milliseconds>
                   (end - start).count();

    double fps = (NUM_FRAMES * 1000.0) / duration;
    std::cout << "Processed " << NUM_FRAMES << " frames in " 
              << duration << "ms\n";
    std::cout << "Throughput: " << fps << " FPS\n";

    // Cleanup
    for (int i = 0; i < NUM_STREAMS; i++) {
        cudaFree(input_frames[i]);
        cudaFree(output_frames[i]);
        cudaFreeHost(host_results[i]);
    }

    return 0;
}
