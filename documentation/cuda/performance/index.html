    Performance Optimization Guide¶

    ⚡ Optimization Techniques¶

    1. Memory Optimization¶

    Pinned Memory for Host-Device Transfer¶

    C++
    // Allocate pinned memory for faster CPU-GPU transfer
void* pinned_buffer = jetbot::pinnedMalloc(image_size);

// Transfer is ~2x faster than pageable memory
cudaMemcpy(gpu_buffer, pinned_buffer, size, cudaMemcpyHostToDevice);

// Don't forget to free properly
jetbot::pinnedFree(pinned_buffer);

    Impact: 40-50% faster host-device transfers

    Memory Pool Reuse¶

    C++
    // Pre-allocate memory pool
class MemoryPool {
    std::vector<uint8_t*> available;
    std::vector<uint8_t*> in_use;

public:
    uint8_t* acquire(size_t size) {
        if (!available.empty()) {
            auto ptr = available.back();
            available.pop_back();
            in_use.push_back(ptr);
            return ptr;
        }
        uint8_t* ptr;
        cudaMalloc(&ptr, size);
        in_use.push_back(ptr);
        return ptr;
    }

    void release(uint8_t* ptr) {
        available.push_back(ptr);
        // Remove from in_use
    }
};

    Impact: Eliminates allocation overhead in loops

    2. Computation Optimization¶

    Async Memory Transfers¶

    C++
    // Overlap computation with memory transfer
cudaStream_t stream1, stream2;
cudaStreamCreate(&stream1);
cudaStreamCreate(&stream2);

// Upload next frame while processing current
cudaMemcpyAsync(gpu_input[0], cpu_input[0], size, 
               cudaMemcpyHostToDevice, stream1);
cudaMemcpyAsync(gpu_input[1], cpu_input[1], size, 
               cudaMemcpyHostToDevice, stream2);

// Process
grayscaleConvert(gpu_input[0], gpu_output[0], ..., stream1);

// Download results
cudaMemcpyAsync(cpu_output[0], gpu_output[0], size,
               cudaMemcpyDeviceToHost, stream1);

    Impact: 25-35% throughput improvement

    Kernel Fusion¶

    C++
    // Instead of separate kernels
grayscaleConvert(input, temp, ...);
gaussianBlur(temp, output, ...);

// Use fused kernel when possible
grayscaleAndBlur(input, output, ...);  // Single kernel

    Impact: Reduced memory bandwidth by 40%

    3. Stream Processing¶

    Multiple Concurrent Streams¶

    C++
    const int NUM_STREAMS = 4;
cudaStream_t streams[NUM_STREAMS];

for (int i = 0; i < NUM_STREAMS; i++) {
    cudaStreamCreate(&streams[i]);
}

// Process multiple frames concurrently
for (int frame = 0; frame < total_frames; frame++) {
    int stream_idx = frame % NUM_STREAMS;

    cudaMemcpyAsync(gpu_input[stream_idx], cpu_input[frame], size,
                   cudaMemcpyHostToDevice, streams[stream_idx]);

    processFrame(gpu_input[stream_idx], gpu_output[stream_idx],
                streams[stream_idx]);

    cudaMemcpyAsync(cpu_output[frame], gpu_output[stream_idx], size,
                   cudaMemcpyDeviceToHost, streams[stream_idx]);
}

cudaDeviceSynchronize();

    Impact: 3-4x throughput improvement with 4 streams

    4. Algorithm Selection¶

    Choose Appropriate Kernel Sizes¶

    YAML
    Gaussian Blur Kernel Impact:
Kernel 3×3:   240 FPS (default, good quality)
Kernel 5×5:   160 FPS (better quality, 30% slower)
Kernel 7×7:   100 FPS (excellent quality, 60% slower)

Recommendation:
- Real-time: Use 3×3 kernel
- Quality: Use 5×5 kernel for detection
- Off-line: Use larger kernels for best results

    Pre-compute when Possible¶

    C++
    // Don't recompute for each frame
class FilterCache {
    std::map<float, float*> gaussian_kernels;

public:
    float* getGaussianKernel(float sigma) {
        if (gaussian_kernels.find(sigma) != gaussian_kernels.end()) {
            return gaussian_kernels[sigma];  // Cache hit!
        }
        // Compute and cache
    }
};

    Impact: Eliminates redundant computation

    🔍 Profiling and Monitoring¶

    NVIDIA Profiler (nsys)¶

    Bash
    # Basic profiling
nsys profile -o profile_report ./your_application

# Profile with metrics
nsys profile --gpu-metrics-device all -o report ./app

# Generate report
nsys stats profile_report.nsys-rep

    NVIDIA Compute Profiler (ncu)¶

    Bash
    # Detailed kernel analysis
ncu -o profile_report ./your_application

# With specific metrics
ncu --set full -o report ./app

# View results
ncu --query report_report.ncu-rep

    Custom Performance Monitoring¶

    C++
    #include <chrono>

class PerformanceMonitor {
private:
    std::map<std::string, std::vector<double>> timings;
    std::map<std::string, cudaEvent_t> events;

public:
    void startTimer(const std::string& name) {
        cudaEventCreate(&events[name + "_start"]);
        cudaEventCreate(&events[name + "_stop"]);
        cudaEventRecord(events[name + "_start"]);
    }

    void stopTimer(const std::string& name) {
        cudaEventRecord(events[name + "_stop"]);
        cudaEventSynchronize(events[name + "_stop"]);

        float milliseconds = 0;
        cudaEventElapsedTime(&milliseconds, events[name + "_start"],
                            events[name + "_stop"]);

        timings[name].push_back(milliseconds);
    }

    void printStats(const std::string& name) {
        auto& times = timings[name];
        double avg = 0, min = DBL_MAX, max = DBL_MIN;

        for (auto t : times) {
            avg += t;
            min = std::min(min, (double)t);
            max = std::max(max, (double)t);
        }
        avg /= times.size();

        std::cout << name << ": avg=" << avg << "ms, "
                  << "min=" << min << "ms, max=" << max << "ms\n";
    }
};

    🎯 Optimization Checklist¶

    • Use pinned memory for host-device transfers
    • Pre-allocate GPU memory outside loops
    • Enable streaming for multi-frame processing
    • Use appropriate kernel sizes (start with 3×3)
    • Profile with nsys/ncu to identify bottlenecks
    • Fuse kernels when processing same data
    • Monitor GPU power consumption
    • Use double buffering for continuous streams
    • Keep memory alignment (32 bytes minimum)
    • Batch operations when possible

    🔧 Advanced Tuning¶

    Block and Thread Configuration¶

    C++
    // Current default: 256 threads per block
// For Jetson Orin Nano (Compute Capability 8.7)
#define THREADS_PER_BLOCK 256
#define MAX_BLOCKS 2048

// Optimal often found through profiling
// Try: 128, 256, 512 threads per block

    Memory Coalescing¶

    Ensure global memory accesses are coalesced:

    CUDA
    // Good: Memory coalescing
__global__ void good_kernel(const uint8_t* input, uint8_t* output) {
    int idx = blockIdx.x * blockDim.x + threadIdx.x;
    output[idx] = process(input[idx]);  // Sequential memory access
}

// Bad: Non-coalesced access
__global__ void bad_kernel(const uint8_t* input, uint8_t* output) {
    int idx = (blockIdx.x * blockDim.x + threadIdx.x) * stride;
    output[idx] = process(input[idx]);  // Strided access
}
