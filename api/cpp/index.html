    C++ API Reference¶

    High-performance native C++ API for Isaac ROS Jetbot NanoOWL.

    📚 Overview¶

    The C++ API provides direct access to all robot functionalities with minimal overhead, ideal for real-time control and performance-critical applications.

    Key Features: - ✅ Sub-millisecond latency - ✅ Minimal memory footprint - ✅ Direct hardware access - ✅ Thread-safe operations - ✅ Comprehensive error handling - ✅ CUDA integration support

    📦 Installation¶

    Step 1: Install Dependencies¶

    Bash
    # Install build tools
sudo apt install -y build-essential cmake git

# Install libraries
sudo apt install -y libopencv-dev libeigen3-dev
sudo apt install -y ros-humble-rclcpp ros-humble-geometry2

    Step 2: Clone and Build¶

    Bash
    # Clone repository
git clone https://github.com/TNG-Blue/Isaac_ROS-Jetbot_NanoOWL.git
cd Isaac_ROS-Jetbot_NanoOWL

# Build C++ libraries
mkdir -p build && cd build
cmake -DCMAKE_BUILD_TYPE=Release ..
make -j$(nproc)
sudo make install

    Step 3: Verify Installation¶

    Bash
    # Check headers are installed
ls -la /usr/local/include/jetbot/

# Check libraries
ls -la /usr/local/lib/libjetbot*

    🚀 Quick Start¶

    Hello World: Move Forward¶

    C++
    #include <jetbot/motor_controller.hpp>
#include <chrono>
#include <thread>

int main() {
    // Create motor controller
    jetbot::MotorController motor;

    // Move forward at 0.5 m/s
    motor.setVelocity(0.5, 0.0);  // linear_x=0.5, angular_z=0.0

    // Sleep for 3 seconds
    std::this_thread::sleep_for(std::chrono::seconds(3));

    // Stop
    motor.setVelocity(0.0, 0.0);

    return 0;
}

    Compile and Run¶

    Bash
    # Create CMakeLists.txt
cat > CMakeLists.txt << 'EOF'
cmake_minimum_required(VERSION 3.10)
project(jetbot_example)

set(CMAKE_CXX_STANDARD 17)

find_package(jetbot REQUIRED)

add_executable(hello_world hello_world.cpp)
target_link_libraries(hello_world jetbot::jetbot)
EOF

# Build
mkdir build && cd build
cmake ..
make

# Run
./hello_world

    📖 Core API Reference¶

    1. Motor Controller¶

    C++
    #include <jetbot/motor_controller.hpp>

namespace jetbot {
    class MotorController {
    public:
        // Constructor
        MotorController();

        // Set velocity (m/s and rad/s)
        void setVelocity(double linear_x, double angular_z);

        // Get current velocity
        std::pair<double, double> getVelocity() const;

        // Set individual motor speeds (0-255)
        void setMotorSpeed(int left_speed, int right_speed);

        // Get motor state
        struct MotorState {
            double left_velocity;
            double right_velocity;
            double left_current;
            double right_current;
        };
        MotorState getMotorState() const;

        // Emergency stop
        void emergencyStop();

        // Get status
        bool isHealthy() const;
    };
}

    Example:

    C++
    jetbot::MotorController motor;

// Set velocity
motor.setVelocity(0.3, 0.1);  // Move forward + slight turn

// Read state
auto state = motor.getMotorState();
std::cout << "Left velocity: " << state.left_velocity << " m/s" << std::endl;

// Check health
if (!motor.isHealthy()) {
    motor.emergencyStop();
}

    2. Camera Interface¶

    C++
    #include <jetbot/camera.hpp>
#include <opencv2/opencv.hpp>

namespace jetbot {
    class Camera {
    public:
        // Constructor
        Camera(int width = 1280, int height = 720, int fps = 30);

        // Get frame
        cv::Mat getFrame();

        // Get with timeout
        bool getFrame(cv::Mat& frame, int timeout_ms = 100);

        // Camera info
        struct CameraInfo {
            int width;
            int height;
            cv::Mat camera_matrix;  // K matrix
            cv::Mat distortion;     // Distortion coefficients
        };
        CameraInfo getCameraInfo() const;

        // Calibration
        void loadCalibration(const std::string& filename);
        cv::Mat undistort(const cv::Mat& frame);

        // Properties
        void setExposure(int exposure);
        void setGain(int gain);
        void setAutoWhiteBalance(bool enabled);
    };
}

    Example:

    C++
    jetbot::Camera camera(1280, 720, 30);

// Load calibration
camera.loadCalibration("camera_calibration.yaml");

// Capture and process
cv::Mat frame = camera.getFrame();
cv::Mat undistorted = camera.undistort(frame);
cv::imshow("Camera Feed", undistorted);

    3. IMU Sensor¶

    C++
    #include <jetbot/imu_sensor.hpp>
#include <Eigen/Dense>

namespace jetbot {
    class IMUSensor {
    public:
        // Constructor
        IMUSensor();

        // Data structure
        struct IMUData {
            Eigen::Vector3d acceleration;  // m/s²
            Eigen::Vector3d angular_velocity;  // rad/s
            Eigen::Vector3d magnetic_field;  // Gauss
            double temperature;  // °C
        };

        // Get latest data
        IMUData getLatestData();

        // Stream with callback
        using Callback = std::function<void(const IMUData&)>;
        void registerCallback(Callback cb);

        // Calibration
        void calibrateAccelerometer();
        void calibrateGyroscope();
        void calibrateMagnetometer();

        // Get calibration
        Eigen::Matrix3d getAccelCalibration() const;
        Eigen::Vector3d getGyroOffset() const;
    };
}

    Example:

    C++
    jetbot::IMUSensor imu;

// Get single measurement
auto data = imu.getLatestData();
std::cout << "Accel X: " << data.acceleration.x() << std::endl;

// Stream with callback
imu.registerCallback([](const jetbot::IMUSensor::IMUData& data) {
    std::cout << "Gyro Z: " << data.angular_velocity.z() << std::endl;
});

    4. Object Detection¶

    C++
    #include <jetbot/detector.hpp>
#include <opencv2/opencv.hpp>

namespace jetbot {
    class ObjectDetector {
    public:
        // Constructor
        ObjectDetector(const std::string& model_path);

        // Detection result
        struct Detection {
            int class_id;
            std::string class_name;
            float confidence;
            cv::Rect2f bbox;  // Normalized coordinates [0, 1]
        };

        // Run detection
        std::vector<Detection> detect(const cv::Mat& frame);

        // Async detection
        void detectAsync(
            const cv::Mat& frame,
            std::function<void(const std::vector<Detection>&)> callback
        );

        // Performance
        struct Stats {
            double inference_time_ms;
            int fps;
            double cpu_usage;
            double memory_usage_mb;
        };
        Stats getStats() const;

        // Configuration
        void setConfidenceThreshold(float threshold);
        void setNMSThreshold(float threshold);
    };
}

    Example:

    C++
    jetbot::ObjectDetector detector("models/yolov8n.engine");

cv::Mat frame = camera.getFrame();
auto detections = detector.detect(frame);

for (const auto& det : detections) {
    std::cout << det.class_name << ": " << det.confidence << std::endl;

    // Draw bbox
    int x = det.bbox.x * frame.cols;
    int y = det.bbox.y * frame.rows;
    int w = det.bbox.width * frame.cols;
    int h = det.bbox.height * frame.rows;

    cv::rectangle(frame, cv::Rect(x, y, w, h), cv::Scalar(0, 255, 0), 2);
}

cv::imshow("Detections", frame);

    🔧 Advanced Topics¶

    Thread-Safe Operations¶

    C++
    #include <jetbot/motor_controller.hpp>
#include <thread>
#include <mutex>

int main() {
    jetbot::MotorController motor;
    std::mutex motor_mutex;

    // Control thread
    std::thread control_thread([&]() {
        while (true) {
            {
                std::lock_guard<std::mutex> lock(motor_mutex);
                motor.setVelocity(0.5, 0.0);
            }
            std::this_thread::sleep_for(std::chrono::milliseconds(100));
        }
    });

    // Monitoring thread
    std::thread monitor_thread([&]() {
        while (true) {
            {
                std::lock_guard<std::mutex> lock(motor_mutex);
                auto state = motor.getMotorState();
                std::cout << "Velocity: " << state.left_velocity << std::endl;
            }
            std::this_thread::sleep_for(std::chrono::milliseconds(500));
        }
    });

    // Wait for threads
    control_thread.join();
    monitor_thread.join();

    return 0;
}

    Custom Callbacks¶

    C++
    jetbot::IMUSensor imu;

// Register lambda callback
imu.registerCallback([](const jetbot::IMUSensor::IMUData& data) {
    // Process IMU data at 200 Hz
    static int count = 0;
    if (++count % 10 == 0) {
        std::cout << "Gravity: " << data.acceleration.norm() << " m/s²" << std::endl;
    }
});

// Keep running
while (true) {
    std::this_thread::sleep_for(std::chrono::milliseconds(100));
}

    📊 Error Handling¶

    C++
    #include <jetbot/exceptions.hpp>

try {
    jetbot::MotorController motor;
    motor.setVelocity(0.5, 0.0);
}
catch (const jetbot::MotorException& e) {
    std::cerr << "Motor error: " << e.what() << std::endl;
}
catch (const jetbot::HardwareException& e) {
    std::cerr << "Hardware error: " << e.what() << std::endl;
}
catch (const std::exception& e) {
    std::cerr << "Error: " << e.what() << std::endl;
}

    🎯 Complete Example: Autonomous Patrol¶

    C++
    #include <jetbot/motor_controller.hpp>
#include <jetbot/camera.hpp>
#include <jetbot/detector.hpp>
#include <chrono>
#include <cmath>

class AutonomousPatrol {
private:
    jetbot::MotorController motor;
    jetbot::Camera camera;
    jetbot::ObjectDetector detector;

public:
    AutonomousPatrol() 
        : camera(1280, 720, 30),
          detector("models/yolov8n.engine") {}

    void run() {
        double distance = 0.0;
        const double target_distance = 5.0;  // 5 meters

        while (distance < target_distance) {
            // Move forward
            motor.setVelocity(0.3, 0.0);

            // Detect obstacles
            cv::Mat frame = camera.getFrame();
            auto detections = detector.detect(frame);

            // Check for obstacles
            bool obstacle_detected = false;
            for (const auto& det : detections) {
                if (det.class_name == "person" && det.confidence > 0.7) {
                    obstacle_detected = true;
                    break;
                }
            }

            if (obstacle_detected) {
                std::cout << "Obstacle detected! Turning..." << std::endl;
                turnRight(90);  // Turn 90 degrees
            }

            // Update distance (approximate from time)
            distance += 0.3 * 0.1;  // 0.3 m/s * 0.1 s

            std::this_thread::sleep_for(std::chrono::milliseconds(100));
        }

        // Stop
        motor.setVelocity(0.0, 0.0);
        std::cout << "Patrol complete!" << std::endl;
    }

private:
    void turnRight(double degrees) {
        double radians = degrees * M_PI / 180.0;
        double angular_velocity = 1.0;  // rad/s
        double duration = radians / angular_velocity;

        motor.setVelocity(0.0, -angular_velocity);
        std::this_thread::sleep_for(
            std::chrono::milliseconds((int)(duration * 1000))
        );
        motor.setVelocity(0.0, 0.0);
    }
};

int main() {
    try {
        AutonomousPatrol patrol;
        patrol.run();
    }
    catch (const std::exception& e) {
        std::cerr << "Error: " << e.what() << std::endl;
        return 1;
    }
    return 0;
}

    📋 Common Patterns¶

    Pattern 1: Control Loop¶

    C++
    auto start = std::chrono::steady_clock::now();
const double target_duration = 10.0;  // seconds

while (true) {
    auto now = std::chrono::steady_clock::now();
    double elapsed = std::chrono::duration<double>(now - start).count();

    if (elapsed > target_duration) break;

    // Control logic here
    motor.setVelocity(0.5, 0.0);

    // Sleep to maintain frequency
    std::this_thread::sleep_for(std::chrono::milliseconds(10));
}

    Pattern 2: State Machine¶

    C++
    enum class State {
    IDLE,
    MOVING,
    DETECTING,
    TURNING,
    STOPPED
};

State current_state = State::IDLE;

while (true) {
    switch (current_state) {
        case State::IDLE:
            current_state = State::MOVING;
            break;
        case State::MOVING:
            motor.setVelocity(0.5, 0.0);
            current_state = State::DETECTING;
            break;
        case State::DETECTING:
            // Detection logic
            current_state = State::MOVING;
            break;
        case State::STOPPED:
            motor.setVelocity(0.0, 0.0);
            return 0;
    }
}

    📊 Performance Tips¶

    1. Reuse objects - Create motors/cameras once, reuse them
    2. Minimize allocations - Pre-allocate vectors/matrices
    3. Use async operations - Non-blocking detection
    4. Thread pooling - For parallel processing
    5. Profile often - Use tools like perf
