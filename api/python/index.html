    API

    Python API Reference¶

    Easy-to-use Python API for Isaac ROS Jetbot NanoOWL.

    📚 Overview¶

    The Python API provides a high-level, Pythonic interface to all robot functionalities, ideal for rapid development, prototyping, and scripting.

    Key Features: - ✅ Simple, intuitive API - ✅ Fast development cycles - ✅ Jupyter notebook support - ✅ Extensive examples - ✅ Great documentation - ✅ Good performance for most applications

    📦 Installation¶

    Method 1: pip (Recommended)¶

    # Install from PyPI
pip install jetbot

# Or with conda
conda install -c conda-forge jetbot

    Method 2: From Source¶

    # Clone repository
git clone https://github.com/TNG-Blue/Isaac_ROS-Jetbot_NanoOWL.git
cd Isaac_ROS-Jetbot_NanoOWL

# Install in development mode
pip install -e ./python

    Verify Installation¶

    python3 -c "import jetbot; print(jetbot.__version__)"
# Output: 1.3.0

    Hello World: Move Forward¶

    from jetbot import MotorController

    📖 Core API Reference¶

    1. Motor Controller¶

    from jetbot import MotorController

class MotorController:
    """Control robot movement"""

    def __init__(self):
        """Initialize motor controller"""

    def set_velocity(self, linear_x: float, angular_z: float) -> None:
        """
        Set robot velocity

        Args:
            linear_x: Forward velocity in m/s (-1.0 to 1.0)
            angular_z: Angular velocity in rad/s (-3.14 to 3.14)

        Examples:
            motor.set_velocity(0.5, 0.0)  # Move forward
            motor.set_velocity(0.0, 1.0)  # Rotate left
            motor.set_velocity(0.5, 0.5)  # Move forward + turn
        """

    def get_velocity(self) -> tuple:
        """
        Get current velocity

        Returns:
            (linear_x, angular_z) tuple
        """

    def set_motor_speed(self, left_speed: int, right_speed: int) -> None:
        """
        Set individual motor speeds (0-255)

        Args:
            left_speed: Left motor speed (0-255)
            right_speed: Right motor speed (0-255)
        """

    def get_motor_state(self) -> dict:
        """
        Get motor state

        Returns:
            {
                'left_velocity': float,
                'right_velocity': float,
                'left_current': float,
                'right_current': float
            }
        """

    def emergency_stop(self) -> None:
        """Perform emergency stop"""

    def is_healthy(self) -> bool:
        """Check if motors are healthy"""

    Example:

    motor = MotorController()

# Move forward
motor.set_velocity(0.3, 0.0)
time.sleep(2)

# Turn and move
motor.set_velocity(0.3, 0.5)
time.sleep(1)

# Check status
velocity = motor.get_velocity()
state = motor.get_motor_state()
print(f"Velocity: {velocity}")
print(f"Motor state: {state}")

# Emergency stop
if not motor.is_healthy():
    motor.emergency_stop()

    2. Camera Interface¶

    from jetbot import Camera
import cv2

class Camera:
    """Access robot camera"""

    def __init__(self, width: int = 1280, height: int = 720, fps: int = 30):
        """Initialize camera"""

    def read(self) -> np.ndarray:
        """
        Get camera frame

        Returns:
            BGR image as numpy array (H × W × 3)
        """

    def get_frame(self, timeout_ms: int = 100) -> tuple:
        """
        Get frame with timeout

        Returns:
            (success: bool, frame: np.ndarray)
        """

    def get_camera_info(self) -> dict:
        """
        Get camera calibration info

        Returns:
            {
                'width': int,
                'height': int,
                'camera_matrix': np.ndarray (3×3),
                'distortion': np.ndarray (5,)
            }
        """

    def load_calibration(self, filename: str) -> None:
        """Load calibration from file"""

    def undistort(self, frame: np.ndarray) -> np.ndarray:
        """Remove lens distortion from frame"""

    def set_exposure(self, exposure: int) -> None:
        """Set exposure (-13 to 0)"""

    def set_gain(self, gain: int) -> None:
        """Set gain (0 to 100)"""

    Example:

    camera = Camera(1280, 720, 30)

# Load calibration
camera.load_calibration("camera_calibration.yaml")

# Capture frame
success, frame = camera.get_frame(timeout_ms=100)

if success:
    # Undistort
    undistorted = camera.undistort(frame)

    # Display
    cv2.imshow("Camera", undistorted)
    cv2.waitKey(1)

# Get calibration info
info = camera.get_camera_info()
K = info['camera_matrix']
dist = info['distortion']

    3. Object Detection¶

    from jetbot import Detector

class Detector:
    """Run object detection"""

    def __init__(self, model_path: str):
        """
        Initialize detector

        Args:
            model_path: Path to model file (.engine, .onnx, .pt)
        """

    def detect(self, frame: np.ndarray) -> list:
        """
        Run detection on frame

        Returns:
            List of detections:
            [
                {
                    'class_id': int,
                    'class_name': str,
                    'confidence': float,
                    'bbox': (x, y, w, h)  # Normalized [0, 1]
                },
                ...
            ]
        """

    async def detect_async(self, frame: np.ndarray, callback) -> None:
        """Asynchronous detection"""

    def get_stats(self) -> dict:
        """
        Get performance statistics

        Returns:
            {
                'inference_time_ms': float,
                'fps': int,
                'cpu_usage': float,
                'memory_usage_mb': float
            }
        """

    def set_confidence_threshold(self, threshold: float) -> None:
        """Set detection confidence threshold (0.0 to 1.0)"""

    Example:

    detector = Detector("models/yolov8n.engine")

camera = Camera()
success, frame = camera.read()

if success:
    detections = detector.detect(frame)

    for det in detections:
        print(f"{det['class_name']}: {det['confidence']:.2f}")

        # Draw bounding box
        x, y, w, h = det['bbox']
        x1 = int(x * frame.shape[1])
        y1 = int(y * frame.shape[0])
        x2 = int((x + w) * frame.shape[1])
        y2 = int((y + h) * frame.shape[0])

        cv2.rectangle(frame, (x1, y1), (x2, y2), (0, 255, 0), 2)
        cv2.putText(frame, det['class_name'], (x1, y1-10),
                   cv2.FONT_HERSHEY_SIMPLEX, 0.5, (0, 255, 0), 2)

    cv2.imshow("Detections", frame)

    4. IMU Sensor¶

    from jetbot import IMUSensor
import numpy as np

class IMUSensor:
    """Access IMU (accelerometer + gyroscope)"""

    def __init__(self):
        """Initialize IMU"""

    def read(self) -> dict:
        """
        Get latest IMU data

        Returns:
            {
                'acceleration': np.array([x, y, z]),  # m/s²
                'angular_velocity': np.array([x, y, z]),  # rad/s
                'temperature': float  # °C
            }
        """

    def register_callback(self, callback) -> None:
        """
        Register callback for real-time updates

        Example:
            def on_imu_data(data):
                print(f"Accel: {data['acceleration']}")

            imu.register_callback(on_imu_data)
        """

    def calibrate_accelerometer(self) -> None:
        """Calibrate accelerometer (keep still)"""

    def calibrate_gyroscope(self) -> None:
        """Calibrate gyroscope (keep still)"""

    def get_calibration(self) -> dict:
        """Get current calibration values"""

    Example:

    imu = IMUSensor()

# Get single reading
data = imu.read()
print(f"Acceleration: {data['acceleration']}")
print(f"Angular velocity: {data['angular_velocity']}")

# Stream with callback
def on_imu_update(data):
    accel = np.linalg.norm(data['acceleration'])
    print(f"Total acceleration: {accel:.2f} m/s²")

imu.register_callback(on_imu_update)

# Keep running
import time
time.sleep(10)

    🎯 Complete Example: Obstacle Avoidance¶

    from jetbot import MotorController, Camera, Detector
import cv2
import time

class ObstacleAvoider:
    def __init__(self):
        self.motor = MotorController()
        self.camera = Camera()
        self.detector = Detector("models/yolov8n.engine")
        self.detector.set_confidence_threshold(0.7)

    def run(self):
        print("Starting obstacle avoidance...")
        running_time = 0

        while running_time < 30:  # Run for 30 seconds
            start = time.time()

            # Get camera frame
            success, frame = self.camera.read()
            if not success:
                continue

            # Detect obstacles
            detections = self.detector.detect(frame)

            # Check for obstacles in front
            obstacle_in_front = False
            for det in detections:
                if det['class_name'] == 'person':
                    # Center of frame is around x=0.5
                    x, w = det['bbox'][0], det['bbox'][2]
                    if 0.25 < x + w/2 < 0.75:  # In center
                        obstacle_in_front = True
                        break

            if obstacle_in_front:
                print("Obstacle detected! Turning...")
                self.motor.set_velocity(0.0, 1.0)  # Turn left
                time.sleep(0.5)
            else:
                print("Moving forward...")
                self.motor.set_velocity(0.3, 0.0)  # Move forward

            # Display frame
            cv2.imshow("Obstacle Avoidance", frame)
            if cv2.waitKey(1) & 0xFF == ord('q'):
                break

            running_time += time.time() - start

        # Stop
        self.motor.set_velocity(0.0, 0.0)
        cv2.destroyAllWindows()
        print("Done!")

if __name__ == "__main__":
    avoider = ObstacleAvoider()
    avoider.run()

    Motion Control¶

    from jetbot import MotorController
from ipywidgets import FloatSlider, Output
import ipywidgets as widgets

motor = MotorController()

# Create sliders
speed_slider = FloatSlider(description='Speed:', min=-1, max=1, value=0.5)
turn_slider = FloatSlider(description='Turn:', min=-3.14, max=3.14, value=0)

# Update motor when sliders change
def on_speed_change(change):
    motor.set_velocity(change.new, turn_slider.value)

def on_turn_change(change):
    motor.set_velocity(speed_slider.value, change.new)

speed_slider.observe(on_speed_change, names='value')
turn_slider.observe(on_turn_change, names='value')

display(speed_slider, turn_slider)

    🔄 Async Operations¶

    import asyncio
from jetbot import Detector, Camera

async def detect_continuously():
    detector = Detector("models/yolov8n.engine")
    camera = Camera()

    while True:
        success, frame = camera.read()
        if success:
            # Non-blocking detection
            detections = await detector.detect_async(frame)
            print(f"Found {len(detections)} objects")

        await asyncio.sleep(0.1)

# Run async loop
asyncio.run(detect_continuously())

    # Tip 1: Reuse objects

# Tip 2: Pre-allocate if using numpy
import numpy as np
frames = np.zeros((10, 720, 1280, 3), dtype=np.uint8)

# Tip 3: Use async for non-blocking operations
async def process_frames():
    while True:
        detections = await detector.detect_async(frame)
        # Process detections
        await asyncio.sleep(0.01)

# Tip 4: Profile your code
import cProfile
cProfile.run('my_function()')

# Tip 5: Use generators for large datasets
def frame_generator():

    📋 Common Patterns¶

    Pattern 1: Simple Loop¶

    motor = MotorController()
camera = Camera()
detector = Detector("models/yolov8n.engine")

while True:
    # Read camera
    success, frame = camera.read()
    if not success:
        continue

    # Process
    detections = detector.detect(frame)

    # Control based on detections
    if len(detections) > 0:
        motor.set_velocity(0.0, 0.0)
    else:
        motor.set_velocity(0.3, 0.0)

    class RobotController:

            self.process_detections(detections)

controller = RobotController()
controller.run()
