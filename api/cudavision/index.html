        🚀 CudaVision API Reference¶

        Professional CUDA-accelerated image processing library with unified Python and C++ interfaces for high-performance computer vision applications.

        🎯 CudaVision Overview

        CudaVision provides GPU-accelerated image processing with 15-20x performance improvements over CPU implementations. Designed for real-time applications, professional photography, and computer vision pipelines.

        🏗️ API Architecture¶

        🐍 Python API Layer

        High-level Python interface with NumPy integration

        • Unified CudaVisionCore class
        • Automatic memory management
        • Context managers support
        • Exception handling

        🔗 C/C++ Native Layer

        High-performance native API for C++ applications

        • Direct CUDA kernel access
        • Manual memory control
        • Zero-copy operations
        • Real-time processing

        ⚡ CUDA Kernel Layer

        Optimized GPU kernels for maximum performance

        • Memory coalescing
        • Shared memory usage
        • Async execution
        • Multi-stream processing

        🖥️ Hardware Abstraction

        Cross-platform GPU compatibility

        • CUDA Compute 3.0+
        • Memory pool management
        • Error recovery
        • Performance monitoring

        class CudaVisionCore:
    """
    Unified interface for all CudaVision functionality with automatic resource management.

    Features:
        - Context manager support
        - Automatic GPU memory management
        - Performance monitoring
        - Error handling and recovery
        - Multi-stream processing

    Example:
        >>> with CudaVisionCore() as processor:
        ...     result = processor.enhance_image(image, mode='professional')
        ...     stats = processor.get_performance_stats()
        ...     print(f"Processing time: {stats['total_time_ms']:.2f}ms")
    """

    def __init__(self, device_id: int = 0, enable_profiling: bool = False):
        """
        Initialize CudaVision processor.

        Args:
            device_id: CUDA device ID to use
            enable_profiling: Enable detailed performance profiling
        """

    def __enter__(self) -> 'CudaVisionCore':
        """Context manager entry."""
        self.initialize()
        return self

    def __exit__(self, exc_type, exc_val, exc_tb):
        """Context manager exit with cleanup."""
        self.cleanup()

    def enhance_image(
        self, 
        image: np.ndarray, 
        mode: str = 'auto',
        **kwargs
    ) -> np.ndarray:
        """
        Enhanced image processing with automatic optimization.

        Args:
            image: Input image
            mode: Processing mode ('auto', 'professional', 'fast')
            **kwargs: Additional processing parameters

        Returns:
            np.ndarray: Processed image
        """

    def get_performance_stats(self) -> Dict[str, Any]:
        """
        Get detailed performance statistics.

        Returns:
            Dict containing:
                - processing_time_ms: Total processing time
                - gpu_utilization: GPU usage percentage  
                - memory_usage_mb: GPU memory consumption
                - throughput_fps: Frames per second capability
                - kernel_times: Individual kernel timing
        """

    def get_capabilities(self) -> Dict[str, Any]:
        """
        Get system capabilities and supported features.

        Returns:
            Dict containing:
                - cuda_version: CUDA runtime version
                - gpu_name: Graphics card name
                - compute_capability: CUDA compute capability
                - memory_total_mb: Total GPU memory
                - supported_formats: Supported image formats
        """

        /**

typedef struct {
    uint32_t width;        ///< Image width in pixels
    uint32_t height;       ///< Image height in pixels
    uint32_t channels;     ///< Number of color channels
    uint32_t stride;       ///< Row stride in bytes
} ImageInfo;

/**
 * RGB color structure
 */
typedef struct {
    uint8_t r, g, b;       ///< Red, green, blue components
} RGB;

/**
 * Camera processing configuration
 */
typedef struct {
    uint32_t width, height;              ///< Frame dimensions
    CudaVisionFormat format;             ///< Pixel format
    CudaVisionProcessingMode mode;       ///< Processing mode
    bool enable_noise_reduction;         ///< Enable denoising
    bool enable_sharpening;              ///< Enable sharpening
    float noise_reduction_strength;      ///< Denoising strength (0.0-1.0)
    float sharpness;                     ///< Sharpening amount (0.0-2.0)
    float white_balance_temp;            ///< Color temperature (2000-10000K)
} CudaVisionCameraConfig;

/**
 * Performance statistics
 */
typedef struct {
    float processing_time_ms;            ///< Total processing time
    float gpu_utilization;               ///< GPU usage percentage
    uint32_t memory_used_mb;             ///< GPU memory consumption
    float throughput_fps;                ///< Processing throughput
} CudaVisionCameraStats;

from cudavision.exceptions import *

try:
    result = cv.auto_enhance_image(image, mode='professional')
except CudaNotAvailableError:
    print("CUDA not available, falling back to CPU processing")
    result = cv.cpu_enhance_image(image)  # CPU fallback
except OutOfMemoryError:
    print("Insufficient GPU memory, reducing image size")
    small_image = cv.resize_image(image, scale=0.5)
    result = cv.auto_enhance_image(small_image, mode='professional')
except InvalidImageFormatError as e:
    print(f"Invalid image format: {e}")
    result = None

class CustomImagePipeline:
    """Custom image processing pipeline with CudaVision."""

    def __init__(self):
        self.processor = cv.CudaVisionCore(enable_profiling=True)
        self.processor.initialize()

    def process_portrait(self, image: np.ndarray) -> np.ndarray:
        """Portrait-specific processing pipeline."""
        # Step 1: Noise reduction with skin tone preservation
        denoised = self.processor.denoise_image(image, strength=0.3)

        # Step 2: Gentle sharpening for facial features
        sharpened = self.processor.sharpen_image(denoised, amount=0.4)

        # Step 3: Skin tone enhancement
        enhanced = self.processor.enhance_skin_tones(sharpened)

        # Step 4: Eye and teeth whitening
        final = self.processor.selective_color_enhancement(
            enhanced, 
            enhance_whites=True, 
            brighten_eyes=True
        )

        return final

    def process_landscape(self, image: np.ndarray) -> np.ndarray:
        """Landscape-specific processing pipeline."""
        # Step 1: HDR tone mapping for sky detail
        tone_mapped = self.processor.apply_tone_mapping(image, method='reinhard')

        # Step 2: Vibrance enhancement for natural colors
        vibrant = self.processor.adjust_vibrance(tone_mapped, amount=0.6)

        # Step 3: Clarity enhancement for distant details
        clear = self.processor.enhance_clarity(vibrant, strength=0.5)

        return clear

    def get_pipeline_stats(self) -> dict:
        """Get detailed pipeline performance statistics."""
        return self.processor.get_performance_stats()

class TestCudaVisionCore(unittest.TestCase):
    """Comprehensive test suite for CudaVision functionality."""

    def setUp(self):
        """Set up test fixtures."""
        self.test_image = np.random.randint(0, 255, (480, 640, 3), dtype=np.uint8)
        self.processor = cv.CudaVisionCore()
        self.processor.initialize()

    def tearDown(self):
        """Clean up after tests."""
        self.processor.cleanup()

        """Test RGB to grayscale conversion."""

        self.assertEqual(gray.dtype, np.uint8)

        self.assertTrue(np.all(gray >= 0))
        self.assertTrue(np.all(gray <= 255))

            self.test_image,
            background_color=(0, 255, 0),
            threshold=30
        )

        # Verify output shape matches input
        self.assertEqual(result.shape, self.test_image.shape)
        self.assertEqual(result.dtype, self.test_image.dtype)

        """Test that performance requirements are met."""
        import time

        start_time = time.time()
        enhanced = self.processor.enhance_image(self.test_image, mode='fast')

        # Verify processing time is reasonable for VGA image
        self.assertLess(processing_time, 50)  # Should be < 50ms for VGA

        # Verify output quality
        self.assertEqual(enhanced.shape, self.test_image.shape)
        self.assertFalse(np.array_equal(enhanced, self.test_image))  # Should be different

        """Test GPU memory management."""
        initial_stats = self.processor.get_memory_stats()

        # Process multiple images
        for _ in range(10):
            result = self.processor.enhance_image(self.test_image)

        final_stats = self.processor.get_memory_stats()

        # Memory usage should not continuously grow
        memory_growth = final_stats['used_mb'] - initial_stats['used_mb']
        self.assertLess(memory_growth, 100)  # Less than 100MB growth

        import cudavision as cv
import time
import numpy as np
import matplotlib.pyplot as plt

def benchmark_cudavision():
    """Comprehensive performance benchmark suite."""

    # Test configurations
    resolutions = [
        (640, 480, "VGA"),
        (1280, 720, "HD"),
        (1920, 1080, "FHD"),
        (3840, 2160, "4K")
    ]

    operations = [
        ('rgb_to_grayscale', cv.rgb_to_grayscale),
        ('remove_background', lambda img: cv.remove_background(img, (0, 255, 0), 30)),
        ('auto_enhance', lambda img: cv.auto_enhance_image(img, mode='auto')),
        ('denoise_image', lambda img: cv.denoise_image(img, strength=0.5))
    ]

    results = {}

    with cv.CudaVisionCore(enable_profiling=True) as processor:
        for width, height, res_name in resolutions:
            print(f"\nBenchmarking {res_name} ({width}x{height})...")
            test_image = np.random.randint(0, 255, (height, width, 3), dtype=np.uint8)

            results[res_name] = {}

            for op_name, op_func in operations:
                # Warm-up runs
                for _ in range(3):
                    _ = op_func(test_image)

                # Benchmark runs
                times = []
                for _ in range(10):
                    start_time = time.perf_counter()
                    result = op_func(test_image)
                    end_time = time.perf_counter()
                    times.append((end_time - start_time) * 1000)  # Convert to ms

                avg_time = np.mean(times)
                std_time = np.std(times)
                fps = 1000.0 / avg_time

                results[res_name][op_name] = {
                    'avg_time_ms': avg_time,
                    'std_time_ms': std_time,
                    'fps': fps
                }

                print(f"  {op_name}: {avg_time:.2f}±{std_time:.2f}ms ({fps:.1f} FPS)")

    return results

def plot_benchmark_results(results):
    """Plot benchmark results."""
    fig, axes = plt.subplots(2, 2, figsize=(15, 10))
    fig.suptitle('CudaVision Performance Benchmarks', fontsize=16)

    # Processing time comparison
    ax1 = axes[0, 0]
    operations = list(results['VGA'].keys())
    resolutions = list(results.keys())

    x = np.arange(len(operations))
    width = 0.2

    for i, res in enumerate(resolutions):
        times = [results[res][op]['avg_time_ms'] for op in operations]
        ax1.bar(x + i*width, times, width, label=res)

    ax1.set_xlabel('Operations')
    ax1.set_ylabel('Processing Time (ms)')
    ax1.set_title('Processing Time by Resolution')
    ax1.set_xticks(x + width * 1.5)
    ax1.set_xticklabels(operations, rotation=45)
    ax1.legend()
    ax1.set_yscale('log')

    # FPS comparison
    ax2 = axes[0, 1]
    for i, res in enumerate(resolutions):
        fps_values = [results[res][op]['fps'] for op in operations]
        ax2.bar(x + i*width, fps_values, width, label=res)

    ax2.set_xlabel('Operations')
    ax2.set_ylabel('Frames Per Second')
    ax2.set_title('Throughput by Resolution')
    ax2.set_xticks(x + width * 1.5)
    ax2.set_xticklabels(operations, rotation=45)
    ax2.legend()
    ax2.set_yscale('log')

    # Speedup analysis
    ax3 = axes[1, 0]
    baseline_times = [results['VGA'][op]['avg_time_ms'] for op in operations]
    for res in resolutions[1:]:  # Skip VGA as baseline
        speedups = [baseline_times[i] / results[res][op]['avg_time_ms'] 
                   for i, op in enumerate(operations)]
        ax3.plot(operations, speedups, 'o-', label=f'{res} vs VGA')

    ax3.set_xlabel('Operations')
    ax3.set_ylabel('Speedup Factor')
    ax3.set_title('Resolution Scaling Analysis')
    ax3.legend()
    ax3.grid(True)

    # Memory efficiency
    ax4 = axes[1, 1]
    memory_usage = [2.8, 6.2, 24.9, 99.6]  # MB for VGA, HD, FHD, 4K
    pixel_counts = [0.3, 0.9, 2.1, 8.3]     # Megapixels

    ax4.scatter(pixel_counts, memory_usage, s=100, alpha=0.7)
    for i, res in enumerate(resolutions):
        ax4.annotate(res, (pixel_counts[i], memory_usage[i]), 
                    xytext=(5, 5), textcoords='offset points')

    ax4.set_xlabel('Image Size (Megapixels)')
    ax4.set_ylabel('Memory Usage (MB)')
    ax4.set_title('Memory Usage vs Image Size')
    ax4.grid(True)

    plt.tight_layout()
    plt.savefig('cudavision_benchmarks.png', dpi=300, bbox_inches='tight')
    plt.show()

# Run benchmarks
if __name__ == '__main__':
    benchmark_results = benchmark_cudavision()
    plot_benchmark_results(benchmark_results)

class PerformanceOptimizer:
    """Advanced performance optimization utilities."""

    @staticmethod
    def optimize_for_batch_processing():
        """Configure CudaVision for optimal batch processing."""
        config = {
            'memory_pool_size_mb': 2048,        # Large memory pool
            'enable_memory_recycling': True,     # Reuse allocations
            'max_concurrent_streams': 8,         # Parallel streams
            'prefer_texture_memory': True,       # Use texture cache
            'enable_async_transfers': True       # Async host-device transfers
        }
        return cv.CudaVisionCore(config=config)

    @staticmethod
    def optimize_for_real_time():
        """Configure for real-time processing with minimal latency."""
        config = {
            'memory_pool_size_mb': 512,          # Smaller pool for lower latency
            'enable_memory_recycling': True,     # Reuse for consistency
            'max_concurrent_streams': 2,         # Reduce context switching
            'prefer_shared_memory': True,        # Fast on-chip memory
            'enable_fast_math': True,            # Trade precision for speed
            'use_tensor_cores': True             # Use Tensor Cores if available
        }
        return cv.CudaVisionCore(config=config)

    @staticmethod
    def profile_pipeline(processor, test_images):
        """Profile processing pipeline performance."""
        print("Performance Profiling Report")
        print("=" * 40)

        total_start = time.time()

        for i, image in enumerate(test_images):
            start_time = time.time()
            result = processor.enhance_image(image, mode='professional')
            end_time = time.time()

            processing_time = (end_time - start_time) * 1000
            print(f"Image {i+1}: {processing_time:.2f}ms")

            if i == 0:  # Detailed stats for first image
                stats = processor.get_performance_stats()
                print(f"  Kernel breakdown:")
                for kernel, time_ms in stats['kernel_times'].items():
                    print(f"    {kernel}: {time_ms:.2f}ms")

        total_time = time.time() - total_start
        avg_time = (total_time / len(test_images)) * 1000
        throughput = len(test_images) / total_time

        print(f"\nSummary:")
        print(f"  Total time: {total_time:.2f}s")
        print(f"  Average per image: {avg_time:.2f}ms")
        print(f"  Throughput: {throughput:.1f} FPS")

        return {
            'total_time': total_time,
            'average_time_ms': avg_time,
            'throughput_fps': throughput
        }
